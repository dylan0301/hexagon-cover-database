\section{Introduction and Proof Outline}
\label{sec:introduction}

\subsection{The theorem and the canonical labeling}

Set $O:=0\in\mathbb R^2$.  With indices in $\mathbb Z/6\mathbb Z$, choose
$V_0\in\mathbb R^2$, let $\operatorname{Rot}_{O,\theta}$ denote ordinary
rotation through $\theta$ about $O$, and define
\[
 \lVert V_0-O\rVert=1,
 \qquad
 V_i:=\operatorname{Rot}_{O,i\pi/3}(V_0).
\]
Put
\[
 \begin{gathered}
 H:=\conv\{V_0,\ldots,V_5\},\qquad
 e_{i,i+1}:=[V_i,V_{i+1}],\qquad r_i:=[O,V_i],\\
 M_i:=\frac{O+V_i}{2},
 \end{gathered}
\]
and
\[
 S:=\partial H\cup\bigcup_{i=0}^5r_i.
\]
Then
\[
 \Haus(\partial H)=6,
 \qquad
 \Haus\left(\bigcup_{i=0}^5r_i\right)=6,
 \qquad
 \Haus(S)=12.
\]

\begin{theorem}[Main theorem]
\label{thm:main}
The hexagon $H$ cannot be covered by seven open equilateral triangles of side
length one.
\end{theorem}

\begin{corollary}[Soifer's Hexagon Problem]
\label{cor:expanded-closed}
For every $L>1$, the regular hexagon $H_L:=O+L(H-O)$ cannot be covered by seven closed
unit equilateral triangles.
\end{corollary}

The seven points $O,V_0,\ldots,V_5$ are pairwise at distance at least one,
whereas any two points in an open unit equilateral triangle are at distance
strictly less than one.  Hence the seven covering triangles admit a unique
labeling
\[
 U_C,U_0,\ldots,U_5
\]
such that
\[
 O\in U_C,
 \qquad
 V_i\in U_i.
\]
We call $U_C$ the \emph{C triangle} and $U_i$ the \emph{V triangle at
$V_i$}.  Set
\[
 T_C=\overline{U_C},\qquad T_i=\overline{U_i}.
\]

\subsection{Finite classifications}

Define the closed C-triangle edge count by
\[
 N_E(T_C):=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 \Haus(T_C\cap e_{i,i+1})>0\right\}\right\rvert.
\]
The closed classifications are
\[
 \begin{aligned}
 T_C\text{ has class }\CEzero&\iff N_E(T_C)=0,\\
 T_C\text{ has class }\CEone&\iff N_E(T_C)=1,\\
 T_C\text{ has class }\CEtwo&\iff N_E(T_C)=2.
 \end{aligned}
\]
Proposition~\ref{prop:ce-classification} proves that these are the only
possibilities.

For the V triangle $T_i$, put
\[
 o(T_i):=\left\lvert\operatorname{vert}(T_i)\setminus H\right\rvert,
 \qquad
 n(T_i):=\left\lvert\left\{j\in\{i-1,i+1\}:
 \Haus(T_i\cap r_j)>0\right\}\right\rvert,
\]
with indices modulo $6$.  The raw pair $(3,0)$ says that all three vertices of
$T_i$ lie outside $H$; it does not say that $T_i$ is disjoint from $H$, since
$V_i\in\interior(T_i)\cap H$.
Proposition~\ref{prop:vd0-exact-trace-normalization} replaces every such
triangle by a same-orientation translate with exactly the same
trace in $H$, still containing $V_i$ in its interior, and with at most two
vertices outside $H$.  We make this exact-trace normalization before defining
any reach.  The normalized exhaustive types used below are
\[
\begin{array}{c|c}
(o(T_i),n(T_i))&\text{type}\\ \hline
(1,0),(2,0)&\Vdzero\\
(1,1)&\Vdone\\
(1,2)&\Vdtwo\\
(2,1)&\Tlike.
\end{array}
\]

Define the edge and radial parametrizations
\[
 X_i(s):=V_i+s(V_{i+1}-V_i),
 \qquad
 R_i(s):=V_i+s(O-V_i)
 \qquad(0\le s\le1).
\]
The actual maximal reaches are defined on the closed V triangle $T_i$ by
\[
 \begin{aligned}
 A_i&:=\max\{s\in[0,1]:V_i+s(V_{i-1}-V_i)\in T_i\},\\
 B_i&:=\max\{s\in[0,1]:X_i(s)\in T_i\},\\
 C_i&:=\max\{s\in[0,1]:R_i(s)\in T_i\}.
 \end{aligned}
\]
Equivalently, in terms of the original open V triangle,
\[
 \begin{aligned}
 A_i&=\sup\{s\in[0,1]:V_i+s(V_{i-1}-V_i)\in U_i\},\\
 B_i&=\sup\{s\in[0,1]:X_i(s)\in U_i\},\\
 C_i&=\sup\{s\in[0,1]:R_i(s)\in U_i\}.
 \end{aligned}
\]
The V triangle $T_i$ is \emph{supercritical} when $A_i+B_i>1$, and
\[
 N_+:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 A_i+B_i>1\right\}\right\rvert.
\]
Lowercase $(a_i,b_i,c_i)$ always denotes selected lower bounds satisfying
\[
 0\le a_i\le A_i,
 \qquad
 0\le b_i\le B_i,
 \qquad
 0\le c_i\le C_i;
\]
these quantities are never the maxima defining $N_+$.

The two incident open traces on $e_{i,i+1}$ are
\[
 U_i\cap e_{i,i+1}=X_i([0,B_i)),
 \qquad
 U_{i+1}\cap e_{i,i+1}=X_i((1-A_{i+1},1]).
\]
Define the boundary-gap set and the number of gap edges by
\[
 G_i:=e_{i,i+1}\setminus
 \bigl((U_i\cap e_{i,i+1})\cup(U_{i+1}\cap e_{i,i+1})\bigr),
 \qquad
 N_{\mathrm{gap}}:=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:G_i\ne\varnothing\right\}\right\rvert.
\]
Thus
\[
 G_i=
 \begin{cases}
 X_i([B_i,1-A_{i+1}]),&B_i+A_{i+1}\le1,\\
 \varnothing,&B_i+A_{i+1}>1.
 \end{cases}
\]
When equality holds in the first branch, $G_i$ is a singleton and remains
uncovered by the two incident open V triangles.

For the case tables, put
\[
 \begin{aligned}
 d&:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 (o(T_i),n(T_i))\in\{(1,1),(1,2)\}\right\}\right\rvert,\\
 t&:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 (o(T_i),n(T_i))=(2,1)\right\}\right\rvert,
 \end{aligned}
\]
and let $N_{\mathrm{sp}}:=d+t$.  The common trace calculation proves that the
number of V triangles that are either supercritical or meet an adjacent
radial segment in positive length is
\[
 N_++N_{\mathrm{sp}}.
\]
In particular, every CE1/CE2 configuration with
$N_++N_{\mathrm{sp}}\ge3$ is impossible by a trace-length bound on $S$.

\subsection{Exhaustive routing}

Lemma~\ref{lem:gap-exhaustion} proves the fundamental equivalence
\[
 N_{\mathrm{gap}}=0
 \quad\Longleftrightarrow\quad
 U_0,\ldots,U_5\text{ cover }\partial H.
\]
Thus Table~\ref{tab:routing} first dispatches every zero-gap case without
reference to the center class.  The nonzero-gap part is read from top to
bottom; the first applicable row is the unique route for a hypothetical
cover.  The full classification proof is
Proposition~\ref{prop:exhaustive-structural-reduction}.

\subsection{Organization of the proof}

Sections~\ref{sec:structure-common}--\ref{sec:finite-enclosure} give the exact
definitions and compact interfaces for the common reductions, trace-length,
area-loss, and finite-enclosure mechanisms.  Appendices A--F solve,
respectively, the structural, shared-local, and signed-center optimizations;
the trace-length optimization; the area-loss optimization; the nonzero-gap
finite-enclosure optimization; the zero-gap nine-point optimization; and the
exact polynomial positivity certificate.  Section~\ref{sec:assembly} applies
the zero-gap matrix before the nonzero-gap rows of Table~\ref{tab:routing} and
closes the proof.
