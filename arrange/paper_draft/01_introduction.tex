\section{Introduction and Proof Outline}
\label{sec:introduction}

\subsection{The theorem and the canonical labeling}

Let
\[
 O=(0,0),\qquad
 V_i=\left(\cos\frac{i\pi}{3},\sin\frac{i\pi}{3}\right),\qquad
 H=\conv\{V_0,\ldots,V_5\},
\]
where indices lie in $\mathbb Z/6\mathbb Z$.  Write
\[
 e_{i,i+1}=[V_i,V_{i+1}],\qquad r_i=[O,V_i],\qquad M_i=\frac12V_i,
\]
and put
\[
 S=\partial H\cup\bigcup_{i=0}^5r_i.
\]
Then
\begin{equation}
\begin{gathered}
 \Haus(\partial H)=6,\\
 \Haus\left(\bigcup_{i=0}^5r_i\right)=6,\\
 \Haus(S)=12.
\end{gathered}
\label{eq:target-lengths}
\end{equation}

\begin{theorem}[Main theorem]
\label{thm:main}
The hexagon $H$ cannot be covered by seven open equilateral triangles of side
length one.
\end{theorem}

\begin{corollary}[Soifer's Hexagon Problem]
\label{cor:expanded-closed}
For every $L>1$, the regular hexagon $H_L=LH$ cannot be covered by seven closed
unit equilateral triangles.
\end{corollary}

The seven points $O,V_0,\ldots,V_5$ are pairwise at distance at least one,
whereas any two points in an open unit equilateral triangle are at distance
strictly less than one.  Hence the seven covering triangles admit a unique
labeling
\[
 U_C,U_0,\ldots,U_5
\]
such that
\[
 O\in U_C,
 \qquad
 V_i\in U_i.
\]
We call $U_C$ the \emph{C triangle} and $U_i$ the \emph{V triangle at
$V_i$}.  Set
\[
 T_C=\overline{U_C},\qquad T_i=\overline{U_i}.
\]

\subsection{Finite classifications}

Set
\[
 N_E(T_C):=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 \Haus(T_C\cap e_{i,i+1})>0\right\}\right\rvert.
\]
The only possible values are $0,1,2$.  We call the corresponding C triangle
types $\CEzero,\CEone,\CEtwo$; thus
\[
 T_C\text{ has type }\CEzero\iff N_E(T_C)=0,
\]
\[
 T_C\text{ has type }\CEone\iff N_E(T_C)=1,
\]
\[
 T_C\text{ has type }\CEtwo\iff N_E(T_C)=2.
\]

For the V triangle $T_i$, let $o(T_i)$ be the number of its triangle vertices
outside $H$ and put
\[
 n(T_i)=\left\lvert\left\{j\in\{i-1,i+1\}:
 \Haus(T_i\cap r_j)>0\right\}\right\rvert,
\]
with indices modulo $6$.  The raw pair $(3,0)$ says that all three vertices of
$T_i$ lie outside $H$; it does not say that $T_i$ is disjoint from $H$, since
$V_i\in\interior(T_i)\cap H$.
Proposition~\ref{prop:vd0-exact-trace-normalization} replaces every such
triangle by a same-orientation translate with exactly the same
trace in $H$, still containing $V_i$ in its interior, and with at most two
vertices outside $H$.  We make this exact-trace normalization before defining
any reach.  The normalized exhaustive types used below are
\[
\begin{array}{c|c}
(o(T_i),n(T_i))&\text{established type}\\ \hline
(1,0)&\Vdzero\\
(2,0)&\Vdzero\\
(1,1)&\Vdone\\
(1,2)&\Vdtwo\\
(2,1)&\Tlike.
\end{array}
\]

The actual maximal reaches are
\[
 A_i:=\max\{x\in[0,1]:V_i+x(V_{i-1}-V_i)\in T_i\},
\]
\[
 B_i:=\max\{x\in[0,1]:V_i+x(V_{i+1}-V_i)\in T_i\},
\]
\[
 C_i:=\max\{x\in[0,1]:V_i+x(O-V_i)\in T_i\}.
\]
The same numbers are the suprema of the corresponding reaches in the open
triangle $U_i$.  The V triangle $T_i$ is \emph{supercritical} when
$A_i+B_i>1$, and
\begin{equation}
 N_+:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 A_i+B_i>1\right\}\right\rvert.
 \label{eq:N-plus-definition}
\end{equation}
Lowercase reaches are used only after a selection satisfying
\[
 0\le a_i\le A_i,
\]
\[
 0\le b_i\le B_i,
\]
\[
 0\le c_i\le C_i.
\]
They never define $N_+$.

Parametrize $e_{i,i+1}$ from $V_i$ to $V_{i+1}$ by
\[
 X_i(x):=V_i+x(V_{i+1}-V_i),\qquad 0\le x\le1.
\]
The incident open traces are
\[
 U_i\cap e_{i,i+1}=X_i([0,B_i)),
\]
\[
 U_{i+1}\cap e_{i,i+1}=X_i((1-A_{i+1},1]).
\]
Define the \emph{boundary gap}
\[
 J_i:=e_{i,i+1}\setminus
 \bigl((U_i\cap e_{i,i+1})\cup(U_{i+1}\cap e_{i,i+1})\bigr).
\]
The displayed trace identities give
\[
 J_i\ne\varnothing\iff B_i+A_{i+1}\le1.
\]
Equality gives a singleton gap, which remains uncovered.  Put
\[
 N_{\mathrm{gap}}:=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:J_i\ne\varnothing
 \right\}\right\rvert.
\]

For the case tables, put
\[
 d:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 (o(T_i),n(T_i))\in\{(1,1),(1,2)\}\right\}\right\rvert,
\]
\[
 t:=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 (o(T_i),n(T_i))=(2,1)\right\}\right\rvert,
\]
and
\[
 N_{\mathrm{sp}}:=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:n(T_i)>0\right\}\right\rvert
 =d+t.
\]
The common trace calculation proves that the
number of V triangles that are either supercritical or meet an adjacent
radial segment in positive length is
\[
 N_++N_{\mathrm{sp}}.
\]
In particular, every CE1/CE2 configuration with
$N_++N_{\mathrm{sp}}\ge3$ is impossible by a trace-length bound on $S$.

\subsection{Exhaustive routing}

Lemma~\ref{lem:gap-exhaustion} proves the fundamental equivalence
\[
 N_{\mathrm{gap}}=0
 \quad\Longleftrightarrow\quad
 U_0,\ldots,U_5\text{ cover }\partial H.
\]
Thus Table~\ref{tab:routing} first dispatches every zero-gap case without
reference to the center class. The entire zero-gap $N_+=1$ row uses one
nine-point theorem, with arbitrary normalized V types. The former Vd1/Vd2
length and T3-like area arguments remain independent alternative proofs.
The nonzero-gap part is read from top to
bottom; the first applicable row is the unique route for a hypothetical
cover.  The full classification proof is
Proposition~\ref{prop:exhaustive-structural-reduction}.

\subsection{Organization of the proof}

Section~\ref{sec:structure-common} proves the structural classification,
strict handoff selection, and midpoint facts.  Appendix~\ref{app:structural-shared-local-signed-center}
proves the signed-center normal form used by the later arguments.
Section~\ref{sec:trace-length} gives the complete trace-length argument.
Section~\ref{sec:area-loss} proves the local and cyclic area-loss estimates.
Section~\ref{sec:finite-enclosure} gives the finite residual-hull principle
and the nonzero-gap and zero-gap enclosure obstructions.
Section~\ref{sec:assembly} then applies the zero-gap matrix before the
nonzero-gap rows of Table~\ref{tab:routing} and closes the proof.
Appendices A--E supply the corresponding solved optimizations; Appendix F is
the exact polynomial certificate for the two mixed zero-gap intersections.
