Throughout this section indices lie in $\mathbb Z/6\mathbb Z$, and we use the
open triangles $U_C,U_0,\ldots,U_5$, their closures $T_C,T_0,\ldots,T_5$,
and the maps $X_i,R_i$ fixed in \zcref{sec:introduction}.  In particular,
all classifications and maximal reaches belong to the closed triangles,
whereas boundary coverage and endpoint strictness belong to the original
open triangles.

\subsection{Open covers and the seven distinguished roles}

\begin{proposition}[Open, closed, and scaled formulations]
\label{prop:open-closed-scaled}
The following assertions are equivalent.
\begin{enumerate}
 \item $H$ is covered by seven open equilateral triangles of side length $1$.
 \item For some $\varepsilon>0$, $H$ is covered by seven closed equilateral
 triangles of side length $1-\varepsilon$.
 \item For some $L>1$, $H_L:=O+L(H-O)$ is covered by seven closed equilateral
 triangles of side length $1$.
\end{enumerate}
\end{proposition}

\begin{proof}
The uniform inward margin needed for (1)$\Rightarrow$(2) is supplied by
\zcref{lem:app-open-closed-scaling}.  Enlarging each closed triangle about
its incenter proves (2)$\Rightarrow$(1), and dilation by
$(1-\varepsilon)^{-1}$ identifies (2) and (3).
\end{proof}

\begin{lemma}[Distinct center and vertex roles]
\label{lem:distinct-roles}
The covering triangles have a unique labeling for which
\[
 O\in U_C,
 \qquad
 V_i\in U_i\quad(0\le i\le5),
\]
and these seven triangles are distinct.  Consequently
$O\in\interior(T_C)$ and $V_i\in\interior(T_i)$.
\end{lemma}

\begin{proof}
The seven displayed points are pairwise at distance at least one, while two
points in an open unit equilateral triangle are at distance strictly less
than its diameter one.  Thus each covering triangle contains at most one of
the points, and the asserted labeling is forced.
\end{proof}

\subsection{Center and vertex classifications}

Recall the exact closed C-triangle count
\[
 \begin{gathered}
 N_E(T_C)=
 \left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 \Haus(T_C\cap e_{i,i+1})>0\right\}\right\rvert,\\
 T_C\text{ is CE}k\iff N_E(T_C)=k
 \qquad(k\in\{0,1,2\}).
 \end{gathered}
\]
A point contact has zero trace length and therefore does not create a second
positive trace.

\begin{proposition}[Exhaustive center classes]
\label{prop:ce-classification}
Every C triangle has exactly one of the classes $\CEzero,\CEone,\CEtwo$;
equivalently, $N_E(T_C)\le2$.  If two edge traces are positive, their edges
are adjacent.
\end{proposition}

\begin{proof}
The two nonadjacent-edge distance calculations in
\zcref{lem:app-center-v-incidence} show that a unit-diameter triangle cannot
have positive traces on three hexagon edges.
\end{proof}

For a closed V triangle put
\[
 o(T_i):=\left\lvert\operatorname{vert}(T_i)\setminus H\right\rvert,
 \qquad
 n(T_i):=\left\lvert\left\{j\in\{i-1,i+1\}:
 \Haus(T_i\cap r_j)>0\right\}\right\rvert.
\]

\begin{lemma}[Local wedge incidence]
\label{lem:local-wedge}
Every original V triangle satisfies $1\le o(T_i)\le3$.  If it has
positive-length intersection with one of $r_{i-1},r_{i+1}$, at least one of
its vertices lies in $H$; if it has positive-length intersection with both,
at least two of its vertices lie in $H$.
\end{lemma}

\begin{proof}
This is the wedge-feasibility calculation in
\zcref{lem:app-center-v-incidence}.
\end{proof}

\begin{proposition}[Exact-trace normalization of raw $(3,0)$ roles]
\label{prop:vd0-exact-trace-normalization}
Let $U$ be an original open $V_i$-triangle, put $T=\overline U$, and suppose
$(o(T),n(T))=(3,0)$.  There is a translate $\widehat U$ of $U$, with the
same orientation and side length and with closure $\widehat T$, such that
\[
 V_i\in\widehat U,
 \qquad
 (o(\widehat T),n(\widehat T))=(2,0),
\]
and
\[
 T\cap H=\widehat T\cap H,
 \qquad
 U\cap H=\widehat U\cap H.
\]
Thus all three actual reaches and the inequality $A_i+B_i>1$ are unchanged.
The normalization may be performed simultaneously for every raw $(3,0)$
role.
\end{proposition}

\begin{proof}
The same-orientation translation and its exact open- and closed-trace
verification are supplied by \zcref{lem:app-exact-trace-normalizations}.
Rolewise equality on $H$ preserves the open cover, so the translations may
be made simultaneously.
\end{proof}

After this normalization, the exact type dictionary is
\[
 \begin{array}{c|c}
 (o(T_i),n(T_i))&\text{type}\\ \hline
 (1,0),(2,0)&\Vdzero\\
 (1,1)&\Vdone\\
 (1,2)&\Vdtwo\\
 (2,1)&\Tlike.
 \end{array}
\]

\begin{proposition}[Exhaustive normalized vertex classes]
\label{prop:vertex-classification}
Every normalized V triangle has exactly one of the four types
$\Vdzero,\Vdone,\Vdtwo,\Tlike$.
\end{proposition}

\begin{proof}
We have $n(T_i)\in\{0,1,2\}$.  By \zcref{lem:local-wedge}, the possible
raw pairs are
\[
 (1,0),(2,0),(3,0),(1,1),(2,1),(1,2).
\]
The normalization removes exactly $(3,0)$ and preserves every other pair.
\end{proof}

\begin{proposition}[T3-like closed-trace majorant]
\label{prop:t3-translation}
Let $T$ be the closure of an original T3-like $V_i$-triangle.  There is a
translate $\widehat T$ of $T$, with the same orientation and side length,
such that
\[
 V_i\in\operatorname{relint}(\text{a side of }\widehat T),
 \qquad
 T\cap H\subsetneq\widehat T\cap H,
\]
and $(o(\widehat T),n(\widehat T))=(2,1)$.  Every old open interior point in
$H\setminus\{V_i\}$ remains interior to $\widehat T$.  Since $V_i$ becomes
a boundary point, $\widehat T$ is only a closed-trace majorant; it is not a
replacement open triangle.
\end{proposition}

\begin{proof}
The exhaustive Type-I/Type-II calculation and the strict-interior check are
given in \zcref{lem:app-exact-trace-normalizations}.
\end{proof}

\subsection{Actual reaches, open traces, and boundary gaps}

The maximal reaches are the closed-triangle maxima
\[
 \begin{aligned}
 A_i&=\max\{s\in[0,1]:X_{i-1}(1-s)\in T_i\},\\
 B_i&=\max\{s\in[0,1]:X_i(s)\in T_i\},\\
 C_i&=\max\{s\in[0,1]:R_i(s)\in T_i\},
 \end{aligned}
\]
and are equally the corresponding suprema with $T_i$ replaced by $U_i$.
Whenever selected demands occur they satisfy, explicitly,
\[
 0\le a_i\le A_i,
 \qquad
 0\le b_i\le B_i,
 \qquad
 0\le c_i\le C_i.
\]
In particular,
\[
 N_+=\left\lvert\left\{i\in\mathbb Z/6\mathbb Z:
 A_i+B_i>1\right\}\right\rvert
\]
is defined only from actual maximal reaches.

\begin{lemma}[Initial reach segments]
\label{lem:reach-initial-segments}
For every normalized V triangle,
\[
 \begin{array}{c|c|c}
 &T_i&U_i\\ \hline
 e_{i-1,i}&X_{i-1}([1-A_i,1])&X_{i-1}((1-A_i,1])\\
 e_{i,i+1}&X_i([0,B_i])&X_i([0,B_i))\\
 r_i&R_i([0,C_i])&R_i([0,C_i)).
 \end{array}
\]
Here each table entry denotes the intersection of the triangle heading its
column with the segment heading its row.
\end{lemma}

\begin{proof}
The convex initial-segment argument, including the open outer endpoint, is
part of \zcref{lem:app-exact-trace-normalizations}.
\end{proof}

The exact gap set is
\[
 G_i:=e_{i,i+1}\setminus
 \bigl((U_i\cap e_{i,i+1})\cup(U_{i+1}\cap e_{i,i+1})\bigr),
 \qquad
 N_{\mathrm{gap}}:=
 \left\lvert\{i\in\mathbb Z/6\mathbb Z:G_i\ne\varnothing\}\right\rvert.
\]

\begin{lemma}[Exhaustiveness of the gap split]
\label{lem:gap-exhaustion}
For every edge,
\[
 G_i=
 \begin{cases}
 X_i([B_i,1-A_{i+1}]),&B_i+A_{i+1}\le1,\\
 \varnothing,&B_i+A_{i+1}>1.
 \end{cases}
\]
Equality in the first branch is a singleton gap.  Every nonempty $G_i$ lies
in a positive-length trace of $U_C$.  Hence $\CEzero,\CEone,\CEtwo$ permit,
respectively, at most $0,1,2$ gap edges, and under a hypothetical cover
\[
 N_{\mathrm{gap}}=0
 \quad\Longleftrightarrow\quad
 \partial H\subseteq\bigcup_{i=0}^5U_i.
\]
\end{lemma}

\begin{proof}
The exact incident trace identities and exclusion of nonincident V triangles
are proved in \zcref{lem:app-exact-trace-normalizations}.  Thus the displayed
set difference is precisely the asserted closed interval, including its
singleton endpoint case.  A gap point must then lie in the open C triangle;
openness supplies a positive relative interval on that edge.  Distinct gap
edges are therefore counted by $N_E(T_C)$, and the conclusions follow from
\zcref{prop:ce-classification}.
\end{proof}

\begin{lemma}[T3-like roles are nonsupercritical]
\label{lem:t3-nonsupercritical}
Every original T3-like V triangle satisfies
\begin{equation}
 A_i+B_i\le1.
 \label{eq:t3-nonsupercritical}
\end{equation}
In particular, it is not counted by $N_+$.
\end{lemma}

\begin{proof}
This is the exact Type-II reach consequence of
\zcref{lem:app-exact-trace-normalizations}.
\end{proof}

\subsection{Strict handoffs and midpoint forcing}

\begin{proposition}[Strict boundary handoffs]
\label{prop:strict-handoffs}
Assume $\partial H\subseteq\bigcup_{i=0}^5U_i$.  There are simultaneous
choices
\[
 \xi_i\in(1-A_{i+1},B_i),
 \qquad
 a_i:=1-\xi_{i-1}<A_i,
 \qquad
 b_i:=\xi_i<B_i,
\]
for which $0<a_i$, $0<b_i$, and
\[
 X_{i-1}(1-a_i),X_i(b_i)\in U_i,
 \qquad
 a_i^2+a_ib_i+b_i^2<1.
\]
Moreover, if $N_+=1$, every such choice satisfies
\[
 \left\lvert\{i:a_i+b_i>1\}\right\rvert=1
\]
with the same unique index; if $N_+\ge2$, some simultaneous choice satisfies
\[
 \left\lvert\{i:a_i+b_i>1\}\right\rvert\ge2.
\]
\end{proposition}

\begin{proof}
The interval feasibility, cyclic selector, and preservation assertions are
proved in \zcref{lem:app-strict-handoff-selection}.  Thus every selected
handoff is strict.
\end{proof}

\begin{lemma}[Self-midpoint obstruction]
\label{lem:self-midpoint}
For $0\le a,b\le1$, if a closed unit equilateral triangle contains
\[
 \left\{V_i,X_{i-1}(1-a),X_i(b),M_i\right\},
\]
then $a+b\le1$.  Consequently,
\[
 M_i\in T_i\quad\Longrightarrow\quad A_i+B_i\le1.
\]
If the four displayed points lie in an open unit equilateral triangle, the
conclusion is strict.
\end{lemma}

\begin{proof}
The four finite calipers and the endpoint-strictness argument are supplied by
\zcref{lem:app-midpoint-caliper}.
\end{proof}

\begin{proposition}[CE1/CE2 exactly-one-midpoint property]
\label{prop:unique-center-midpoint}
If $O\in\interior(T_C)$ and $N_E(T_C)\in\{1,2\}$, then
\[
 \left\lvert T_C\cap\{M_0,\ldots,M_5\}\right\rvert=1.
\]
\end{proposition}

\begin{proof}
After normalizing a positive trace by a hexagon symmetry, the signed-center
optimization in \zcref{prop:app-signed-center-optimization} gives the midpoint set
$\{M_0\}$.  Undoing the symmetry proves the assertion.
\end{proof}

\begin{remark}
The interior hypothesis is essential: $\conv\{O,V_0,V_1\}$ has a boundary
edge overlap and contains both $M_0$ and $M_1$.
\end{remark}

\begin{proposition}[Exhaustive structural reduction]
\label{prop:exhaustive-structural-reduction}
Every hypothetical cover by seven open unit equilateral triangles admits the
distinct center and V triangles used in \zcref{tab:routing}, and its center
class, normalized V classes, actual $N_+$, and exact gap sets determine
exactly one row of that table.
\end{proposition}

\begin{proof}
The role labeling is \zcref{lem:distinct-roles}; the two classifications are
\zcref{prop:ce-classification,prop:vertex-classification}.  The actual
reaches determine $N_+$ and the open traces determine $N_{\mathrm{gap}}$.

If $N_{\mathrm{gap}}=0$, \zcref{lem:gap-exhaustion} gives the five
zero-gap alternatives in \zcref{tab:routing}, split by $N_+$ and then by
$(d,t)$.  If $N_{\mathrm{gap}}\ge1$, the same lemma forces
$N_E(T_C)\in\{1,2\}$, and \zcref{prop:unique-center-midpoint} gives the
unique radial midpoint.  When $N_+\ge2$,
\zcref{lem:positive-support-rescuer} gives $N_{\mathrm{sp}}\ge1$, so
\zcref{thm:common-skeleton-count} applies.  For $N_+\in\{0,1\}$ the same
theorem removes $N_++d+t\ge3$.  The remaining formula-only alternatives are
exactly the nonzero-gap rows of \zcref{tab:routing}; they are disjoint and
exhaustive.
\end{proof}
