\subsection{Structural classification}

\begin{proposition}[Structural classification]
\label{prop:body-structural-reduction}
Every hypothetical cover has the canonical labeling described in
Section~\ref{sec:introduction}.  The two classification conclusions are
\[
 \text{center type}\in\{\CEzero,\CEone,\CEtwo\},
 \qquad
 \text{vertex type}\in\{\Vdzero,\Vdone,\Vdtwo,\Tlike\}.
\]
These types, the maximal reaches, and the boundary-gap data determine exactly
one row of Table~\ref{tab:routing}.

In the CE1/CE2 case the C triangle contains exactly one radial midpoint.
If the six V triangles alone cover the boundary, strict handoff points may
be selected while preserving whether the maximal-reach supercritical pattern
has none, exactly one, or at least two indices.
\end{proposition}

\begin{proof}
Distinctness is Lemma~\ref{lem:distinct-roles}; the center and vertex
classifications are Propositions~\ref{prop:ce-classification} and
\ref{prop:vertex-classification}; the unique midpoint is
Proposition~\ref{prop:unique-center-midpoint}; the count
$N_++N_{\mathrm{sp}}$ is Lemma~\ref{lem:short-role-count}; strict handoff
selection is Proposition~\ref{prop:strict-handoffs}; and exhaustiveness is
Proposition~\ref{prop:exhaustive-structural-reduction}.
\end{proof}

\subsection{Reach coordinates and the local admissible set}

\begin{definition}[Maximal reaches and selected lower bounds]
\label{def:body-vtriangle-coordinates}
All indices lie in $\mathbb Z/6\mathbb Z$.  For the closed V triangle
$T_i$, define
\[
 \begin{aligned}
 A_i&=\sup_{\substack{t\in[0,1]\\
 V_i+t(V_{i-1}-V_i)\in T_i}}t,\\
 B_i&=\sup_{\substack{t\in[0,1]\\
 V_i+t(V_{i+1}-V_i)\in T_i}}t,\\
 C_i&=\sup_{\substack{t\in[0,1]\\
 V_i+t(O-V_i)\in T_i}}t.
 \end{aligned}
\]
Passing from $U_i$ to its closure does not change these suprema.  A
\emph{selected reach triple} is any $(a_i,b_i,c_i)\in[0,1]^3$ satisfying
\[
 (a_i,b_i,c_i)\le(A_i,B_i,C_i)
\]
coordinatewise.  The V triangle $T_i$ is subcritical, critical, or
strictly supercritical according as
\[
 A_i+B_i<1,
 \qquad
 A_i+B_i=1,
 \qquad
 A_i+B_i>1,
\]
respectively.  Thus
\[
 N_+=\left\lvert
 \left\{i\in\mathbb Z/6\mathbb Z:A_i+B_i>1\right\}
 \right\rvert
\]
is defined from maximal reaches, never from selected lower bounds.
\end{definition}

\begin{definition}[Anchor hull and admissible set]
\label{def:body-local-admissible-set}
Let
\[
 u=\left(\frac12,\frac{\sqrt3}{2}\right),
 \qquad
 v=\left(\frac12,-\frac{\sqrt3}{2}\right).
\]
For $(a,b,c)\in[0,1]^3$, define the \emph{anchor hull}
\[
 K(a,b,c)=\conv\{0,au,bv,c(u+v)\}.
\]
Define
\[
 \mathcal A=
 \left\{(a,b,c)\in[0,1]^3:
 \begin{array}{c}
 K(a,b,c)\subseteq\Delta\text{ for some closed}\cr
 \text{unit equilateral triangle }\Delta
 \end{array}
 \right\}.
\]
Thus every selected reach triple for a V triangle belongs to
$\mathcal A$.
\end{definition}

\begin{definition}[Equilateral enclosure gauge]
\label{def:body-enclosure-gauge}
Let $\mathsf R$ be counterclockwise rotation through $2\pi/3$.  For every
nonempty compact $K\subset\mathbb R^2$, define
\[
 h_K(n)=\max_{z\in K}\langle z,n\rangle,
 \qquad
 \Lambda(K)=\frac2{\sqrt3}
 \min_{\lVert n\rVert=1}\sum_{j=0}^2h_K(\mathsf R^jn).
\]
\end{definition}
Proposition~\ref{prop:universal-enclosure-gauge} proves that $\Lambda(K)$ is
the least side length of a closed equilateral triangle containing $K$.

\begin{proposition}[Structure of the admissible set]
\label{prop:body-admissible-description}
The set $\mathcal A$ is compact, coordinatewise downward closed, and invariant
under interchange of $a$ and $b$.  Moreover,
\[
 (a,b,c)\in\mathcal A
 \quad\Longleftrightarrow\quad
 \Lambda(K(a,b,c))\le1.
\]
Every one-coordinate fiber of $\mathcal A$ is a nonempty compact initial
interval.  Consequently every maximum used below is attained.
\end{proposition}

\begin{proof}
Compactness and downward closure follow from the containment definition and
compactness of the oriented unit-triangle family modulo translation.
Reflection in the radial axis exchanges $a$ and $b$.  The gauge equivalence is
Proposition~\ref{prop:universal-enclosure-gauge}.  Compactness and downward
closure then give the initial-interval fibers.  The explicit support formulas
and cell conditions are proved in
Appendix~\ref{sec:admissible-set-derivation}.
\end{proof}

\subsection{Signed center coordinates}

For $t\in\mathbb R$, write $[t]_+=\max\{t,0\}$, and for a closed triangle $T$
put
\[
 L_{\partial H}(T)=\Haus(T\cap\partial H).
\]

\begin{definition}[Signed center coordinates]
\label{def:body-signed-center-data}
Normalize a positive center trace to $e_{0,1}$.  There are parameters
\[
 0<R<1,
 \qquad
 W=1-R,
 \qquad
 E=\sqrt{1-RW},
 \qquad
 \eta=1-E,
 \qquad
 P=E(1-E),
\]
and positive slacks $\alpha,\delta$ such that, with
\[
 k=\eta+\alpha+\delta,
 \qquad
 \Delta_R=P-\alpha-W\delta>0,
 \qquad
 \Delta_L=P-R\alpha-\delta,
\]
the normalized center traces are
\[
 T_C\cap e_{0,1}=\left[\frac{k}{R},W+\delta\right],
 \qquad
 \Haus(T_C\cap e_{0,1})=\frac{\Delta_R}{R},
\]
and, when $\Delta_L>0$,
\[
 T_C\cap e_{5,0}=\left[\frac{k}{W},R+\alpha\right],
 \qquad
 \Haus(T_C\cap e_{5,0})=\frac{\Delta_L}{W}.
\]
The six center exits, measured from $O$ toward $V_i$, are
\[
 \begin{aligned}
 d_0^C&=E-\alpha-\delta,&
 d_1^C&=\frac{\delta}{R},&
 d_2^C&=\delta,\\
 d_3^C&=\min\left\{\frac{\alpha}{R},\frac{\delta}{W}\right\},&
 d_4^C&=\alpha,&
 d_5^C&=\frac{\alpha}{W}.
 \end{aligned}
\]
The complementary radial lower bound for $T_i$ is
\[
 c_i^C=1-d_i^C.
\]
The signed parameter domain is
\[
 \mathcal D_{\rm sgn}=
 \left\{(R,\alpha,\delta):0<R<1,\ \alpha>0,\ \delta>0,\ \Delta_R>0\right\}.
\]
Its CE1 and CE2 parts are
\[
 \mathcal D_{\rm C1}=\{\Delta_L\le0\},
 \qquad
 \mathcal D_{\rm C2}=\{\Delta_L>0\},
\]
with the ambient condition $(R,\alpha,\delta)\in\mathcal D_{\rm sgn}$
understood.
\end{definition}

\begin{proposition}[Center trace and radial data]
\label{prop:body-signed-center-interface}
The C triangle is CE1 exactly when $\Delta_L\le0$, and CE2 exactly when
$\Delta_L>0$; equality is a single contact point and hence belongs to CE1.
Furthermore,
\[
 L_{\partial H}(T_C)
 =\frac{\Delta_R}{R}+\frac{[\Delta_L]_+}{W},
\]
and
\[
 L_{\partial H}(T_C)\le\frac{\sqrt3}{2}-\frac34
 \quad\text{in CE1},
 \qquad
 L_{\partial H}(T_C)<\frac12
 \quad\text{in CE2}.
\]
\end{proposition}

\begin{proof}
These statements are Proposition~\ref{prop:signed-center-normal-form} and
Corollary~\ref{cor:signed-center-boundary}.
\end{proof}
