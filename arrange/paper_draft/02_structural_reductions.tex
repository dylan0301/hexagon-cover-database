\subsection{Module A.1: open--closed shrinking feasibility}

\paragraph{Exact interface.}
The three cover formulations are those of
\zcref{prop:open-closed-scaled}.

\paragraph{Variables and feasible set.}
Let $U_1,\ldots,U_7$ be open unit equilateral triangles covering the compact
hexagon $H$.  For $p\in H$ define
\[
 m_j(p):=\operatorname{dist}(p,\mathbb R^2\setminus U_j),
 \qquad
 m(p):=\max_{1\le j\le7}m_j(p).
\]

\paragraph{Objective.}
Find a single inward displacement that may be applied to all twenty-one side
lines while retaining a cover of $H$.

\paragraph{Exact result.}

\begin{lemma}[Uniform open--closed shrinking]
\label{lem:app-open-closed-scaling}
There is $\rho>0$ such that moving every side of every $U_j$ inward through
distance $\rho$ produces seven closed equilateral triangles covering $H$,
each of side length $1-2\sqrt3\rho>0$.
\end{lemma}

\paragraph{Solution.}

\begin{proof}
Each $m_j$ is continuous, hence so is $m$.  The open cover makes $m$ positive
on $H$, and compactness gives
\[
 \eta:=\min_{p\in H}m(p)>0.
\]
Choose
\[
 0<\rho<\min\left\{\eta,\frac{\sqrt3}{6}\right\}.
\]
For every $p\in H$ some $j$ satisfies $m_j(p)\ge\eta>\rho$, so $p$ remains
in the closed triangle obtained from $U_j$ by the inward displacement.  The
inradius of an equilateral triangle of side $s$ is $s\sqrt3/6$; moving all
three sides inward by $\rho$ therefore decreases the side by
$2\sqrt3\rho$.  The choice of $\rho$ keeps the resulting side positive.
\end{proof}

\paragraph{Body consequence.}
This is the quantitative implication used in
\zcref{prop:open-closed-scaled}.

\subsection{Module A.2: center and V-triangle incidence feasibility}

\paragraph{Exact interface.}
The C-triangle count $N_E(T_C)$ and the V-triangle incidence pair
$(o(T_i),n(T_i))$ are defined in \zcref{sec:introduction,sec:structure-common}.

\paragraph{Variables and feasible set.}
For the center calculation take relative-interior parameters $0<s,t<1$ on
two nonadjacent edges.  For the V calculation normalize to $i=0$ and use the
affine wedge chart
\[
 X=V_0+x(V_1-V_0)+y(V_5-V_0),
\]
with $V_0$ as the chart origin.  Its squared Euclidean norm is
\begin{equation}
 \lVert(x,y)\rVert^2=x^2+y^2-xy
 =\left(y-\frac x2\right)^2+\frac{3x^2}{4}
 =(x-y)^2+xy.
 \label{eq:app-wedge-norm}
\end{equation}
Within the closed unit ball about $V_0$, the part of $H$ is exactly
\[
 W:=\{(x,y):x\ge0,\ y\ge0\},
\]
and the adjacent radial segments are
\[
 r_1=\{(1,y):0\le y\le1\},
 \qquad
 r_5=\{(x,1):0\le x\le1\}.
\]

\paragraph{Objective.}
Determine the maximum number of positive C-triangle edge incidences and the
minimum number of V-triangle vertices forced into the local wedge by one or
two positive radial incidences.

\paragraph{Exact result.}

\begin{lemma}[Center and V-triangle incidence calculation]
\label{lem:app-center-v-incidence}
A closed unit equilateral triangle with $O$ in its interior has positive
trace on at most two edges of $H$, and any two such edges are adjacent.  An original V triangle has
$1\le o(T_i)\le3$; one positive adjacent radial trace forces at least one of
its vertices into $H$, and two such traces force at least two vertices into
$H$.
\end{lemma}

\paragraph{Solution.}

\begin{proof}
For the C triangle, among any three edges of the six-cycle two are
nonadjacent.  By symmetry take
\[
 x=(1-t)V_0+tV_1.
\]
If $y=(1-s)V_2+sV_3$, direct calculation in the global chart gives
\[
 \lVert x-y\rVert^2-1=(1-t)(2-t)+s(s+t)>0.
\]
If instead $y=(1-s)V_3+sV_4$ and $u=t+s$, then
\[
 \lVert x-y\rVert^2-1=(u-1)^2+2>0.
\]
Thus a unit-diameter triangle cannot contain relative-interior points of two
nonadjacent edges, proving the center assertion.

For the V triangle, \eqref{eq:app-wedge-norm} proves the asserted description
of $H$ inside the unit ball.  Suppose $T_0\cap r_1$ has positive length and
let $m$ be the largest $x$-coordinate of a vertex of $T_0$.  If $m>1$, a
maximizing vertex $(m,y)$ lies in the unit ball, whence $0<y<1$ and that
vertex lies in $H$.  If $m=1$, positive-length contact with the support line
$x=1$ forces a side of $T_0$ to lie there; both endpoints satisfy
$1+y^2-y\le1$ and hence lie in $H$.  The same argument applies to $r_5$.

If both $r_1$ and $r_5$ are met but only one triangle vertex lies in $H$,
that vertex must have $x>1$ and $y>1$, contradicting
$(x-y)^2+xy>1$.  Finally, if all three vertices belonged to the convex set
$H$, then $T_0\subseteq H$, which is impossible because the extreme point
$V_0$ lies in the interior of $T_0$.  This also proves $o(T_0)\ge1$.
\end{proof}

\paragraph{Body consequence.}
The first conclusion supplies \zcref{prop:ce-classification}; the remaining
conclusions supply \zcref{lem:local-wedge} and the finite vertex-type list.

\subsection{Module A.3: exact-trace normalization and interval geometry}

\paragraph{Exact interface.}
The input is an original V triangle $U_i$, its closure $T_i$, the incidence
pair $(o(T_i),n(T_i))$, and the actual reaches $A_i,B_i,C_i$ defined in
\zcref{sec:introduction}.  The output must preserve the exact open and closed
traces whenever a role is normalized.

\paragraph{Variables and feasible set.}
Normalize to $i=0$ in the wedge chart of Module A.2.  Put
\[
 D_t:=1-t+t^2,
 \qquad
 z:=\sqrt{D_t},
 \qquad
 0\le t\le1.
\]
After relabeling the triangle vertices, every orientation is in one of the
two families
\[
 \begin{array}{c|cc}
 &\text{first side vector}&\text{second side vector}\\ \hline
 \mathrm I&z^{-1}(1,t)&z^{-1}(1-t,1)\\
 \mathrm {II}&z^{-1}(t,-(1-t))&z^{-1}(1,t).
 \end{array}
\]
For Type I set
\begin{equation}
 U=\alpha+y-tx,
 \qquad
 V=\beta+x-(1-t)y,
 \qquad
 T=\{U\ge0,V\ge0,U+V\le z\},
 \label{eq:app-orientation-I}
\end{equation}
and for Type II set
\begin{equation}
 U=\alpha+(1-t)x+ty,
 \qquad
 V=\beta+tx-y,
 \qquad
 T=\{U\ge0,V\ge0,U+V\le z\}.
 \label{eq:app-orientation-II}
\end{equation}
In both cases
\begin{equation}
 V_0\in\interior(T)
 \iff
 \alpha>0,
 \quad
 \beta>0,
 \quad
 \alpha+\beta<z.
 \label{eq:app-orientation-interior}
\end{equation}
The Type-II vertex in the wedge is
\begin{equation}
 Q=\left(
 \frac{z-\alpha-t\beta}{D_t},
 \frac{t(z-\alpha)+(1-t)\beta}{D_t}
 \right),
 \label{eq:app-orientation-II-Q}
\end{equation}
whose coordinates are positive on the feasible set.

\paragraph{Objective.}
For a raw $(3,0)$ role, find a same-orientation translation with identical
trace in $H$ and type $(2,0)$.  For a T3-like role, find the exact
closed-trace majorant and determine its actual boundary-reach sum.  Finally,
determine all open and closed initial reach intervals and their exact gap set.

\paragraph{Exact result.}

\begin{lemma}[Exact-trace translations and initial intervals]
\label{lem:app-exact-trace-normalizations}
The following hold.
\begin{enumerate}
 \item Every raw $(3,0)$ role admits a same-orientation translate
 $\widehat U$ for which
 \[
  V_i\in\widehat U,
  \quad (o(\overline{\widehat U}),n(\overline{\widehat U}))=(2,0),
  \quad U\cap H=\widehat U\cap H,
  \quad \overline U\cap H=\overline{\widehat U}\cap H.
 \]
 \item Every T3-like role has Type II orientation and admits the
 closed-trace majorant of \zcref{prop:t3-translation}.  Its exact reaches
 satisfy $A_i+B_i\le1$.
 \item For every V triangle,
 \[
  \begin{array}{c|c|c}
  &T_i&U_i\\ \hline
  e_{i-1,i}&X_{i-1}([1-A_i,1])&X_{i-1}((1-A_i,1])\\
  e_{i,i+1}&X_i([0,B_i])&X_i([0,B_i))\\
  r_i&R_i([0,C_i])&R_i([0,C_i)).
  \end{array}
 \]
 Consequently
 \[
 G_i=
 \begin{cases}
 X_i([B_i,1-A_{i+1}]),&B_i+A_{i+1}\le1,\\
 \varnothing,&B_i+A_{i+1}>1,
 \end{cases}
 \]
 including a singleton in the equality case.
\end{enumerate}
\end{lemma}

\paragraph{Solution.}

\begin{proof}
For a raw $(3,0)$ role, the Type-II vertex
\eqref{eq:app-orientation-II-Q} lies in the wedge; since $T$ contains the
origin and has diameter one, \zcref{lem:app-center-v-incidence} puts it in
$H$.  Thus only Type I is feasible.  Its vertices are
\[
 Q_0=\left(
 -\frac{(1-t)\alpha+\beta}{D_t},
 -\frac{\alpha+t\beta}{D_t}
 \right),
\]
\begin{equation}
 \begin{aligned}
 Q_1&=\left(
 \frac{z-(1-t)\alpha-\beta}{D_t},
 \frac{tz-\alpha-t\beta}{D_t}
 \right),\\
 Q_2&=\left(
 \frac{(1-t)z-(1-t)\alpha-\beta}{D_t},
 \frac{z-\alpha-t\beta}{D_t}
 \right).
 \end{aligned}
 \label{eq:app-o3-vertices}
\end{equation}
The vertex $Q_0$ has two negative coordinates, the first coordinate of
$Q_1$ is positive, and the second coordinate of $Q_2$ is positive.  Hence
$o=3$ is equivalent to
\begin{equation}
 \alpha+t\beta>tz,
 \qquad
 (1-t)\alpha+\beta>(1-t)z,
 \label{eq:app-o3-outside}
\end{equation}
which forces $0<t<1$.  Put
\[
 s:=\alpha+\beta,
 \qquad
 L:=z-s>0.
\]
Then \eqref{eq:app-o3-outside} becomes
\begin{equation}
 \alpha>\frac{tL}{1-t},
 \qquad
 \beta>\frac{(1-t)L}{t}.
 \label{eq:app-o3-redundancy}
\end{equation}
On the wedge cut
\[
 (1-t)x+ty\le L,
\]
the minimum of $U$ occurs at $(L/(1-t),0)$ and that of $V$ occurs at
$(0,L/t)$.  Therefore \eqref{eq:app-o3-redundancy} makes both inequalities
strictly redundant there and
\begin{equation}
 T\cap W=\{(x,y)\in W:(1-t)x+ty\le L\}.
 \label{eq:app-o3-wedge-trace}
\end{equation}

Keep $t,s$ fixed and replace the affine constants by
\begin{equation}
 \widehat\alpha:=\frac{tL}{1-t},
 \qquad
 \widehat\beta:=s-\widehat\alpha.
 \label{eq:app-o3-translation}
\end{equation}
Then $0<\widehat\alpha<\alpha$, $\widehat\beta>\beta>0$, and
$\widehat\alpha+\widehat\beta=s<z$, so this is a same-orientation
translation retaining $V_0$ in its interior.  Its third side still gives
the cut in \eqref{eq:app-o3-wedge-trace}; the new $\widehat U$ inequality
has equality only at the already-boundary point $(L/(1-t),0)$ and the new
$\widehat V$ inequality remains strict.  Thus both the closed and open
traces on $W$, hence on $H$, are unchanged.  Formula
\eqref{eq:app-o3-translation} puts $Q_1$ on $H$, leaves $Q_0$ outside, and
keeps $Q_2$ outside by the second inequality in
\eqref{eq:app-o3-redundancy}; the new pair is therefore $(2,0)$.  This proves
part 1.

For part 2, direct inversion of the Type-I vertices shows that when exactly
two vertices are outside $W$, the sole wedge vertex has both coordinates
strictly below one: in the two possible cases the relevant bounds are
$(1-t)/z$ and $t/z$, with strictness from $\alpha>0$ or $\beta>0$.
The supporting-line argument in Module A.2 then excludes positive contact
with $x=1$ or $y=1$.  A T3-like role must therefore have Type II orientation.
Reflect in $x=y$ if needed so that $T_0\cap r_1$ has positive length.  Its
outside vertices have respectively negative $x$- and $y$-coordinates, its
wedge vertex is \eqref{eq:app-orientation-II-Q}, and $0<t<1$.  The exact axis
reaches are
\begin{equation}
 A=\beta,
 \qquad
 B=z-\alpha-\beta,
 \label{eq:app-t3-reaches}
\end{equation}
while the T3-like incidence is equivalent to $Q_x>1$ and $Q_y\le1$.

Replace $\alpha$ by $0$ and retain $t,\beta$.  At the origin the new slacks
are $\widehat U=0$ and $0<\widehat V=\beta<z$, placing $V_0$ in the relative
interior of a side.  For $(x,y)\in W$,
\[
 \widehat U=(1-t)x+ty\ge0,
 \qquad
 \widehat V=V,
 \qquad
 \widehat U+\widehat V=U+V-\alpha.
\]
Hence $T\cap W\subset\widehat T\cap W$; the old $x$-axis reach is $B$ and
the new one is $B+\alpha$, so inclusion is strict in $H$.  The two outside
vertices remain outside and $\widehat Q_x>Q_x>1$.  From $Q_x>1$ one obtains
$t\beta<z-\alpha-D_t$, and therefore
\[
 tD_t(1-\widehat Q_y)
 >D_t(1-z)+(1-t)\alpha>0.
\]
Thus $\widehat Q_y<1$, so the translated role remains of type $(2,1)$.
Every nonzero point of $W$ has $\widehat U>0$, which proves the asserted
strict-interior preservation.  Finally, \eqref{eq:app-t3-reaches} gives
\[
 A+B=z-\alpha\le1.
\]

For part 3, intersect the convex set $T_i$ with each of the three segments
issuing from its interior point $V_i$.  The result is a nonempty compact
initial interval, and the defining maximum is its outer endpoint.  Each
maximum is strictly below one: an interior point of a unit-diameter triangle
is at distance strictly below one from every point of its closure.  Every
point before the outer endpoint is a strict convex combination of $V_i$ and
a point of $T_i$, hence is in $U_i=\interior(T_i)$; if the outer endpoint
were in $U_i$, openness would extend the interval past it.  This proves all
six interval identities and the equality of the closed maxima with the open
suprema.

It remains only to exclude nonincident V triangles from the relative
interior of $e_{i,i+1}$.  Normalize to $e_{0,1}$ and write
$x=X_0(s)$ with $0<s<1$.  For the nearest nonincident vertices one finds
\[
 \lVert x-V_2\rVert^2=1+(1-s)(2-s)>1,
 \qquad
 \lVert x-V_5\rVert^2=1+s+s^2>1;
\]
all other cases follow by symmetry or are farther away.  Since the
distinguished vertex of a V triangle lies in its open interior, its triangle
cannot also contain such an $x$.  Subtracting the two incident open initial
intervals from the edge now gives the displayed formula for $G_i$, with the
closed singleton retained when their two missing endpoints agree.
\end{proof}

\paragraph{Body consequence.}
Part 1 proves \zcref{prop:vd0-exact-trace-normalization}.  Part 2 proves
\zcref{prop:t3-translation,lem:t3-nonsupercritical}; part 3 proves
\zcref{lem:reach-initial-segments,lem:gap-exhaustion}.
