\section{Universal Enclosure and Propagation Calculus}
\label{sec:universal-calculus}

This section collects the geometry shared by the propagation and direct-witness
arguments.  The same equilateral enclosure functional governs the local
admissible set of Method~2 and the forced witness set of Method~4.  The
transfer statements below distinguish three situations that must not be
conflated: a center-free edge, an edge with a specified center interval, and a
free strict-supercritical outgoing envelope.

\subsection{The equilateral enclosure gauge}

Let $\mathsf R$ denote counterclockwise rotation through $2\pi/3$.  For a
nonempty compact set $K\subset\mathbb R^2$, put
\[
 h_K(n)=\max_{x\in K}\langle x,n\rangle
\]
and
\begin{equation}
 \Lambda(K)=\frac2{\sqrt3}
 \min_{\lVert n\rVert=1}
 \sum_{j=0}^2h_K(\mathsf R^jn).
 \label{eq:universal-lambda}
\end{equation}

\begin{proposition}[Equilateral enclosure gauge]
\label{prop:universal-enclosure-gauge}
The number $\Lambda(K)$ is the least side length of a closed equilateral
triangle containing $K$.  It is monotone under inclusion, translation
invariant, and positively homogeneous.  If $K$ is a finite polygonal hull, a
minimizing normal may be chosen perpendicular to one of its hull edges.
\end{proposition}

\begin{proof}
For a fixed unit normal $n$, the three supporting half-planes with outward
normals $n,\mathsf Rn,\mathsf R^2n$ form an equilateral triangle.  Its side is
$2/\sqrt3$ times the sum of the three support numbers, proving
\eqref{eq:universal-lambda}.  Monotonicity and homogeneity are immediate, while
translation invariance follows from
\[
 n+\mathsf Rn+\mathsf R^2n=0.
\]
On an open normal-direction cell with three fixed support points, the support
sum has the form $A\cos\theta+B\sin\theta$ and has second derivative equal to
its negative.  In the nondegenerate case an interior critical point is a
maximum.  Hence a minimum occurs at a cell boundary, where one support line
contains a hull edge.
\end{proof}

For the anchor hull
\[
 K(a,b,c)=\conv\{0,au,bv,c(u+v)\},
\]
where $u$ and $v$ are the two unit boundary directions meeting at $120^\circ$,
the admissible set is
\begin{equation}
 \mathcal A
 =\{(a,b,c)\in[0,1]^3:\Lambda(K(a,b,c))\le1\}.
 \label{eq:universal-admissible-gauge}
\end{equation}
% STATEMENTS-ONLY-REPLACE-BEGIN: universal-support-reference
The complete support calculation is given in
Subsection~\ref{sec:admissible-set-derivation} below.
% STATEMENTS-ONLY-REPLACE-END: universal-support-reference

\subsection{One equilateral radical}

% STATEMENTS-ONLY-REPLACE-BEGIN: universal-radical-summary
For $0\le x\le1$, define
\begin{equation}
 \omega(x)=\sqrt{1-x+x^2},
 \qquad
 \sigma(x)=1-\omega(x).
 \label{eq:universal-radical}
\end{equation}
Then
\begin{equation}
 \omega(1-x)=\omega(x),
 \qquad
 \sigma(x)=\frac{x(1-x)}{1+\omega(x)},
 \label{eq:universal-radical-identities}
\end{equation}
and direct differentiation gives
\begin{equation}
 \sigma''(x)=-\frac3{4\omega(x)^3}<0.
 \label{eq:universal-sigma-concavity}
\end{equation}
Thus $\sigma$ is strictly concave.  In the signed center calculus,
\[
 E=\omega(R),\qquad \eta=\sigma(R).
\]

For $0<x<1$, put $z=\sigma(x)/x$.  Eliminating the square root gives
\begin{equation}
 x=\frac{1-2z}{1-z^2},
 \qquad
 \sigma(x)=\frac{z(1-2z)}{1-z^2},
 \qquad 0<z<\frac12.
 \label{eq:universal-rational-radical}
\end{equation}
% STATEMENTS-ONLY-REPLACE-END: universal-radical-summary

\subsection{Forward envelope and nonsupercritical propagation}

For $0\le a,c\le1$, define the raw forward envelope
\begin{equation}
M_c(a)=\max\{b\in[0,1]:(a,b,c)\in\mathcal A\}.
\label{eq:reader-raw-capped-maps}
\end{equation}
Its zero-radial specialization, introduced geometrically in
Lemma~\ref{lem:signed-diameter-transfer}, is
\begin{equation}
 M_0(a)=\frac{-a+\sqrt{4-3a^2}}2.
 \label{eq:reader-zero-radial-envelope}
\end{equation}
The exact nonsupercritical forward cap and the following-edge propagation map
are
\begin{equation}
 \overline M_c(a)=
 \begin{cases}
 1-a,&0\le c\le1/2,\\
 M_c(a),&1/2<c\le1,
 \end{cases}
 \qquad
 \Phi_c(a)=1-\overline M_c(a).
 \label{eq:reader-technical-aliases}
\end{equation}
Equivalently,
\[
 \overline M_c(a)=\min\{M_c(a),1-a\}.
\]
\begin{lemma}[Midpoint threshold]
\label{lem:reader-midpoint-threshold}
For $0\le a\le1$,
\[
 M_c(a)\le1-a\qquad\left(\frac12\le c\le1\right),
 \qquad
 M_{1/2}(a)=1-a.
\]
Consequently $\Phi_c(a)\ge a$ whenever $c\ge1/2$.
\end{lemma}

\begin{proof}
If $(a,b,c)\in\mathcal A$ and $c\ge1/2$, the containing triangle also
contains the radial midpoint $(u+v)/2$.  Lemma~\ref{lem:self-midpoint} gives
$a+b\le1$, hence $b\le1-a$.

For the reverse inequality at $c=1/2$, set $b=1-a$.  In the exact cell
description of Proposition~\ref{prop:exact-admissible-set}, one has
$s=a+b=1$, $d=a^2+ab+b^2\le1$, $q=ab\ge0$, and, with
$M=\max\{a,b\}\ge1/2$,
\[
 F_T(a,b,1/2)=M(1/2-M)\le0,
 \qquad
 1/2\le2M.
\]
Thus $(a,b,1/2)\in\mathcal A$, proving equality.
\end{proof}

\begin{lemma}[Raw envelope and nonsupercritical cap]
\label{lem:reader-raw-capped-transfer}
Let $T$ be an actual V triangle satisfying
$A(T)\ge a$ and $C(T)\ge c$, and let $T_{\mathrm{next}}$ be the next
incident actual V triangle. Then
\[
 B(T)\le M_c(a).
\]
If no center interval intervenes on the next edge, then
\[
 A(T_{\mathrm{next}})\ge1-M_c(a).
\]
If $T$ is nonsupercritical, then
\begin{equation}
 B(T)\le\overline M_c(a)
 \label{eq:reader-branchwise-outgoing}
\end{equation}
and, on a center-free next edge,
\begin{equation}
 A(T_{\mathrm{next}})\ge\Phi_c(a)\ge a.
 \label{eq:reader-branchwise-transfer}
\end{equation}
\end{lemma}

\begin{proof}
The closure of $T$ realizes $(a,B(T),c)$, so the definition of $M_c$ gives
the raw outgoing bound. On a center-free edge, coverage gives
$A(T_{\mathrm{next}})\ge1-B(T)$. Under nonsupercriticality,
$B(T)\le1-A(T)\le1-a$; combining this with the raw bound gives
$B(T)\le\overline M_c(a)$. Taking complements proves the propagation bound.
\end{proof}

\begin{proposition}[Zero-radial warning]
\label{prop:reader-zero-radial-g}
For $0<a<1$,
\[
 M_0(a)>1-a,
 \qquad
 1-M_0(a)<a.
\]
Thus a nonsupercritical zero-radial handoff uses the cap
$\overline M_0(a)=1-a$, not the raw envelope $M_0(a)$.
\end{proposition}

\begin{proof}
By \eqref{eq:reader-zero-radial-envelope}, the inequality
$M_0(a)>1-a$ is equivalent, after squaring positive
quantities, to $4a(1-a)>0$.
\end{proof}

\subsection{Center-assisted propagation and the free envelope}

Let $J\subseteq[0,1]$ be empty or a closed interval.  For $0\le p\le1$, put
\begin{equation}
 e_J(p)=\sup\{x\in[0,1]:[0,x]\subseteq[0,p]\cup J\},
 \qquad
 \mathcal R_J(p)=1-e_J(p).
 \label{eq:reader-residual-operator}
\end{equation}
If $J=[L,U]$, then
\begin{equation}
 \mathcal R_{[L,U]}(p)=
 \begin{cases}
  1-p,&p<L,\\
  1-\max\{p,U\},&p\ge L,
 \end{cases}
 \qquad
 \mathcal R_\varnothing(p)=1-p.
 \label{eq:reader-residual-formula}
\end{equation}
The map is nonincreasing in $p$ and under enlargement of $J$.

\begin{lemma}[Generalized boundary handoff]
\label{lem:reader-generalized-handoff}
Suppose a unit boundary edge is covered by a left trace contained in $[0,p]$,
a C triangle trace $J$, and a right trace contained in $[1-q,1]$.  Then
\[
 q\ge\mathcal R_J(p).
\]
For an actual V triangle satisfying $A(T)\ge a$ and $C(T)\ge c$,
\begin{equation}
 A(T_{\mathrm{next}})\ge\mathcal R_J(M_c(a)).
 \label{eq:reader-generalized-raw-transfer}
\end{equation}
If $T$ is nonsupercritical, then
\begin{equation}
 A(T_{\mathrm{next}})\ge\mathcal R_J(\overline M_c(a)).
 \label{eq:reader-generalized-transfer}
\end{equation}
For $J=\varnothing$, the latter is exactly
$A(T_{\mathrm{next}})\ge\Phi_c(a)$.
\end{lemma}

\begin{proof}
The component of $[0,p]\cup J$ containing $0$ is $[0,e_J(p)]$.  If
$1-q>e_J(p)$, the intervening open interval is uncovered.  Apply this with
the raw outgoing bound $B(T)\le M_c(a)$, or with the nonsupercritical bound
$B(T)\le\overline M_c(a)$, and use monotonicity of $\mathcal R_J$.
\end{proof}

For $0\le c<1/2$, define
\begin{equation}
 M_c^{\rm sup}
 =\sup\{M_c(a):M_c(a)>1-a\}
 =\frac{c+\sqrt{c^2-8c+4}}2.
 \label{eq:reader-free-supercritical-envelope}
\end{equation}
The supremum is not attained. Hence an actual strict-supercritical V triangle
$T$ with $C(T)\ge c$ satisfies
\begin{equation}
 B(T)<M_c^{\rm sup}.
 \label{eq:reader-free-outgoing-envelope}
\end{equation}
If, in addition, its outgoing edge is center-free and $T_{\mathrm{next}}$ is
the next incident actual V triangle, then coverage gives
\begin{equation}
 A(T_{\mathrm{next}})>1-M_c^{\rm sup}.
 \label{eq:reader-free-raw-envelope}
\end{equation}
At $c=1/2$ the strict feasible set is empty. When convenient we write
$M_{1/2}^{\rm sup}=1/2$ only for the continuous extension of the displayed
formula, never for an attained strict supremum.

\subsection{Relaxed composition and path budgets}

For maps listed in geometric V triangle order, write
\begin{equation}
 [\Psi_1\mid\cdots\mid\Psi_r](x)
 =(\Psi_r\circ\cdots\circ\Psi_1)(x).
 \label{eq:reader-chain-notation}
\end{equation}
The leftmost slot acts first.  Write $\mathrm I(x)=x$.

\begin{lemma}[Relaxed composition]
\label{lem:reader-relaxed-composition}
Suppose propagated reach lower bounds satisfy
\[
 x_j\ge\Psi_j(x_{j-1})
 \qquad(1\le j\le r),
\]
where every $\Psi_j$ is nondecreasing.  If
$\underline\Psi_j\le\Psi_j$, then
\[
 x_r\ge[\underline\Psi_1\mid\cdots\mid\underline\Psi_r](x_0).
\]
\end{lemma}

\begin{proof}
Set $y_0=x_0$ and $y_j=\underline\Psi_j(y_{j-1})$.  Monotonicity gives
$x_j\ge\Psi_j(x_{j-1})\ge\Psi_j(y_{j-1})\ge y_j$ inductively.
\end{proof}

\begin{lemma}[Boundary-path budget]
\label{lem:reader-boundary-path}
Let $E_0,\ldots,E_k$ be consecutive boundary edges and let
$T_1,\ldots,T_k$ be the triangles at their intermediate vertices.  Suppose:
\begin{enumerate}
\item on $E_0$, the component covered from the endpoint opposite $T_1$ by all
triangles other than $T_1$ has extent at most $u$;
\item on $E_k$, the analogous component opposite $T_k$ has extent at most $v$;
\item on every middle edge, no triangle other than the two incident path triangles has
positive-length trace;
\item the chain is covered.
\end{enumerate}
Then
\begin{equation}
 \sum_{j=1}^k(A_j+B_j)\ge k+1-u-v.
 \label{eq:reader-path-budget}
\end{equation}
Consequently, if all $k$ internal V triangles are nonsupercritical, the chain is
impossible whenever $u+v<1$.
\end{lemma}

\begin{proof}
The first and last component hypotheses give $A_1\ge1-u$ and
$B_k\ge1-v$.  On every middle edge the C triangle and all nonincident triangles have
been excluded, so $B_j+A_{j+1}\ge1$.  Adding proves the formula.
\end{proof}

\subsection{Low roots, selected \(Q_+\) curves, and thresholds}

For the high-radial maps define
\begin{equation}
 e(d)=\frac{1-d}{2}
 \left(1-\sqrt{4(1-d)^2-3}\right)
 \qquad\left(0<d<1-\frac{\sqrt3}{2}\right),
 \label{eq:signed-low-root-definition}
\end{equation}
and, for the low-radial order transition,
\begin{equation}
 h(c)=\frac{2c}{1+\sqrt{16c^2-3}}
 \qquad\left(\frac12<c\le\frac{\sqrt3}{2}\right).
 \label{eq:signed-order-transition}
\end{equation}

\begin{proposition}[Universal selected-$Q_+$ curve]
\label{prop:reader-universal-tplus}
Fix $0<d<1/2$, put $c=1-d$, and suppose an input $p$ belongs to a selected
$Q_+$ arc of $\Phi_c$.  Write
\[
 q=\Phi_c(p),\qquad \nu=q-p,
 \qquad x=\frac{q-d}{1-d}.
\]
Then
\begin{equation}
 \nu=\sigma(x)=1-\sqrt{1-x+x^2},
 \label{eq:reader-psi}
\end{equation}
and
\begin{equation}
 q=d+(1-d)x,
 \qquad
 p=d+(1-d)x-\sigma(x).
 \label{eq:reader-tplus-normal}
\end{equation}
Consequently the selected arc is increasing and strictly concave.
\end{proposition}

\begin{proof}
The selected quadratic is
\[
 (1-q)(q-d)=(1-d)^2\nu(2-\nu).
\]
Substitution gives $x(1-x)=\nu(2-\nu)$, and the selected component uses the
smaller root $\nu=\sigma(x)$.  The input function is increasing and strictly
convex because $\sigma$ is strictly concave; inversion gives the assertion.
\end{proof}

The corresponding chord bounds are
\begin{equation}
 \Phi_{1-d}(p)
 \ge p+\frac{d}{e(d)-d}(p-d),
 \label{eq:reader-high-chord}
\end{equation}
and, with $t=h(1-d)$,
\begin{equation}
 \Phi_{1-d}(p)
 \ge p+\frac{1-2t}{t-d}(p-d).
 \label{eq:reader-low-chord}
\end{equation}
% STATEMENTS-ONLY-REPLACE-BEGIN: universal-affine-coefficient
For every coefficient $\lambda$ proved valid on the relevant selected arc,
% STATEMENTS-ONLY-REPLACE-END: universal-affine-coefficient
write
\begin{equation}
 \Phi_{1-d}^{\lambda}(p)
 :=p+\lambda(p-d)
 \le\Phi_{1-d}(p).
 \label{eq:reader-affine-relaxation}
\end{equation}
% STATEMENTS-ONLY-OMIT-BEGIN: universal-beta-rationalization
The rational form \eqref{eq:universal-rational-radical} gives, in the
historical notation $x=1-\beta$,
\begin{equation}
 \beta=\frac{z(2-z)}{1-z^2},
 \qquad
 m_\beta=\frac{1-z+z^2}{1-z^2}.
 \label{eq:reader-beta-rational}
\end{equation}
% STATEMENTS-ONLY-OMIT-END: universal-beta-rationalization

For $0<d<1-\sqrt3/2$, the exact high-reach lower-bound threshold is
\begin{equation}
 z>e(d)
 \quad\Longrightarrow\quad
 \Phi_{1-d}(z)\ge1-e(d).
 \label{eq:reader-high-threshold}
\end{equation}
Define
\begin{equation}
 \Phi_{1-d}^{\rm th}(z)=
 \begin{cases}
  z,&z\le e(d),\\
  1-e(d),&z>e(d),
 \end{cases}
 \qquad
 \Phi_{1-d}^{\rm th}\le\Phi_{1-d}.
 \label{eq:reader-threshold-transfer}
\end{equation}

\begin{lemma}[Threshold routing]
\label{lem:reader-threshold-routing}
Let $z_j=\Psi_j(z_{j-1})$, where every $\Psi_j$ is nondecreasing and extensive.
If $\Psi_k=\Phi_{1-d}$ and $z_{k-1}>e(d)$, then
$z_j\ge1-e(d)$ for all $j\ge k$.  If a chain contains later occurrences of
$\Phi_{1-d_1}$ and $\Phi_{1-d_2}$ and
\[
 z_0>e(d_1),
 \qquad
 \min\{e(d_1),e(d_2)\}<Q,
\]
then at least one of those two V triangles forces every subsequent output above
$1-Q$.
\end{lemma}

\begin{proof}
The first assertion is \eqref{eq:reader-high-threshold} followed by
extensivity.  If $e(d_1)<Q$, use the first threshold.  Otherwise
$z_0>e(d_1)\ge Q>e(d_2)$, so the second threshold fires.  The conclusion is
``at least one'', not uniqueness; both thresholds may be available.
\end{proof}
