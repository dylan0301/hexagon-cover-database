\section{Method 1: Trace-Length Bounds}
\label{sec:strategy1-summary}

For a closed triangle $T$ and a rectifiable target $E$, write
\[
 L_E(T)=\Haus(T\cap E).
\]
If the seven open triangles cover $E$, closure and subadditivity give
\[
 \Haus(E)\le L_E(T_C)+\sum_{i=0}^5L_E(T_i).
\]
Method~1 contradicts this inequality on either $\partial H$ or $S$.

\begin{proposition}[Trace bounds]
\label{prop:body-trace-register}
For every classified V triangle $T_i$, its boundary trace is supported on
its two incident boundary edges, and therefore
\[
 L_{\partial H}(T_i)=A_i+B_i.
\]
Under the indicated hypotheses, the following bounds hold.
\begin{center}
\small
\renewcommand{\arraystretch}{1.08}
\begin{tabular}{p{.64\textwidth}c}
\toprule
triangle and target&upper bound\\
\midrule
CE1/CE2 C triangle on $S$&$L_S(T_C)<3/2$\\
supercritical V triangle on $S$&$L_S(T_i)<3/2$\\
$T_i$ with $\Haus(T_i\cap r_j)>0$ for some $j\in\{i-1,i+1\}$, on $S$&
$L_S(T_i)<3/2$\\
nonsupercritical V triangle with no such radial intersection on $S$&
$L_S(T_i)\le2$\\
CE1 C triangle on $\partial H$&
$L_{\partial H}(T_C)\le\sqrt3/2-3/4$\\
CE2 C triangle on $\partial H$&$L_{\partial H}(T_C)<1/2$\\
supercritical $\Vdzero$ triangle on $\partial H$&
$L_{\partial H}(T_i)\le2/\sqrt3$\\
$\Vdone$ or $\Vdtwo$ triangle on $\partial H$&
$L_{\partial H}(T_i)<1/2$\\
nonsupercritical $\Vdzero$ triangle on $\partial H$&
$L_{\partial H}(T_i)\le1$\\
$\Tlike$ triangle on $\partial H$&$L_{\partial H}(T_i)<1$\\
\bottomrule
\end{tabular}
\end{center}
\end{proposition}

\begin{proof}
The four skeleton bounds are Lemmas~\ref{lem:center-skeleton-cap},
\ref{lem:supercritical-skeleton-cap},
\ref{lem:positive-support-skeleton-cap}, and
\ref{lem:no-support-skeleton-cap}.  The boundary bounds are proved in
Appendix~\ref{sec:length}.
\end{proof}

\begin{lemma}[Skeleton trace criterion]
\label{lem:body-skeleton-trace-criterion}
If $N_++N_{\mathrm{sp}}\ge3$, the seven triangles do not cover $S$.
\end{lemma}

\begin{proof}
At least three V triangles have skeleton trace strictly below $3/2$,
while every other V triangle has trace at most $2$.  Hence
\[
 L_S(T_C)+\sum_iL_S(T_i)
 <\frac32+(N_++N_{\mathrm{sp}})\frac32
 +(6-N_+-N_{\mathrm{sp}})2
 \le12=\Haus(S),
\]
contradicting subadditivity for a cover.
\end{proof}

\begin{lemma}[Perimeter trace criterion]
\label{lem:body-perimeter-trace-criterion}
Put
\[
 \theta=\frac2{\sqrt3}-1.
\]
Suppose the center trace has length at most $L_C$, there are $n$
supercritical $\Vdzero$ triangles, there are $s\ge1$ distinguished
nonsupercritical V triangles with strict bounds
$\lambda_1,\ldots,\lambda_s<1$, and every remaining V triangle has
boundary trace at most one.  If
\[
 L_C+n\theta\le\sum_{j=1}^s(1-\lambda_j),
\]
then the seven triangles do not cover $\partial H$.
\end{lemma}

\begin{proof}
The total boundary trace is strictly less than
\[
 L_C+n\frac2{\sqrt3}+\sum_{j=1}^s\lambda_j+(6-n-s)
 =6+L_C+n\theta-\sum_{j=1}^s(1-\lambda_j)
 \le6,
\]
contradicting $\Haus(\partial H)=6$.
\end{proof}

\begin{proposition}[Cases excluded by trace length]
\label{prop:body-length-branches}
Method~1 excludes:
\begin{enumerate}
\item $\CEzero$, $N_+=0$;
\item $\CEzero$, $N_+=1$, with a $\Vdone/\Vdtwo$ triangle;
\item CE1/CE2 with $N_++N_{\mathrm{sp}}\ge3$;
\item CE1/CE2, $N_+=0$, with a $\Vdone/\Vdtwo$ triangle;
\item CE1, $N_+=1$, with exactly one $\Vdone/\Vdtwo$ triangle and no
$\Tlike$ triangle;
\item the CE2 subcase in which the unique $\Vdtwo$ triangle $T_i$ contains
$M_{i-1}$ or $M_{i+1}$.
\end{enumerate}
\end{proposition}

\begin{proof}
The first case is the cyclic boundary sum.  The third follows from
Lemma~\ref{lem:body-skeleton-trace-criterion}.  The remaining cases substitute
the appropriate center and vertex bounds into
Lemma~\ref{lem:body-perimeter-trace-criterion}.  The exact substitutions and
strict inequalities are Proposition~\ref{prop:length-branches} and
Corollary~\ref{cor:signed-budget-branches}.
\end{proof}
