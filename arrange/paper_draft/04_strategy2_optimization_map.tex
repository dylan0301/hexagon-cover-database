\subsection{The primitive high-radial output function}

For
\[
 \frac12<c<1,\qquad 0\le x\le1,
\]
define
\[
 h(c)=\frac{2c}{1+\sqrt{16c^2-3}},
 \qquad
 Q_-(c,x)=\frac{\sqrt{x^2-cx+c^2}}{c}-x.
\]
For
\[
 \frac{\sqrt3}{2}\le c<1,
\]
define
\[
 \ell(c)=\frac c2\left(1-\sqrt{4c^2-3}\right),
 \qquad
 r(c)=c-\ell(c).
\]
Finally put
\[
 D_+(c,x)
 =(2xc^2+c)^2-4(1-c^2)(1-x^2)c^2
\]
and, whenever \(D_+(c,x)\ge0\), define
\[
 Q_+(c,x)=
 \frac{2(1-x^2)c^2}
 {2xc^2+c+\sqrt{D_+(c,x)}}.
\]

\begin{lemma}[Radical and denominator domains]
\label{lem:s2-pure-map-well-defined}
If \(1/2<c<1\) and \(1-c\le x\le1\), then
\[
 D_+(c,x)
 =c^2\left(4x^2+4cx+4c^2-3\right)
 \ge c^2(2c-1)^2>0.
\]
Every denominator in \(h,Q_-,Q_+\) is positive on the branch on which the
corresponding expression is used.  The functions \(\ell,r\) are real-valued on
their stated domain.
\end{lemma}

\begin{proof}
The displayed factorization is direct.  The quadratic
\(x\mapsto4x^2+4cx+4c^2-3\) is increasing for \(x\ge0\), so its value at
\(x=1-c\) is \((2c-1)^2\).  The remaining assertions follow from
\(c>1/2\), positivity of square roots, and \(2xc^2+c>0\).
\end{proof}

\begin{definition}[Pure high-radial output function]
\label{def:s2-pure-map}
Define \(F:(1/2,1)\times[0,1]\to[0,1]\) as follows.  For
\(1/2<c\le\sqrt3/2\), put
\[
 F(c,x)=
 \begin{cases}
  1-x,&0\le x\le1-c,\\
  Q_+(c,x),&1-c<x<h(c),\\
  h(c),&x=h(c),\\
  Q_-(c,x),&h(c)<x<c,\\
  1-x,&c\le x\le1.
 \end{cases}
\]
For \(\sqrt3/2<c<1\), put
\[
 F(c,x)=
 \begin{cases}
  1-x,&0\le x\le1-c,\\
  Q_+(c,x),&1-c<x\le \ell(c),\\
  \ell(c),&\ell(c)<x<r(c),\\
  Q_-(c,x),&r(c)\le x<c,\\
  1-x,&c\le x\le1.
 \end{cases}
\]
Define the complementary update
\[
 T(c,x):=1-F(c,x).
\]
\end{definition}

By Proposition~\ref{prop:branchwise-safe-output}, this function is exactly
$F(c,x)=g_c(1-x)$ on its stated high-radial domain.

The equality assignments in Definition~\ref{def:s2-pure-map} are part of the
problem specification.  In particular, the value at \(x=\ell(c)\) is the
selected \(Q_+\) value \(r(c)\), while the open interval immediately to its
right has value \(\ell(c)\).

\begin{definition}[Pure lower relaxations]
\label{def:s2-pure-relaxations}
For \(0<d<1-\sqrt3/2\), define
\[
 E_0(d):=\ell(1-d).
\]
For \(\lambda\in\mathbb R\), define
\[
 A_{d,\lambda}(x):=x+\lambda(x-d).
\]
It is licensed on an interval \(I\) only after proving
\[
 A_{d,\lambda}(x)\le T(1-d,x)
 \qquad(x\in I).
\]
Define
\[
 \Theta_d(x)=
 \begin{cases}
 x,&0\le x\le E_0(d),\\
 1-E_0(d),&E_0(d)<x\le1.
 \end{cases}
\]
The verification appendix proves
\[
 \Theta_d(x)\le T(1-d,x).
\]
These are proof devices for Problems~\ref{prob:s2-ce1-five-chain} and
\ref{prob:s2-ce2-five-chain}; they are not additional primitive functions in
the problem statements.
\end{definition}
