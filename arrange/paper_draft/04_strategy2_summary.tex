\section{Strategy 2: Monotone Transfer Certificates}
\label{sec:strategy2-summary}

Strategy~2 turns boundary coverage into lower bounds propagated around the
six rows.  The geometric input is Proposition~\ref{prop:body-transfer-interface};
the remaining work is to retain a small number of exact endpoint maps, replace
other center-free nonsupercritical rows by lower relaxations, and prove one
terminal inequality.

\begin{definition}[Geometric-order composition and identity slots]
\label{def:body-chain-notation}
Let $r\in\mathbb Z_{\ge1}$, let
$X_0,\ldots,X_r$ be sets, and let
$\Phi_j:X_{j-1}\to X_j$ for $1\le j\le r$.  Define
\[
 [\Phi_1\mid\cdots\mid\Phi_r]
 :=\Phi_r\circ\cdots\circ\Phi_1:X_0\to X_r.
\]
For $X_0=\cdots=X_r=[0,1]$, define
\[
 \mathrm I:[0,1]\to[0,1],\qquad \mathrm I(x):=x,
\]
\[
 \mathrm I^j:=
 \underbrace{[\mathrm I\mid\cdots\mid\mathrm I]}_{j\text{ slots}}
 \quad(j\in\mathbb Z_{\ge1}),
 \qquad
 \mathrm I^0:=\mathrm I.
\]
\end{definition}

\subsection{The all-Vd0 kernel}

\begin{definition}[V-gaps and active-gap rank]
\label{def:body-active-gap-rank}
For $i\in\mathbb Z/6\mathbb Z$, parametrize $e_{i,i+1}$ by
\[
 \xi_i:[0,1]\to e_{i,i+1},
\qquad
 \xi_i(t):=V_i+t(V_{i+1}-V_i).
\]
The set missed on this edge by its two incident open vertex roles is
\[
 \Gamma_i:=
 \begin{cases}
  \xi_i\bigl([B_i,1-A_{i+1}]\bigr),
       &B_i\le1-A_{i+1},\\
  \varnothing,&B_i>1-A_{i+1}.
 \end{cases}
\]
A nonempty $\Gamma_i$ is a \emph{V-gap}; in particular,
$B_i=1-A_{i+1}$ gives a singleton V-gap.  In a hypothetical cover define
\[
 \mathrm{gr}:=
 \left\lvert
 \left\{i\in\mathbb Z/6\mathbb Z:
 \Gamma_i\ne\varnothing,\ \Gamma_i\subset U_C
 \right\}
 \right\rvert .
\]
Boundary locality implies that every nonempty $\Gamma_i$ is contained in
$U_C$, so $\mathrm{gr}$ is equivalently the number of nonempty V-gaps, or the
number of positive center traces containing one.
\end{definition}

\begin{figure}[H]
\centering
\resizebox{.99\linewidth}{!}{\input{figures/strategy2_gap_rank_skeleton}}
\caption{The three active-gap ranks.  A singleton $\Gamma_i$ remains a
nonempty V-gap and is counted.}

The two row counts and the active-gap rank form the single kernel
\[
\begin{array}{c|ccc}
 &\mathrm{gr}=0&\mathrm{gr}=1&\mathrm{gr}=2\\ \hline
N_+=0&\mathrm I^6\text{ cycle}&
       \text{two endpoints}&
       \text{paired endpoints}\\
N_+=1&\text{nine points}&
       \text{five-row chain}&
       \text{paired endpoints}.
\end{array}
\]
The last column is CE2-only.  For $N_+=0$ and $\mathrm{gr}=0$, the center has
no trace on any missed handoff and the six incident vertex traces alone cover
every edge.  Thus
\[
 A_{i+1}>1-B_i
 \ge\widehat g_{C_i}^\vee(A_i)\ge A_i,
\]
which gives the impossible strict cycle
\[
 A_0<A_1<\cdots<A_5<A_0.
\]

\begin{proposition}[All-Vd0 endpoint certificates]
\label{prop:body-all-vd0-endpoints}
\begin{enumerate}
\item Suppose
\[
 (R,\alpha,\delta)\in
 \mathcal D_{\rm CE1}\cup\mathcal D_{\rm CE2}.
\]
Assume $N_+=0$, all six roles are Vd0, and exactly the normalized right
center trace contains a V-gap.  Use the one-gap data of
Definition~\ref{def:body-gap-endpoint-data}.
Then the two exact hatted outgoing caps satisfy
\[
 \widehat g_{c_5^{\mathrm{one}}}(1-a_5^{\mathrm{one}})
 +\widehat g_{c_1^{\mathrm{one}}}(1-a_1^{\mathrm{one}})<1.
\]
The companion edge is center-free in CE1 and gap-free in CE2; the three middle
rows satisfy Definition~\ref{def:body-center-free-path} and may be replaced
by $\mathrm I^3$.

\item Suppose
\[
 (R,\alpha,\delta)\in\mathcal D_{\rm CE2}.
\]
Assume the two center traces both contain V-gaps, and rows
$T_1,\ldots,T_5$ are nonsupercritical Vd0.  Use the paired-gap data of
Definition~\ref{def:body-gap-endpoint-data}.
Then
\[
 \widehat g_{c_5^{\mathrm{two}}}(1-a_5^{\mathrm{two}})
 +\widehat g_{c_1^{\mathrm{two}}}(1-a_1^{\mathrm{two}})<1.
\]
Again the three middle rows satisfy
Definition~\ref{def:body-center-free-path}.
\end{enumerate}
Singleton gaps are included.
\end{proposition}

\begin{proof}
These are the one-side and paired-endpoint theorems in
Appendix~\ref{sec:exact-local-demand-calculus}, applied through the corrected
boundary-path lemma, Lemma~\ref{lem:reader-boundary-path}.  The endpoint
residuals include every center contribution before the ordinary path is used.
\end{proof}

\begin{definition}[Low root and decorated lower relaxations]
\label{def:body-decorated-relaxations}
Set
\[
 d_*:=1-\frac{\sqrt3}{2}.
\]
Define the high-radial low root
\[
 e:(0,d_*)\to(0,1),\qquad
 e(d):=\frac{1-d}{2}
 \left(1-\sqrt{4(1-d)^2-3}\right),
\]
and the low-radial order transition
\[
 h:\left(\frac12,\frac{\sqrt3}{2}\right]\to(0,1),\qquad
 h(c):=\frac{2c}{1+\sqrt{16c^2-3}}.
\]
For $0<d<1/2$, define the selected-$T_+$ input interval
\[
 \mathcal T_d:=[d,\tau(d)],\qquad
 \tau(d):=
 \begin{cases}
  e(d),&0<d<d_*,\\
  h(1-d),&d_*\le d<1/2.
 \end{cases}
\]
Define the set of certified affine coefficients
\[
 \mathcal C_d:=
 \left\{\lambda\in\mathbb R:
 z+\lambda(z-d)\le\widehat g_{1-d}^{\vee}(z)
 \quad\forall z\in\mathcal T_d
 \right\}.
\]
For $\lambda\in\mathcal C_d$, define the affine lower relaxation on its
certified interval by
\[
 \widehat g_{1-d}^{\vee,\lambda}:\mathcal T_d\to\mathbb R,\qquad
 \widehat g_{1-d}^{\vee,\lambda}(z):=z+\lambda(z-d).
\]
For $0<d<d_*$, define the threshold lower relaxation on all of $[0,1]$ by
\[
\widehat g_{1-d}^{\vee,\mathrm{th}}:[0,1]\to[0,1],
\qquad
 \widehat g_{1-d}^{\vee,\mathrm{th}}(z):=
 \begin{cases}
  z,&0\le z\le e(d),\\
  1-e(d),&e(d)<z\le1.
 \end{cases}
\]
It satisfies
\[
 \widehat g_{1-d}^{\vee,\mathrm{th}}(z)
 \le\widehat g_{1-d}^{\vee}(z)\qquad(0\le z\le1).
\]
\end{definition}

\begin{proposition}[Five-row one-gap certificate]
\label{prop:body-five-row-certificate}
Assume
\[
 (R,\alpha,\delta)\in
 \mathcal D_{\rm CE1}\cup\mathcal D_{\rm CE2},
\]
all six roles are Vd0, $N_+=1$, and the normalized right center trace
contains a V-gap.  If $(R,\alpha,\delta)\in\mathcal D_{\rm CE2}$, assume the
companion trace is gap-free.  Use the data of
Definition~\ref{def:body-five-row-data}.
The actual rows satisfy
\[
 A_0\ge Z,
 \qquad
 A_0<1-h_{\mathrm{term}}.
\]
Exact capped-map duality reduces the required contradiction
$Z>1-h_{\mathrm{term}}$ to
\[
 [\widehat g_{1-\alpha}^\vee
  \mid\widehat g_{1-d_3^C}^\vee
  \mid\widehat g_{1-\delta}^\vee](h_{\mathrm{term}})
 >1-x_{\mathrm{in}}.
\]
For $(R,\alpha,\delta)\in\mathcal D_{\rm aff}$, the two certified affine
iterates of Definition~\ref{def:body-ce1-affine-domain} satisfy
\[
 \ell_2>e(\delta),
\qquad
 \widehat g_{1-\delta}^{\vee,\mathrm{th}}(\ell_2)
 >1-x_{\mathrm{in}}.
\]
The remaining CE1 capped-map branches are closed by the exact branch audit.

For $(R,\alpha,\delta)\in\mathcal D_{\rm CE2}$,
\[
 0<\alpha<d_*,\qquad 0<\delta<d_*,
\]
\[
 x_{\mathrm{in}}>e(\alpha),\qquad
 \min\{e(\alpha),e(\delta)\}<h_{\mathrm{term}}.
\]
Therefore at least one of the $\alpha$- or $\delta$-threshold slots may be
chosen; every other slot is replaced by $\mathrm I$.  Both thresholds may be
available, and no uniqueness is asserted.  Reflection exchanges
$(R,\alpha)$ with $(W,\delta)$ and reverses the row order.
\end{proposition}

\begin{proof}
The actual-row induction and terminal diameter bound are
Proposition~\ref{prop:signed-one-gap-interface}.  The CE1 affine calculation is
Proposition~\ref{prop:reader-ce1-one-gap}; the CE2 alternatives are
Proposition~\ref{prop:signed-ce2-one-gap}.  Every formal iterate is a lower
bound for the corresponding actual incoming reach.
\end{proof}

\begin{definition}[Vd1--supercritical axis-replacement data]
\label{def:body-vd-replacement-data}
Let $t\ge1$ and let $a,b\in[0,1]$ satisfy
\[
 b>\frac{t-1}{2t},
\qquad
 a+tb<\sqrt{t^2+t+1}-1-\frac t2.
\]
Define
\[
 d:=\sqrt{t^2+t+1},
\]
\[
 X_{\rm loc}:\mathbb R^2\to\mathbb R^2,\qquad
 X_{\rm loc}(x,y):=
 V_0+x(V_5-V_0)+y(V_1-V_0),
\]
\[
 T_0^{\rm loc}:=
 \left\{(x,y)\in\mathbb R^2:
 \begin{array}{rcl}
 x-(t+1)y&\le&a,\\
 ty-(t+1)x&\le&tb,\\
 tx+y&\le&d-a-tb
 \end{array}
 \right\},
\]
\[
 c:=\frac{d-a-tb}{t+1},
\qquad
 \lambda:=\frac{t(1-b)}{t+1},
\qquad
 \mu:=\frac{d-a-tb-1}{t}.
\]
Let
\[
 (A_1,B_1,C_1,d_1^C)\in[0,1]^4,
\qquad
 (A_1,B_1,C_1)\in\mathcal A,

 A_1+B_1>1,

 A_1\ge1-b,\qquad C_1\ge\lambda,\qquad
 c_1^{\rm req}:=1-d_1^C,
\qquad
 c_1^{\rm req}<\max\{C_1,\mu\}.
\]

\[
 a<p_1<p_2<1-B_1,\qquad
 p_1<\frac12,\qquad
 1-p_1>c,\qquad
 \max\{p_2,1-p_2\}>c_1^{\rm req}.

\end{definition}

\begin{proposition}[Adjacent rescuer and Vd terminal certificates]
\label{prop:body-special-terminal-certificates}
The following exact geometric terminals hold.
\begin{enumerate}
\item For the data of Definition~\ref{def:body-adjacent-rescuer-data}, the
local profile gives
\[
 a_{\mathrm{res}}\le1-g_{c_{\mathrm{res}}}^{\rm sc},
 \qquad
 \vartheta_{\mathrm{res}}\le1-g_{c_{\mathrm{res}}}^{\rm sc}.
\]
Radial isolation forces the opposite boundary demand to be at least
$g_{c_{\mathrm{res}}}^{\rm sc}$, whereas the unique supercritical row has
outgoing reach strictly below $g_{c_{\mathrm{res}}}^{\rm sc}$.  The
intervening four-row boundary path is center-free on its internal edges, so
Lemma~\ref{lem:reader-boundary-path} gives a contradiction.

\item In the adjacent CE2 one-Vd1/Vd2 placement, use
Definition~\ref{def:body-vd-residual-data} and reflect the
$\tau=5$ case to the normalized representative $\tau=1$.  Then
\[
 a_{\mathrm R}^{\mathrm{res}}+a_{\mathrm L}^{\mathrm{res}}<\frac12,
 \qquad
 \delta<\frac{a_{\mathrm L}^{\mathrm{res}}}{4}.
\]
More precisely,
\[
 U_{1\to2}<1-a_{\mathrm L}^{\mathrm{res}},
 \qquad
 C_2\le1-\frac{a_{\mathrm L}^{\mathrm{res}}}{4}
 <1-\delta.
\]
The first bound also lies below $1-\delta$; hence $r_2$ is uncovered.

\item In a nonadjacent placement, use
Definition~\ref{def:body-vd-residual-data}.  Then
\[
 a_{\mathrm R}^{\mathrm{res}}>s_C,\qquad
 a_{\mathrm L}^{\mathrm{res}}>s_C,
\]
\[
 d_\tau^C<
 \min\{a_{\mathrm R}^{\mathrm{res}},a_{\mathrm L}^{\mathrm{res}}\},
 \qquad
 C_\tau<
 1-\min\{a_{\mathrm R}^{\mathrm{res}},a_{\mathrm L}^{\mathrm{res}}\}.
\]
Thus the Vd radial trace and the center trace are disjoint.

\item In the remaining adjacent Vd1--supercritical placement, use
Definition~\ref{def:body-vd-replacement-data}.  The two roles $T_0'$ and
$T_1'$ are open nonsupercritical Vd0 roles and preserve every boundary and
radial demand used by the all-Vd0 theorem.
\end{enumerate}
\end{proposition}

\begin{proof}
Items 1--3 are Corollary~\ref{cor:signed-adjacent-rescuers} and
Lemmas~\ref{lem:signed-vd-adjacent-placement} and
\ref{lem:signed-vd-nonadjacent-placement}.  Item 4 is
Proposition~\ref{prop:paper-vd1-pair-replacement}.
\end{proof}

\begin{proposition}[Branches closed by Strategy 2]

All singleton-gap states are included.
\end{proof}
