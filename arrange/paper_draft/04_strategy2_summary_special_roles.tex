\subsection{Special-role terminals}

\begin{definition}[Adjacent-rescuer record]
\label{def:body-adjacent-rescuer-data}
An adjacent T3-like or Vd1 rescuer is reduced to three scalar reaches: its
boundary contribution, the entry point of its adjacent radial trace, and the
uncovered radial tail absorbed by the center.  These scalars form the rescuer
record of Appendix~\ref{sec:t3-vd-demand-branches}.  No placement coordinate
is used after this record has been extracted.
\end{definition}

\begin{definition}[CE2 Vd residual record]
\label{def:body-vd-residual-data}
For a CE2 state with exactly one Vd1 or Vd2 role, the geometric placement
audit produces two center-assisted boundary residuals and one of two terminal
radial records: adjacent or nonadjacent.  These real records are the only data
passed to the terminal inequalities in
Appendix~\ref{sec:short-vd-placements}.
\end{definition}

\begin{definition}[Vd1--supercritical replacement data]
\label{def:body-vd-replacement-data}
The remaining adjacent Vd1--supercritical placement uses two separate local
vertex charts.  The strict placement margins determine two open,
nonsupercritical Vd0 replacements that preserve the full skeleton coverage.
This is a geometric reduction, not a function-optimization problem.
\end{definition}

\begin{proposition}[Adjacent rescuer and Vd terminal certificates]
\label{prop:body-special-terminal-certificates}
The special-role branches terminate as follows.

\begin{enumerate}
\item An adjacent T3-like or Vd1 rescuer forces an opposite endpoint demand at
least the free strict-supercritical envelope, while the unique
supercritical role has outgoing reach strictly below that envelope.  The
intervening four-V-triangle path is center-free.
\item In an adjacent CE2 one-Vd1/Vd2 placement, both possible radial bridges
end strictly before the center trace begins.
\item In a nonadjacent placement, the Vd radial trace and the center trace are
strictly separated.
\item In the remaining adjacent Vd1--supercritical placement, the two-chart
replacement reduces the configuration to the already excluded all-Vd0
skeleton state.
\end{enumerate}
\end{proposition}

\begin{proof}
Items 1--3 are the exact rescuer, quarter-radial, and Vd-corner terminal
inequalities in Appendices~\ref{sec:t3-vd-demand-branches} and~\ref{sec:short-vd-placements}.  Item 4 is
Proposition~\ref{prop:paper-vd1-pair-replacement}.
\end{proof}

\begin{proposition}[Branches closed by Strategy 2]
\label{prop:body-demand-branches}
Strategy~2 closes:

\begin{enumerate}
\item the $N_+=0$ all-Vd0 and T3-like-without-Vd1/Vd2 branches for CE1 and
CE2;
\item every nonempty-gap $N_+=1$ all-Vd0 state;
\item the exactly-one-T3-like $N_+=1$ state with no Vd1/Vd2 role;
\item the complementary transfer placements in the CE2, $N_+=1$,
exactly-one-Vd1/Vd2 branch.
\end{enumerate}

Singleton V-gaps are included.
\end{proposition}

\begin{proof}
Use Propositions~\ref{prop:body-all-vd0-endpoints},
\ref{prop:body-five-V-triangle-certificate},
\ref{prop:body-t3-endpoint}, and
\ref{prop:body-special-terminal-certificates}.  The complete CE2 one-Vd
placement assembly is Proposition~\ref{prop:paper-ce2-one-vd-placements}.
The zero-gap $N_+=1$ all-Vd0 cell remains the Strategy~4 nine-point
obstruction.
\end{proof}
