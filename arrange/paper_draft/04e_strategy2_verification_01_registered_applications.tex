\subsection{All-Vd0 applications}

\begin{proposition}[Common CE2 two-gap application]
\label{prop:reader-common-two-gap}
Assume both CE2 center traces contain V-gaps and
\(T_1,\ldots,T_5\) are nonsupercritical Vd0 roles.  Then the perimeter is not
covered.  No hypothesis on the criticality of \(T_0\) is required.
\end{proposition}

\begin{proof}
The bridge identifies the endpoint record with Problem S2-E2.  Theorem~\ref{thm:s2-pure-e2} makes its objective negative, so the two endpoint capacities sum to less than one.  The three
intervening roles form the center-free path required by
Lemma~\ref{lem:reader-boundary-path}, giving the contradiction.
\end{proof}

For the one-gap return, all five radial demands exceed $1/2$; retain the exact raw forward notation
\begin{equation}
 Z=
 [g_{c_1}^\vee\mid
  g_{c_2}^\vee\mid
  g_{c_3}^\vee\mid
  g_{c_4}^\vee\mid
  g_{c_5}^\vee](X).
 \label{eq:reader-exact-five-chain}
\end{equation}
The bridge gives \(A_0\ge Z\) and \(A_0<1-H\).  The registered reverse
three-update target is
\begin{equation}
 [g_{1-\alpha}^\vee
  \mid g_{1-m_3}^\vee
  \mid g_{1-\delta}^\vee](H)
 >1-X.
 \label{eq:reader-signed-three-map}
\end{equation}

\begin{proposition}[CE1 one-gap obstruction]
\label{prop:reader-ce1-one-gap}
The CE1, \(N_+=1\), all-Vd0, one-gap state is impossible.
\end{proposition}

\begin{proof}
Proposition~\ref{prop:s2-return-bridge} identifies the required scalar sign
with Problem S2-R1.  Theorem~\ref{thm:s2-pure-r1} proves that problem,
and the exact forward interface yields \(Z>1-H\).
\end{proof}

\begin{corollary}[CE2 one-gap obstruction]
\label{prop:reader-ce2-one-gap}
The CE2, \(N_+=1\), all-Vd0, exactly-one-gap state is impossible in either
orientation.
\end{corollary}

\begin{proof}
Problem S2-R2 is Theorem~\ref{thm:s2-pure-r2}.  Its threshold
condition is an inclusive disjunction: at least one of the two locations
fires, and both may fire.  Reflection reverses the five-role order and
exchanges the two signed slacks.
\end{proof}

\subsection{Registered special-role applications}

The universal proof owners are:
\begin{itemize}
\item high-radial S2-T3: Theorem~\ref{thm:s2-pure-t3}, with objective
identification in Proposition~\ref{prop:s2-t3-objective-bridge}; the
low-radial endpoint is the direct right-Full row of
Theorem~\ref{thm:paper-complete-t3-endpoint};
\item S2-SC: Theorem~\ref{thm:s2-pure-sc};
\item adjacent S2-VD: Theorem~\ref{thm:s2-pure-vd-adjacent};
\item nonadjacent S2-VD: Theorem~\ref{thm:s2-pure-vd-nonadjacent}.
\end{itemize}
The remaining two-chart replacement is
Proposition~\ref{prop:paper-vd1-pair-replacement}; it is geometric rather
than an optimization problem.

\begin{proposition}[Branches closed by registered Strategy 2 problems]
\label{prop:reader-demand-branches}
Strategy~2 closes:
\begin{enumerate}
\item the \(N_+=0\) all-Vd0 and T3-like-without-Vd1/Vd2 branches for CE1 and
CE2;
\item every nonempty-gap \(N_+=1\) all-Vd0 state;
\item the exactly-one-T3-like \(N_+=1\) state with no Vd1/Vd2 role;
\item the complementary transfer placements in the CE2, \(N_+=1\),
exactly-one-Vd1/Vd2 branch.
\end{enumerate}
Singleton V-gaps are included.
\end{proposition}

\begin{proof}
Use Corollary~\ref{cor:s2-pure-registry} through the bridge propositions above,
and use Theorem~\ref{thm:paper-complete-t3-endpoint} for the direct
low-radial right-Full T3 endpoint.  The zero-gap \(N_+=1\) all-Vd0 cell
remains the Strategy~4 nine-point obstruction.
\end{proof}
