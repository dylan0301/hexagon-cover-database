\subsection{Propagation roadmap and the T3 endpoint state}
\label{sec:strategy2-reader}

The proof below has eight geometric outputs: the one- and two-gap endpoint
inequalities, the CE1 and CE2 return inequalities, the support-isolated T3
endpoint inequality, the common adjacent-exception obstruction, and the two
Vd separation statements. Their proof-bearing locations carry the labels
\ref{thm:s2-geom-e1}, \ref{thm:s2-geom-e2},
\ref{thm:s2-geom-r1}, \ref{thm:s2-geom-r2},
\ref{thm:s2-geom-t3}, \ref{thm:s2-geom-sc},
\ref{thm:s2-geom-vd-adjacent}, and
\ref{thm:s2-geom-vd-nonadjacent}, respectively. This roadmap introduces no
second registry or universal optimization layer.

\begin{definition}[Support-isolated T3-like endpoint state]
\label{def:body-t3-endpoint-state}
Assume $T_C$ is CE1 or CE2, $N_+=0$, no V triangle is Vd1 or Vd2, and,
after normalization and reflection, $T_0$ is T3-like with
$\Haus(T_0\cap r_1)>0$. Apply the closed-trace domination of
Proposition~\ref{prop:t3-translation} and retain the symbol $T_0$ for the
resulting trace-dominating majorant. The support-isolation hypotheses are
\[
 \Haus(T_j\cap r_1)=0\quad(j\notin\{0,1\}),
 \qquad
 \Haus(T_j\cap r_5)=0\quad(j\ne5).
\]
Parameterize $e_{0,1}$ and $e_{5,0}$ from $V_0$, and $r_1,r_5$ from $O$.
Write
\[
 T_0\cap e_{0,1}=[0,p_1],\qquad
 T_0\cap e_{5,0}=[0,p_5],
\]
\[
 J_1=T_C\cap e_{0,1},\qquad
 J_5=
 \begin{cases}
 T_C\cap e_{5,0},&T_C\text{ is CE2},\\
 \varnothing,&T_C\text{ is CE1}.
 \end{cases}
\]
In CE1 a possible closure-only point contact is discarded: it has zero trace,
and the original open C triangle supplies neither open coverage nor a
positive boundary interval there. Write
\[
 T_C\cap r_i=[0,\gamma_i]\quad(i\in\{1,5\}),\qquad
 T_0\cap r_1=[1-u,1-c].
\]
With the residual operator $\mathcal R_J$ of
\eqref{eq:reader-residual-operator}, define
\[
 (a_5^\ast,c_5^\ast)
 =\left(\mathcal R_{J_5}(p_5),1-\gamma_5\right),
\]
\[
 (a_1^\ast,c_1^\ast)=
 \begin{cases}
 \left(\mathcal R_{J_1}(p_1),c\right),
   &1-u\le\gamma_1\le1-c,\\
 \left(\mathcal R_{J_1}(p_1),1-\gamma_1\right),
   &\text{otherwise}.
 \end{cases}
\]
For $i\in\{1,5\}$ the endpoint capacity is
$\overline M_{c_i^\ast}(a_i^\ast)$; it is the direct nonsupercritical value
$1-a_i^\ast$ when $c_i^\ast\le1/2$.
If $A_i,C_i$ denote the actual maximal boundary and radial reaches of $T_i$,
then the displayed residual quantities are the selected lower bounds
\[
 a_i^\ast\le A_i,\qquad c_i^\ast\le C_i
 \qquad(i\in\{1,5\}).
\]
\end{definition}

\begin{proposition}[Common CE2 two-gap application]
\label{prop:reader-common-two-gap}
Assume both CE2 center traces contain boundary gaps and
$T_1,\ldots,T_5$ are nonsupercritical Vd0 triangles. Then the perimeter is
not covered. No hypothesis on the criticality of $T_0$ is required.
\end{proposition}

\begin{proof}
The two-endpoint inequality, Theorem~\ref{thm:s2-geom-e2}, makes the endpoint
capacities sum to less than one. The three intervening V triangles form the
center-free path required by Lemma~\ref{lem:reader-boundary-path}, giving the
contradiction.
\end{proof}
