\begin{theorem}[Complete support-isolated T3-like endpoint certificate]
\label{thm:paper-complete-t3-endpoint}
For every support-isolated state of
Definition~\ref{def:body-t3-endpoint-state},
\[
 F_{c_5^*}(a_5^*)+F_{c_1^*}(a_1^*)<1.
\]
\end{theorem}

\begin{proof}
If $a_1^*+a_5^*>1$, the direct bounds $F_c(a)\le1-a$ already give the strict
inequality.  Assume $a_1^*+a_5^*\le1$.  If an endpoint radial demand is at
most $1/2$, Proposition~\ref{prop:branchwise-safe-output} makes that endpoint
the Full value $1-a$.  If it is larger than $1/2$,
Proposition~\ref{prop:capped-demand-map} assigns one of the same four labels.
Thus Table~\ref{tab:paper-407-complete-audit} remains an exhaustive partition
of the ordered label pairs, including the low-radial right-Full case.  The
incorporated source in each row proves either that the pair is impossible or
that its two exact branchwise outputs have sum strictly below one.  File
`407d` checks that the geometric residual reduction in `4073` reaches exactly
these branches and no others.  Therefore the displayed endpoint inequality
holds throughout the complete support-isolated domain.
\end{proof}

\subsection{The explicit Vd1--supercritical replacement}

\begin{proposition}[Vd1--supercritical two-chart replacement]
\label{prop:paper-vd1-pair-replacement}
Assume the CE2 branch in which an adjacent Vd1 role and the unique
supercritical Vd0 role lie away from the vertex role at the center's unique
midpoint. Assume the shared boundary edge is center-free, all other vertex
roles are nonsupercritical Vd0, and no additional role has positive support
on the two radial arms of the pair. Then the pair can be replaced by two open
nonsupercritical Vd0 roles while preserving coverage of the full skeleton.
\end{proposition}

\begin{proof}
After a fresh cyclic renumbering, let the Vd1 role be based at $V_0$, the
supercritical role at $V_1$, and let the Vd1 role contain $M_1$. Write $a,b$
for the Vd1 boundary reaches and use the Vd corner parameter $t\ge1$, with
$d=\sqrt{t^2+t+1}$. Put
\[
 c=\frac{d-a-tb}{t+1},\qquad
 \lambda=\frac{t(1-b)}{t+1},\qquad
 u_{\rm adj}=\frac{d-a-tb-1}{t}.
\]
The strict midpoint and one-sided-support inequalities give
\[
 a+c<1,\qquad a<\lambda\le\frac12,
 \qquad b<\frac12,
 \qquad u_{\rm adj}<1-a.
\tag{1}
\]

Let $(A_1,B_1,C_1)$ be the actual reaches of the supercritical role. The
center-free shared edge and radial coverage give
\[
 A_1\ge1-b>\frac12,
 \qquad C_1\ge\lambda.
\]
The selected supercritical admissible cell has $C_1\le1/2$. The half-square
lemma
\[
 (x,y,z)\in\mathcal A,\qquad x\ge\frac12,\qquad z\le\frac12
 \quad\Longrightarrow\quad y\le1-z
\]
therefore gives $B_1\le1-C_1\le1-\lambda$, and hence
\[
 a+B_1<1.
\tag{2}
\]
If $c_1^{\rm req}=1-d_1^C$ is the vertex-side beginning of the center trace
on $r_1$, open skeleton coverage gives
\[
 c_1^{\rm req}<\max\{C_1,u_{\rm adj}\}
 \le\max\{1-B_1,1-a\}.
\tag{3}
\]

Choose $p_2\in(a,1-B_1)$ with
\[
 \max\{p_2,1-p_2\}>c_1^{\rm req};
\]
such a point exists because the supremum of the left side on that interval is
$\max\{1-B_1,1-a\}$. Then choose
\[
 a<p_1<p_2,
 \qquad p_1<\frac12,
 \qquad1-p_1>c.
\]
Finally choose $\varepsilon>0$ smaller than
\[
 p_1-a,\quad p_2-p_1,\quad1-B_1-p_2,\quad1-p_1-c,
 \quad\max\{p_2,1-p_2\}-c_1^{\rm req}.
\tag{4}
\]

Use two distinct local charts:
\[
 X_0(x,y)=V_0+x(V_5-V_0)+y(V_1-V_0),
\]
\[
 X_1(x,y)=V_1+x(V_0-V_1)+y(V_2-V_1).
\]
For $p\le1/2$ put
\[
 \Delta_p^-=\conv\{(0,1-p),(1,1-p),(0,-p)\},
\]
and for $p>1/2$ put
\[
 \Delta_p^+=\conv\{(p,0),(p,1),(p-1,0)\}.
\]
Define
\[
 U_0'=X_0\!\left(\interior(\Delta_{p_1}^-)+(-\varepsilon,0)\right)
\]
and
\[
 U_1'=\begin{cases}
 X_1\!\left(\interior(\Delta_{p_2}^-)+(-\varepsilon,0)\right),
   &p_2\le1/2,\\
 X_1\!\left(\interior(\Delta_{p_2}^+)+(0,-\varepsilon)\right),
   &p_2>1/2.
 \end{cases}
\]

In its own chart the shifted minus template contains the origin and has reach
triple
\[
 (p-\varepsilon,1-p,1-p),
\]
while the shifted plus template contains the origin and has reach triple
\[
 (p,1-p-\varepsilon,p).
\]
Their closures stay strictly below both adjacent-arm lines $x=1$ and $y=1$.
Thus $V_0\in U_0'$, $V_1\in U_1'$, both roles are Vd0, and both boundary
sums equal $1-\varepsilon<1$.

The first replacement reaches beyond $a$ on $e_{5,0}$ and beyond $c$ on
$r_0$. On the shared edge its reach is $1-p_1$, while the second replacement
reaches at least $p_2-\varepsilon$ from $V_1$; their sum exceeds one by (4).
The second replacement reaches beyond $B_1$ on $e_{1,2}$ and has own-radial
reach $\max\{p_2,1-p_2\}>c_1^{\rm req}$ on $r_1$. Hence every boundary and
radial component formerly supplied by the special pair remains covered, and
all other skeleton components are unchanged.

The modified roles therefore cover the full skeleton and have six
nonsupercritical Vd0 vertex roles. Proposition~\ref{prop:nplus-zero-all-vd0}
gives the contradiction.
\end{proof}
