\subsection{Closure of the legacy Strategy 2 references}
\label{subsec:strategy2-cross-reference-closure}

\begin{corollary}[Authoritative replacement for the compact CE2 citations]
\label{cor:paper-strategy2-legacy-closure}
The CE2 exactly-one-Vd1/Vd2 item in
Propositions~\ref{prop:body-demand-branches} and
\ref{prop:reader-demand-branches} is proved by
Proposition~\ref{prop:paper-ce2-one-vd-placements}.  The replacement conclusion
stated in Lemma~\ref{lem:vd1-pair-replacement} is proved in full by
Proposition~\ref{prop:paper-vd1-pair-replacement}.
\end{corollary}

\begin{proof}
The first assertion is the exhaustive placement proof of
Proposition~\ref{prop:paper-ce2-one-vd-placements}.  In its final placement, the
shared edge is center-free and the complete two-ordered-half admissibility,
strict-margin, and open-axis-replacement argument is exactly
Proposition~\ref{prop:paper-vd1-pair-replacement}.  Thus the earlier compact
citations are summaries of these later complete proofs and are not independent
proof dependencies.
\end{proof}
