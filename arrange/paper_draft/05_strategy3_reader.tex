Put
\[
 A_\triangle:=\frac{\sqrt3}{4}.
\]

\begin{definition}[Local feasible family and normalized loss]
\label{def:area-loss-family}
For $(a,b)\in[0,1]^2$, define
\[
 \mathcal T(a,b):=
 \left\{T:\begin{array}{l}
 T\text{ is a closed unit equilateral triangle},\\
 \{V_0,V_0+a(V_5-V_0),V_0+b(V_1-V_0)\}\subseteq T
 \end{array}\right\}.
\]
For every closed unit equilateral triangle $T$, define
\[
 \mathcal L(T):=
 \frac{\operatorname{area}(T\setminus H)}{A_\triangle},
 \qquad
 \mathcal L(a,b):=
 \inf_{T\in\mathcal T(a,b)}\mathcal L(T).
\]
\end{definition}

The corresponding family at $V_i$ is obtained by the hexagon symmetry already
fixed in Section~\ref{sec:introduction}.  In particular, if
\[
 0\le a_i\le A_i,
 \qquad
 0\le b_i\le B_i,
\]
then
\[
 \operatorname{Rot}_{O,-i\pi/3}(T_i)\in\mathcal T(a_i,b_i),
 \qquad
 \mathcal L\bigl(\operatorname{Rot}_{O,-i\pi/3}(T_i)\bigr)
 =\mathcal L(T_i).
\]

\begin{theorem}[Unconditional local square loss]
\label{thm:local-square-loss}
For every $(a,b)\in[0,1]^2$ with $\mathcal T(a,b)\ne\varnothing$,
\begin{equation}
 \mathcal L(a,b)\ge\min\{a,b\}^2.
 \label{eq:min-square-loss}
\end{equation}
If $a+b>1$, then
\begin{equation}
 \mathcal L(a,b)\ge\max\{a,b\}^2.
 \label{eq:max-square-loss}
\end{equation}
\end{theorem}

\begin{proof}
This is the exact optimum comparison proved in
\zcref{lem:app-local-square-loss-calculation}.
\end{proof}

\begin{theorem}[Direct T3-like loss]
\label{thm:t3-direct-loss}
Let $T_i$ be an actual T3-like V triangle.  Then
\begin{equation}
 A_i+B_i\le1.
 \label{eq:t3-area-nonsupercritical}
\end{equation}
Moreover, if
\[
 0\le a_i\le A_i,
 \qquad
 0\le b_i\le B_i,
 \qquad
 0\le\mu\le\frac12,
 \qquad
 a_i,b_i\ge\mu,
\]
then
\begin{equation}
 \mathcal L(T_i)\ge2\mu-4\mu^2.
 \label{eq:t3-loss}
\end{equation}
\end{theorem}

\begin{proof}
The nonsupercritical conclusion is Lemma~\ref{lem:t3-nonsupercritical}; the
actual-triangle loss optimization is
\zcref{lem:app-t3-direct-loss-calculation}.
\end{proof}

\begin{lemma}[Cyclic area-loss certificate]
\label{lem:cyclic-area-loss}
Assume the strict handoff data
\[
 \xi_i\in(1-A_{i+1},B_i),
 \qquad
 a_i:=1-\xi_{i-1}<A_i,
 \qquad
 b_i:=\xi_i<B_i.
\]
Then
\[
 \#\{i:a_i+b_i>1\}\ge2
 \quad\Longrightarrow\quad
 \sum_{i=0}^5\mathcal L(T_i)>1,
\]
and
\[
 \begin{gathered}
 \#\{i:a_i+b_i>1\}=1,\\
 \forall i\;[T_i\text{ is }\Vdzero\text{ or }\Tlike],\\
 \exists j\;[T_j\text{ is }\Tlike]
 \end{gathered}
 \quad\Longrightarrow\quad
 \sum_{i=0}^5\mathcal L(T_i)>1.
\]
\end{lemma}

\begin{proof}
Both exact cyclic minima are proved in
\zcref{lem:app-area-cyclic-calculation}.
\end{proof}

\begin{proposition}[Branches closed by area]
\label{prop:area-branches}
The area-loss mechanism excludes:
\begin{enumerate}
\item $N_{\mathrm{gap}}=0$ and $N_+\ge2$;
\item
      \[
       \begin{gathered}
       N_{\mathrm{gap}}=0,\qquad N_+=1,\\
       \#\{i:T_i\text{ is }\Vdone\text{ or }\Vdtwo\}=0,\qquad
       \#\{i:T_i\text{ is }\Tlike\}\ge1.
       \end{gathered}
      \]
\end{enumerate}
\end{proposition}

\begin{proof}
By Lemma~\ref{lem:gap-exhaustion}, $N_{\mathrm{gap}}=0$ is exactly the
condition that the six open V triangles cover $\partial H$.  Apply the
strict selector of Proposition~\ref{prop:strict-handoffs}.  Its at-least-two
clause and exact-one clause place the two respective cases under the two
implications of Lemma~\ref{lem:cyclic-area-loss}.  Thus the six V triangles
have total normalized area inside $H$ strictly below five; the C triangle has
normalized area one, whereas $H$ has normalized area six.  This contradicts
area subadditivity for a cover.
\end{proof}
