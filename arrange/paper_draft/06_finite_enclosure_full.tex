\section{Finite-Enclosure Obstructions}
\label{sec:finite-enclosure}

The third strategy places explicit points in the part of the hexagon missed by
the six open V triangles.  These points are therefore forced into the open C
triangle.  The contradiction is geometric: their least enclosing equilateral
triangle has side at least one.  No boundary-reach map is iterated in this
section.

\subsection{Trace-exact AB envelopes at an actual gap}
\label{subsec:trace-exact-ab-envelopes}

The endpoints of a displayed V-gap are actual maximal reaches, not selected
lower bounds.  If the gap on $e_{i,i+1}$ is
\[
 J_i=X_i([\ell,r]),
 \qquad X_i(t)=V_i+t(V_{i+1}-V_i),
\]
then
\[
 \boxed{\ell=B_i,\qquad r=1-A_{i+1}.}
 \tag{6.24a}
\]
Consequently the two incident open traces are exactly
\[
 U_i\cap e_{i,i+1}=X_i([0,\ell)),
 \qquad
 U_{i+1}\cap e_{i,i+1}=X_i((r,1]),
 \tag{6.24b}
\]
and the whole closed interval $J_i$, including a singleton, is missed by the
six open V roles and therefore lies in $U_C$.

For visualization and local overapproximation it is useful to condition the
ordinary AB-union on this actual endpoint.  If $S$ is a closed unit
equilateral triangle with $V_i\in\interior S$, let $A_i(S),B_i(S)$ be its
actual maximal backward and forward reaches.  Define
\[
 \mathcal E_i^{\rightarrow}(a\mid b)
 =\bigcup\left\{S\cap\mathcal C_i:
 A_i(S)\ge a,\ B_i(S)=b\right\},
 \tag{6.24c}
\]
\[
 \mathcal E_i^{\leftarrow}(a\mid b)
 =\bigcup\left\{S\cap\mathcal C_i:
 A_i(S)=a,\ B_i(S)\ge b\right\},
 \tag{6.24d}
\]
where the union ranges over such closed unit triangles and
$\mathcal C_i$ is the corner cone at $V_i$.  We call these sets the
\emph{trace-exact AB envelopes}.  They are source-conditioned subunions of
the ordinary AB-set, and their relevant closed edge sections are exactly
\[
 \mathcal E_i^{\rightarrow}(a\mid b)\cap e_{i,i+1}
 =X_i([0,b]),
\]
\[
 \mathcal E_{i+1}^{\leftarrow}(1-r\mid b')\cap e_{i,i+1}
 =X_i([r,1]).
 \tag{6.24e}
\]
Thus the actual incident roles are locally contained in
\[
 \mathcal E_i^{\rightarrow}(a_i\mid\ell),
 \qquad
 \mathcal E_{i+1}^{\leftarrow}(1-r\mid b_{i+1}),
 \tag{6.24f}
\]
while the interval between the two exact traces is center-forced.

The word ``trace-exact'' is essential.  These sets are not obtained by
intersecting an ordinary AB-union with an affine half-plane through the gap
endpoint: a source triangle whose edge trace stops at that endpoint can cross
such a line elsewhere.  The valid operation restricts the source family to
triangles having the prescribed actual maximal reach and then takes their
union.  Since
\[
 \mathcal E_i^{\rightarrow}(a\mid b),\quad
 \mathcal E_i^{\leftarrow}(a\mid b)
 \subseteq \mathcal K_i(a,b),
 \tag{6.24g}
\]
the already proved capacities $c_{\max},C_+,C_-$ remain valid radial upper
bounds.  The proof uses these exact set inclusions; the figures below show
dense samples of valid source triangles only as geometric guides.

\subsection{Radial witnesses}

Suppose $T_i=\overline{U_i}$ contains selected boundary anchors at reaches
$a_i,b_i$.  Its exact own-ray capacity is $c_{\max}(a_i,b_i)$.  If the
preceding role has positive support on $r_i$, let its exact neighboring-ray
capacity be $C_+(a_{i-1},b_{i-1})$; define the reflected capacity $C_-$ for
the following role.  Terms not permitted by the actual V type are omitted.
Put
\[
 \Gamma_i=
 \max\{c_{\max}(a_i,b_i),C_+(a_{i-1},b_{i-1}),
 C_-(a_{i+1},b_{i+1})\}
\]
and
\[
 D_i=(1-\Gamma_i)V_i.
 \tag{6.25}
\]

\begin{lemma}[Type-aware radial forcing]
\label{lem:new-type-aware-radial-forcing}
Every $D_i$ is missed by all six open V roles.  Hence, under a hypothetical
cover,
\[
 \boxed{D_i\in U_C.}
\]
\end{lemma}

\begin{proof}
If $D_i\in U_i$, openness increases the own-radial demand beyond
$c_{\max}(a_i,b_i)$.  If an adjacent role contains $D_i$, openness either
creates forbidden positive adjacent support or increases the exact neighboring
capacity beyond $C_+$ or $C_-$.  For the three nonlocal roles, writing
$D_i=dV_i$ gives
\[
 \lVert dV_i-V_{i\pm2}\rVert^2=1+d+d^2>1,
 \qquad
 \lVert dV_i-V_{i+3}\rVert=1+d>1.
\]
A unit equilateral triangle has diameter one.  Thus every V role misses
$D_i$, and coverage forces it into $U_C$.
\end{proof}

\begin{lemma}[Common-pair domination]
\label{lem:new-common-pair-domination}
Let $p,q\ge0$ and $p+q\le1$.  Put
\[
 c_*=c_{\max}(p,q),
 \qquad m=\min\{p,q\}.
\]
Then
\[
 c_*\ge1-m,
 \qquad
 C_+(p,q),C_-(p,q)\le1-m\le c_*.
 \tag{6.26}
\]
Consequently any actual local pair dominating $(p,q)$ has all permitted
own- and neighboring-ray capacities at most $c_*$.
\end{lemma}

\begin{proof}
Assume $q=m\le p$.  The demand $(p,q,1-q)$ has one exact support side of
length $q+\max\{p,1-q\}=1$, so it is admissible; hence $c_*\ge1-q$.
The exact neighboring-ray formula has linear value $1-q$, a plateau no
larger than $1-q$, and a final radical branch no larger than its plateau.
Thus $C_+(p,q)\le1-q$.  Reflection treats the other order and $C_-$.  The
last assertion follows from coordinatewise antitonicity.
\end{proof}

\subsection{A disk and one boundary point}

\begin{lemma}[Disk-plus-point formula]
\label{lem:new-disk-point-formula}
Let $\mathcal D_\eta$ be the disk of radius $\eta$ about $O$, and let
$\lVert G\rVert=\rho$.  Then
\[
 \Lambda(\mathcal D_\eta\cup\{G\})=
 \begin{cases}
  3\eta/h,&\rho\le2\eta,\\[2mm]
  \dfrac{3\eta+\sqrt{3(\rho^2-\eta^2)}}{2h},&\rho>2\eta.
 \end{cases}
 \tag{6.27}
\]
\end{lemma}

\begin{proof}
For a unit normal $n$, write
$s_j=\langle G,\mathsf R^jn\rangle$.  Then
$s_0+s_1+s_2=0$ and
$\sum s_j^2=3\rho^2/2$.  The support sum is
$\sum\max\{\eta,s_j\}$.  If $\rho\le2\eta$, orient the point projections as
$\rho/2,\rho/2,-\rho$.  If $\rho>2\eta$, an angular minimum occurs when one
point projection equals $\eta$.  The other active projection is
\[
 \frac{-\eta+\sqrt{3(\rho^2-\eta^2)}}2.
\]
Substitution in Proposition~\ref{prop:new-enclosure-gauge} gives (6.27).
\end{proof}

For $p+q\le1$, let
\[
 J(p,q)=[X(q),X(1-p)]\subset e_{0,1},
 \qquad X(t)=V_0+t(V_1-V_0),
\]
and put
\[
 c_*=c_{\max}(p,q),
 \qquad d=1-c_*.
\]

\begin{theorem}[Complementary-gap enclosure]
\label{thm:new-complementary-gap}
\[
 \boxed{
 \Lambda\left(\mathcal D_{hd}\cup J(p,q)\right)\ge1.
 }
 \tag{6.28}
\]
\end{theorem}

\begin{proof}
Put $m=\min\{p,q\}$.  The farther endpoint of $J(p,q)$ has squared norm
\[
 \rho^2=1-m+m^2.
\]
By Lemma~\ref{lem:new-disk-point-formula}, it is enough to prove
\[
 3c_*(1-c_*)\ge m(1-m).
 \tag{6.29}
\]
Lemma~\ref{lem:new-common-pair-domination} gives $c_*\ge1-m\ge1/2$.
If $c_*\le h$, then the left side of (6.29) is at least
$3h(1-h)>1/4$, while $m(1-m)\le1/4$.

Assume $c_*>h$.  In the exact local support formula, let
$x=\cos|t-\pi/6|$.  The pure radial support gives $x\le h/c_*$.  The two
mixed supports imply
\[
 mx+\frac{c_*}{2}
 \left(x+\sqrt3\sqrt{1-x^2}\right)\le h.
\]
The left side is concave in $x$.  Its value at $x=h$ is greater than $h$
unless $c_*=1-m$, in which case (6.29) is immediate.  Hence its value at
$x=h/c_*$ is at most $h$, giving
\[
 m\le \zeta(c_*):=c_*
 \left(\frac12-\sqrt{c_*^2-\frac34}\right).
\]
A direct calculation gives
\[
\begin{aligned}
&3c_*(1-c_*)-\zeta(c_*)(1-\zeta(c_*))\\
&\quad=
\frac{c_*(1-c_*)}{2}
\left(5-2c_*-2c_*^2+2\sqrt{c_*^2-\frac34}\right)\ge0.
\end{aligned}
\]
Since $0\le m\le \zeta(c_*)<1/2$, this proves (6.29).
\end{proof}

\subsection{The CE2 two-gap short-ray theorem}

\begin{theorem}[CE2 short-ray obstruction]
\label{thm:new-ce2-short-ray}
In the strict signed CE2 domain put
\[
 p=W-\alpha,
 \qquad q=R-\delta,
 \qquad e=\min\{\alpha,\delta\}.
\]
Then
\[
 \boxed{c_{\max}(p,q)<1-e.}
 \tag{6.30}
\]
Consequently, whenever both CE2 traces contain V-gaps and the five roles
$T_1,\ldots,T_5$ realize the common pair $(p,q)$, one of the radial witnesses
on $r_2,r_4$ lies beyond the corresponding center exit.
\end{theorem}

\begin{proof}
The CE2 inequalities are
\[
 \alpha+W\delta<P,
 \qquad R\alpha+\delta<P.
\]
Using $RW=\eta+P$, they give
\[
 p,q>\eta+e.
 \tag{6.31}
\]
If $M=\max\{R,W\}$, then
\[
 e<\frac{P}{1+M},
 \qquad
 \frac{\eta}{e}>\frac{1+M}{E}\ge\sqrt3.
\]
Hence
\[
 p,q>(1+\sqrt3)e.
 \tag{6.32}
\]
Moreover
\[
 0<e<e_*:=\frac{2\sqrt3-3}{6}.
 \tag{6.33}
\]
Put $c=1-e$ and
\[
 f_c(t)=c^4-c^2+ct-t^2.
\]
Its roots are
\[
 t_\pm=\frac c2\left(1\pm\sqrt{4c^2-3}\right).
\]
At $t_0=(1+\sqrt3)e$,
\[
 f_{1-e}(t_0)=eB(e),
\]
where
\[
 B(e)=e^3-4e^2-3\sqrt3e+\sqrt3-1.
\]
The function $B$ decreases on $[0,e_*]$ and
\[
 B(e_*)=\frac{302\sqrt3-501}{72}>0.
\]
Thus $t_-<(1+\sqrt3)e<p,q$.  Since
$p+q\le1-2e=c-e$, each of $p,q$ is also below $t_+$.  Therefore
\[
 f_c(p)>0,
 \qquad f_c(q)>0.
 \tag{6.34}
\]
At radial demand $c=1-e$, the four exact support-contact side lengths of
$K(p,q,c)$ are all greater than one: two are $p+c,q+c$, and the other two
exceed one exactly by (6.34).  Thus $c$ is not admissible, proving (6.30).

If $e=\delta$, the point $(1-c_{\max}(p,q))V_2$ lies farther than $\delta$
from $O$, while the center exit on $r_2$ is $\delta$.  If $e=\alpha$, the
analogous point on $r_4$ lies beyond the exit $\alpha$.
\end{proof}

\subsection{Nonzero-gap cases with $N_+=0$}

\begin{proposition}[All-$\Vdzero$ and T3-like gap closures]
\label{prop:new-nplus-zero-gap-closures}
Every nonzero-gap state with $N_+=0$, no Vd1/Vd2 role, and at most two
T3-like roles is impossible.
\end{proposition}

\begin{proof}
For one gap, normalize it to $[X(\ell),X(r)]$.  Strict handoffs on the other
five edges and nonsupercriticality give
\[
 \ell\le x_5\le x_4\le x_3\le x_2\le x_1\le r.
\]
Put $p=1-r$, $q=\ell$, and $c_*=c_{\max}(p,q)$.  Every local boundary pair
dominates $(p,q)$.  Lemmas~\ref{lem:new-type-aware-radial-forcing} and
\ref{lem:new-common-pair-domination} force all six points
$(1-c_*)V_i$ into $U_C$, even when one or two T3-like roles contribute an
adjacent support term.  Their convex hull contains
$\mathcal D_{h(1-c_*)}$.  The gap is $J(p,q)$, so Theorem
\ref{thm:new-complementary-gap} contradicts containment in an open unit
triangle.

For two gaps, the state is CE2.  The four intervening center-free handoffs
give the common pair
\[
 p=W-\alpha,
 \qquad q=R-\delta
\]
on $T_1,\ldots,T_5$.  Type-aware neighboring capacities remain below the
common own-ray capacity.  Theorem~\ref{thm:new-ce2-short-ray} forces one of
the radial witnesses on $r_2,r_4$ beyond the center exit.
\end{proof}

\subsection{One gap, $N_+=1$, all V triangles $\Vdzero$}

Let the gap be $[X(\ell),X(r)]$, and choose strict handoffs
\[
 x_5<x_4<x_3<x_2<x_1<r,
 \qquad x_5<\ell.
\]
The selected pairs are
\[
 (a_0,b_0)=(1-x_5,\ell),
 \qquad
 (a_1,b_1)=(1-r,x_1),
\]
\[
 (a_i,b_i)=(1-x_{i-1},x_i),
 \qquad2\le i\le5.
\]
Let $C_i$ be the actual own-radial reach of $T_i$ and put
\[
 P_i=(1-C_i)V_i.
 \tag{6.35}
\]
Every $P_i$ is missed by all V roles: it is an own-trace endpoint, adjacent
roles are Vd0, and nonlocal roles are excluded by diameter.  Hence
$P_i\in U_C$.

\begin{proposition}[Anisotropic one-gap enclosure]
\label{prop:new-nplus-one-all-vd0}
The compact set
\[
 K_{410}=\{O,M_0,X(\ell),X(r),P_0,\ldots,P_5\}
\]
has
\[
 \boxed{\Lambda(K_{410})\ge1.}
 \tag{6.36}
\]
\end{proposition}

\begin{proof}
Suppose an open unit triangle $U_C'$ contains $K_{410}$.  Together with the
fixed V roles it covers the full skeleton: it covers the gap and, on every
$r_i$, overlaps the V role at $P_i$.  Give $U_C'$ the signed center variables
$R,W,E,\eta,P,\alpha,\delta$, and put
\[
 k=\eta+\alpha+\delta,
 \qquad X=R-\delta,
 \qquad Q=\frac{k}{2R}.
\]
The active trace gives
\[
 B_0\ge\frac{k}{R},
 \qquad A_1\ge X.
 \tag{6.37}
\]
Containment of $P_0$ gives $C_0\ge k$.  The adjacent-edge diameter inequality
therefore gives
\[
 \boxed{A_0<1-Q.}
 \tag{6.38}
\]

Assume first that $U_C'$ is CE2.  The signed inequalities are
\[
 \alpha+W\delta<P,
 \qquad R\alpha+\delta<P.
\]
The low-root estimates give
\[
 \boxed{X>\epsilon(\alpha),
 \qquad
 \min\{\epsilon(\alpha),\epsilon(\delta)\}<Q.}
 \tag{6.39}
\]
The first follows from
$WX>\eta+\alpha$ and the selected quadratic; the second follows by adding
the two signed inequalities after multiplication by $W,R$ and testing the
same quadratic on $[0,(\alpha+\delta)]$.

The five nonsupercritical roles satisfy the direct monotonicity
\[
 A_1\le A_2\le A_3\le A_4\le A_5\le A_0,
 \tag{6.40}
\]
because each center-free edge gives $B_i+A_{i+1}\ge1$ and
$A_i+B_i\le1$.  If $\epsilon(\alpha)<Q$, then
$A_4>\epsilon(\alpha)$; Lemma~\ref{lem:new-direct-threshold} at $T_4$ gives
$B_4\le\epsilon(\alpha)$, hence
$A_5>1-Q$, contradicting (6.38).  Otherwise (6.39) gives
$\epsilon(\delta)<Q$, and
\[
 A_2\ge X>\epsilon(\alpha)\ge Q>\epsilon(\delta).
\]
The threshold at $T_2$ gives $B_2\le\epsilon(\delta)$ and therefore
$A_3>1-Q$, again contradicting (6.38).

If $U_C'$ is CE1, Proposition~\ref{prop:new-ce1-direct-certificate} gives
$A_0>1-Q$, contradicting (6.38).  Thus no open unit triangle contains
$K_{410}$.
\end{proof}
