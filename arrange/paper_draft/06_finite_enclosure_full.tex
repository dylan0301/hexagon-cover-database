\section{Finite-Enclosure Obstructions}
\label{sec:finite-enclosure}

The finite-enclosure strategy forces explicit points or compact sets into the
open C triangle.  Global rows end with an enclosing-side inequality
$\Lambda\ge1$; local rows end when a forced point lies beyond a signed
C-triangle exit.  The simplified proof uses three recurring forms:

\[
 \text{common radial disk}+\text{actual gap},
\]
\[
 \text{two transverse radial points in CE2},
\]
and
\[
 K_{\rm tr}
 =
 \{O,M_0,X(\ell),X(r),\allowbreak P_2,P_3,P_4\}
\]
for the one-gap, one-supercritical all-Vd0 row.

\subsection{Trace-exact AB envelopes at an actual gap}
\label{subsec:trace-exact-ab-envelopes}

If the actual V-gap on $e_{i,i+1}$ is
\[
 J_i=X_i([\ell,r]),
 \qquad X_i(t)=V_i+t(V_{i+1}-V_i),
\]
then
\[
 \boxed{\ell=B_i,\qquad r=1-A_{i+1}.}
 \tag{6.24a}
\]
Consequently
\[
 U_i\cap e_{i,i+1}=X_i([0,\ell)),
 \qquad
 U_{i+1}\cap e_{i,i+1}=X_i((r,1]),
 \tag{6.24b}
\]
and the entire closed interval $J_i$, including a singleton, is
center-forced.

For a closed unit V triangle $S$, define the source-conditioned unions
\[
 \mathcal E_i^{\rightarrow}(a\mid b)
 =
 \bigcup\{S\cap\mathcal C_i:A_i(S)\ge a,\ B_i(S)=b\},
 \tag{6.24c}
\]
\[
 \mathcal E_i^{\leftarrow}(a\mid b)
 =
 \bigcup\{S\cap\mathcal C_i:A_i(S)=a,\ B_i(S)\ge b\}.
 \tag{6.24d}
\]
Their relevant edge sections stop and start exactly at the actual gap
endpoints, and
\[
 \mathcal E_i^{\rightarrow}(a\mid b),\quad
 \mathcal E_i^{\leftarrow}(a\mid b)
 \subseteq\mathcal K_i(a,b).
 \tag{6.24e}
\]
Thus the ordinary capacities remain valid while endpoint ownership is exact.

\subsection{Type-aware radial witnesses}

Suppose $T_i=\overline{U_i}$ contains selected boundary anchors with
$a_i\le A_i$ and $b_i\le B_i$.  Put
\[
 \Gamma_i=
 \max\{c_{\max}(a_i,b_i),C_+(a_{i-1},b_{i-1}),
 C_-(a_{i+1},b_{i+1})\},
\]
omitting neighboring terms forbidden by the actual V types, and define
\[
 D_i=(1-\Gamma_i)V_i.
 \tag{6.25}
\]

\begin{lemma}[Type-aware radial forcing]
\label{lem:new-type-aware-radial-forcing}
Every $D_i$ is missed by all six open V roles and therefore lies in $U_C$.
\end{lemma}

\begin{proof}
Membership in the own role would exceed $c_{\max}$.  Membership in an
adjacent role would create forbidden support or exceed $C_+$ or $C_-$.
For $D_i=dV_i$,
\[
 \|dV_i-V_{i\pm2}\|^2=1+d+d^2>1,
 \qquad
 \|dV_i-V_{i+3}\|=1+d>1,
\]
so nonlocal roles are excluded by diameter one.
\end{proof}

\begin{lemma}[Common-pair domination]
\label{lem:new-common-pair-domination}
Let $p,q\ge0$, $p+q\le1$, and put
\[
 c_*=c_{\max}(p,q),\qquad m=\min\{p,q\}.
\]
Then
\[
 c_*\ge1-m,
 \qquad
 C_+(p,q),C_-(p,q)\le1-m\le c_*.
 \tag{6.26}
\]
The same conclusion holds for every actual pair dominating $(p,q)$.
\end{lemma}

\begin{proof}
Assume $q=m\le p$.  The demand $(p,q,1-q)$ is admissible because the
relevant support side has length $q+\max\{p,1-q\}=1$.  Thus
$c_*\ge1-q$.  Every branch of the exact neighboring formula is at most
$1-q$.  Reflection and coordinatewise antitonicity complete the proof.
\end{proof}

\subsection{A disk and one boundary point}

\begin{lemma}[Disk-plus-point formula]
\label{lem:new-disk-point-formula}
Let $\mathcal D_\eta$ be centered at $O$, and let $\|G\|=\rho$.  Then
\[
 \Lambda(\mathcal D_\eta\cup\{G\})=
 \begin{cases}
  3\eta/h,&\rho\le2\eta,\\[2mm]
  \dfrac{3\eta+\sqrt{3(\rho^2-\eta^2)}}{2h},&\rho>2\eta.
 \end{cases}
 \tag{6.27}
\]
\end{lemma}

\input{figures/finite_enclosure/fe02_disk_plus_point}

\begin{proof}
Apply Proposition~\ref{prop:new-disk-finite-caliper}.  In the disk-only
regime the support sum is $3\eta$.  Otherwise a point--disk tie gives the
other active projection
\[
 \frac{-\eta+\sqrt{3(\rho^2-\eta^2)}}2.
\]
\end{proof}

For $p+q\le1$, put
\[
 J(p,q)=[X(q),X(1-p)]\subset e_{0,1},
 \qquad
 c_*=c_{\max}(p,q),\qquad d=1-c_*.
\]

\begin{theorem}[Complementary-gap enclosure]
\label{thm:new-complementary-gap}
\[
 \boxed{
 \Lambda(\mathcal D_{hd}\cup J(p,q))\ge1.
 }
 \tag{6.28}
\]
\end{theorem}

\input{figures/finite_enclosure/fe03_complementary_gap}

\begin{proof}
Let $m=\min\{p,q\}$.  The farther endpoint of the gap has squared norm
$1-m+m^2$.  Lemma~\ref{lem:new-disk-point-formula} reduces the claim to
\[
 3c_*(1-c_*)\ge m(1-m).
 \tag{6.29}
\]
Common-pair domination gives $c_*\ge1-m\ge1/2$.  If $c_*\le h$, the left
side is at least $3h(1-h)>1/4$.  If $c_*>h$, the exact local support
inequalities give
\[
 m\le
 c_*\left(\frac12-\sqrt{c_*^2-\frac34}\right)=:\zeta(c_*),
\]
and direct calculation gives
\[
 3c_*(1-c_*)-\zeta(c_*)(1-\zeta(c_*))\ge0.
\]
\end{proof}

\subsection{The shortened CE2 two-gap theorem}

\begin{theorem}[CE2 short-ray obstruction]
\label{thm:new-ce2-short-ray}
In the strict signed CE2 domain put
\[
 p=W-\alpha,\qquad q=R-\delta,\qquad
 e=\min\{\alpha,\delta\}.
\]
Then
\[
 \boxed{c_{\max}(p,q)<1-e.}
 \tag{6.30}
\]
\end{theorem}

\input{figures/finite_enclosure/fe04_ce2_short_ray}

\begin{proof}
Let $M=\max\{R,W\}$.  The two CE2 inequalities imply
\[
 P>(1+M)e.
\]
Since $E^2=1-M+M^2$,
\[
 (1+M)^2-(3M-1)^2E^2
 =
 3M(1-M)(3M^2-2M+3)>0,
\]
so $\eta=P/E>(3M-1)e$.  Using $RW=\eta+P$ gives
\[
 Wq>\eta+e>3Me,\qquad Rp>\eta+e>3Me.
\]
Hence
\[
 3e<p,q<1-5e.
 \tag{6.31}
\]
Also
\[
 e<\frac{P}{1+M}
 \le\frac{2\sqrt3-3}{6}<\frac1{12}.
 \tag{6.32}
\]

Set $c=1-e$ and $f_c(t)=c^4-c^2+ct-t^2$.  This function is concave and
\[
 f_c(3e)=e(1-7e-4e^2+e^3)>0,
\]
\[
 f_c(1-5e)=e(2-15e-4e^2+e^3)>0.
\]
Thus $f_c(p),f_c(q)>0$.  Since $p+c,q+c>1$ and $c>p+q$, all four exact
support-contact side lengths at radial demand $c$ exceed one.  Therefore
$c$ is inadmissible.

If $e=\delta$, the radial witness on $r_2$ lies beyond the exit $\delta$.
If $e=\alpha$, the witness on $r_4$ lies beyond the exit $\alpha$.
\end{proof}

\subsection{Nonzero-gap cases with $N_+=0$}

\begin{proposition}[All-Vd0 and T3-like gap closures]
\label{prop:new-nplus-zero-gap-closures}
Every nonzero-gap state with $N_+=0$, no Vd1/Vd2 role, and at most two
T3-like roles is impossible.
\end{proposition}

\begin{proof}
For one gap, strict handoffs give a common pair $(p,q)$ on all six roles.
Type-aware forcing puts the regular radial hexagon with radius
$1-c_{\max}(p,q)$ in $U_C$, hence also its incircle.  The actual gap is
$J(p,q)$, so Theorem~\ref{thm:new-complementary-gap} applies.

For two gaps, the state is CE2.  The common pair is
$(W-\alpha,R-\delta)$ on $T_1,\ldots,T_5$.  Common-pair domination controls
the permitted T3-like neighboring terms, and
Theorem~\ref{thm:new-ce2-short-ray} places $D_2$ or $D_4$ beyond the C exit.
\end{proof}

\subsection{The one-gap all-Vd0 case}

Let the actual gap be $[X(\ell),X(r)]\subset e_{0,1}$, and let $C_i$ be the
actual own-radial reach of $T_i$.  Put
\[
 P_i=(1-C_i)V_i.
\]

\begin{lemma}[The transverse witness is center-forced]
\label{lem:readable-k410-forced}
Every $P_i$ is missed by all six open V roles.  In particular,
\[
 K_{\rm tr}
 =
 \{O,M_0,X(\ell),X(r),\allowbreak P_2,P_3,P_4\}
 \subset U_C.
 \tag{6.33}
\]
\end{lemma}

\input{figures/finite_enclosure/fe05_k410_actual_reach}

\begin{proof}
Each $P_i$ is the closed own-trace endpoint.  Adjacent roles are Vd0 and the
nonlocal roles fail by diameter.  The actual gap is also center-forced.
\end{proof}

\begin{lemma}[Endpoint upper squeeze]
\label{lem:readable-k410-upper-bound}
If an open unit triangle $U'$ contains $K_{\rm tr}$, give it signed center
variables and put
\[
 X=R-\delta,\qquad Q=\frac{\eta+\alpha+\delta}{2R}.
\]
Then
\[
 \boxed{A_0<1-Q.}
 \tag{6.34}
\]
\end{lemma}

\begin{proof}
The gap endpoints lie strictly inside the trace of $U'$, so
\[
 B_0=\ell>2Q,\qquad A_1=1-r>X.
\]
The diameter inequality for the two boundary endpoints of $T_0$ gives
\[
 A_0
 \le M_0(B_0)
 <M_0(2Q)
 =-Q+\sqrt{1-3Q^2}
 <1-Q.
\]
No radial point at $r_0$ is used.
\end{proof}

\begin{proposition}[CE2 transverse threshold terminal]
\label{prop:readable-k410-ce2}
No CE2 open unit triangle contains $K_{\rm tr}$.
\end{proposition}

\begin{proof}
Nonsupercriticality and the five V-gap-free edges give
\[
 A_1<A_2<A_3<A_4<A_5<A_0.
\]
The preceding lemma therefore implies $B_2,B_4>Q$.  Containment of
$P_2,P_4$ gives
\[
 C_2>1-\delta,\qquad C_4>1-\alpha.
\]
The signed CE2 estimates are
\[
 X>\epsilon(\alpha),\qquad
 \min\{\epsilon(\alpha),\epsilon(\delta)\}<Q.
\]
If $\epsilon(\alpha)<Q$, the threshold at $T_4$ gives
$B_4<Q$.  Otherwise the threshold at $T_2$ gives $B_2<Q$.
Both alternatives contradict the preceding lower bounds.
\end{proof}

\begin{proposition}[CE1 three-transverse return]
\label{prop:readable-k410-ce1}
No CE1 open unit triangle contains $K_{\rm tr}$.
\end{proposition}


\begin{proof}
Containment of $P_4,P_3,P_2$ supplies, in order,
\[
 C_4>1-\alpha,\qquad
 C_3>1-\frac{\alpha}{R},\qquad
 C_2>1-\delta.
\]
These are exactly the radial hypotheses used by
Proposition~\ref{prop:new-ce1-direct-certificate}, which proves
$A_0>1-Q$.  This contradicts Lemma~\ref{lem:readable-k410-upper-bound}.
\end{proof}

\begin{proposition}[Transverse seven-point enclosure]
\label{prop:new-nplus-one-all-vd0}
\[
 \boxed{\Lambda(K_{\rm tr})\ge1.}
 \tag{6.35}
\]
Consequently the former ten-point superset $K_{410}$ also has enclosure
number at least one.
\end{proposition}

\begin{proof}
Every candidate open unit triangle is CE1 or CE2, and the preceding two
propositions exclude both.  A compact subset of an open unit triangle would
have enclosure number below one.
\end{proof}

For two gaps, Theorem~\ref{thm:new-ce2-short-ray} applies independently of
the criticality of $T_0$.

\label{prop:finite-enclosure-nonzero-gap-branches}

impossible.

\end{proof}
