\subsection{Enclosure numbers and local capacities}

\begin{definition}[Equilateral enclosure number]
\label{def:finite-enclosure-number}
For $X\in\mathbb R^2$ and $r\ge0$, put
\[
 \mathbb B(X,r):=\{Y\in\mathbb R^2:\lVert Y-X\rVert\le r\}.
\]
For every nonempty compact $K\subset\mathbb R^2$, put
\[
 \Lambda(K):=
 \inf\left\{\operatorname{side}(Q):
 \begin{array}{l}
 Q\text{ is a closed equilateral triangle},\\
 K\subseteq Q
 \end{array}
 \right\}.
\]
The support-function evaluation of this infimum is
\zcref{lem:app-shared-enclosure-gauge}.
\end{definition}

\begin{lemma}[Compact set in an open unit triangle]
\label{lem:new-compact-open-shrink}
\[
 \left.
 \begin{gathered}
 K\subset\mathbb R^2\text{ is compact},\qquad K\subset U,\\
 U\text{ is an open unit equilateral triangle}
 \end{gathered}
 \right\}
 \Longrightarrow \Lambda(K)<1.
\]
\end{lemma}

\begin{proof}
The distance from $K$ to each side line of $U$ is positive.  Moving the three
side lines inward through a common smaller distance leaves a closed
equilateral triangle of side strictly below one that still contains $K$.
\end{proof}

Normalize at $V_0$ and use the established radial map $R_i$.

\begin{definition}[Anchor hull and radial capacity]
\label{def:finite-anchor-admissible-set}
For $(a,b,c)\in[0,1]^3$, define
\[
 K(a,b,c):=\conv\{V_0,X_5(1-a),X_0(b),R_0(c)\},
\]
\[
 \mathcal A:=\{(a,b,c)\in[0,1]^3:\Lambda(K(a,b,c))\le1\}.
\]
For
\[
 0\le a,b\le1,
 \qquad a^2+ab+b^2\le1,
\]
put
\[
 c_{\max}(a,b):=\max\{c\in[0,1]:(a,b,c)\in\mathcal A\}.
\]
\end{definition}

\begin{proposition}[Projection and symmetry of the anchor capacity]
\label{prop:finite-anchor-capacity-interface}
One has
\[
 \{(a,b)\in[0,1]^2:\exists c\in[0,1]\ ((a,b,c)\in\mathcal A)\}
 =\{(a,b)\in[0,1]^2:a^2+ab+b^2\le1\},
\]
and the maximum in Definition~\ref{def:finite-anchor-admissible-set} is
attained.  Moreover,
\[
 \left.
 \begin{gathered}
 0\le p,q\le1,\\
 p^2+pq+q^2\le1
 \end{gathered}
 \right\}
 \Longrightarrow c_{\max}(p,q)=c_{\max}(q,p).
\]
\end{proposition}

\begin{proof}
These statements are proved by the solved local enclosure problem
\zcref{lem:app-local-admissible-set}.
\end{proof}

\begin{definition}[Neighboring-radial capacities]
\label{def:finite-neighbor-capacities}
For $0\le a,b\le1$ and $a+b\le1$, put
\[
 C_+(a,b):=
 \max\left\{c\in[0,1]:
 \begin{array}{l}
 \exists\Delta\ [\Delta\text{ is a closed unit equilateral triangle},\\
 \{V_0,X_5(1-a),X_0(b),R_1(c)\}\subseteq\Delta]
 \end{array}
 \right\},
\]
\[
 C_-(a,b):=C_+(b,a).
\]
The maxima are attained and evaluated in
\zcref{prop:app-neighbor-ray-formula}.
\end{definition}

\begin{definition}[Type-aware radial witness]
\label{def:finite-type-aware-witness}
Fix $i\in\mathbb Z/6\mathbb Z$ and selected reaches satisfying
\[
 0\le a_j\le A_j,
 \qquad 0\le b_j\le B_j
 \qquad(0\le j\le5),
\]
\[
 a_j^2+a_jb_j+b_j^2\le1
 \qquad(0\le j\le5),
\]
\[
 \begin{aligned}
 \Haus(T_{i-1}\cap r_i)>0&\Longrightarrow a_{i-1}+b_{i-1}\le1,\\
 \Haus(T_{i+1}\cap r_i)>0&\Longrightarrow a_{i+1}+b_{i+1}\le1.
 \end{aligned}
\]
Define
\[
\begin{aligned}
 \Gamma_i:=\max\Bigl(&\{c_{\max}(a_i,b_i)\}\\
 &\cup\{C_+(a_{i-1},b_{i-1}):\Haus(T_{i-1}\cap r_i)>0\}\\
 &\cup\{C_-(a_{i+1},b_{i+1}):\Haus(T_{i+1}\cap r_i)>0\}\Bigr),
 \qquad D_i:=(1-\Gamma_i)V_i.
\end{aligned}
\]
\end{definition}

\begin{lemma}[Type-aware radial forcing]

Consequently,

\end{lemma}

\begin{proposition}[One-gap common-enclosure terminal]
