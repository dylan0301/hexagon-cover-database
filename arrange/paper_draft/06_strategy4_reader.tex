Under a hypothetical cover, the zero-gap branch considered here is
\[
 N_{\mathrm{gap}}=0,
 \qquad N_+=1,
 \qquad (d,t)=(0,0).
\]
Rotate the unique supercritical role to index~$4$.  Then
\begin{equation}
 \xi_3<\xi_2<\xi_1<\xi_0<\xi_5<\xi_4,
 \label{eq:reader-extreme-chain}
\end{equation}
and, on putting
\begin{equation}
 a:=1-\xi_3,
 \qquad b:=\xi_4,
 \qquad p:=1-b,
 \qquad q:=1-a,
 \label{eq:reader-ab-parameters}
\end{equation}
one has
\begin{equation}
 \begin{gathered}
 0<a,b<1,
 \qquad a=a_4<A_4,
 \qquad b=b_4<B_4,\\
 a+b>1,
 \qquad a^2+ab+b^2<1,
 \qquad
 a_i\ge p,
 \quad b_i\ge q
 \quad(0\le i\le5).
 \end{gathered}
 \label{eq:reader-ab-domain}
\end{equation}
These relations are supplied by \zcref{lem:ab-extreme-jump}.

Define the two boundary anchors
\[
 P_A:=V_4+a(V_3-V_4),
 \qquad
 P_B:=V_4+b(V_5-V_4),
\]
and the feasible union
\[
 \mathcal R_{AB}(a,b):=
 \bigcup\left\{
 T\cap H:
 \begin{array}{l}
 T\text{ is a closed unit equilateral triangle},\\
 V_4,P_A,P_B\in T
 \end{array}
 \right\}.
\]
Using the previously defined radial envelope $c_{\max}$, put
\[
 c_\ast:=c_{\max}(p,q),
 \qquad
 h:=\frac{\sqrt3}{2},
 \qquad
 r_\ast:=h(1-c_\ast),
\]
\[
 P_i^{\mathrm{rad}}:=(1-c_\ast)V_i
 \quad(0\le i\le5),
 \qquad
 \mathbb B(0,r_\ast):=\{X:\lVert X\rVert\le r_\ast\}.
\]
The direct radial forcing lemma gives
\begin{equation}
 \mathbb B(0,r_\ast)
 \subseteq
 \conv\{P_0^{\mathrm{rad}},\ldots,P_5^{\mathrm{rad}}\}
 \subset U_C.
 \label{eq:reader-forced-disk}
\end{equation}
Indeed, this is precisely \zcref{lem:symmetric-core-witness}.

For the asymmetric witnesses, define the two adjacent-handoff circles by
\[
 \mathcal C_-:=\partial\mathbb B(X_2(b),1),
 \qquad
 \mathcal C_+:=\partial\mathbb B(X_5(1-a),1).
\]
The corresponding maximal nondegenerate line pieces of the interior
frontier are defined intrinsically by
\[
 \{\ell_-\}:=
 \left\{
 [X,Y]:
 \begin{array}{l}
 X,Y\in\mathbb R^2,\quad X\ne Y,\quad
 [X,Y]\subseteq
 \partial\mathcal R_{AB}(a,b)\cap\interior(H),\\
 {}[X,Y]\cap\mathcal C_-\ne\varnothing,\\
 \nexists X',Y'\in\mathbb R^2\quad
 {}[X,Y]\subsetneq[X',Y']
 \subseteq\partial\mathcal R_{AB}(a,b)\cap\interior(H)
 \end{array}
 \right\},
\]
\[
 \{\ell_+\}:=
 \left\{
 [X,Y]:
 \begin{array}{l}
 X,Y\in\mathbb R^2,\quad X\ne Y,\quad
 [X,Y]\subseteq
 \partial\mathcal R_{AB}(a,b)\cap\interior(H),\\
 {}[X,Y]\cap\mathcal C_+\ne\varnothing,\\
 \nexists X',Y'\in\mathbb R^2\quad
 {}[X,Y]\subsetneq[X',Y']
 \subseteq\partial\mathcal R_{AB}(a,b)\cap\interior(H)
 \end{array}
 \right\}.
\]
Finally, define the three frontier points by
\[
 \{Q_0\}:=\ell_-\cap\ell_+,
 \qquad
 \{Q_-\}:=\ell_-\cap\mathcal C_-,
 \qquad
 \{Q_+\}:=\ell_+\cap\mathcal C_+.
\]
Their existence and uniqueness follow from
\zcref{thm:strict-ab-union,lem:fixed-line-signs}; their direct exclusion
from all six V triangles is \zcref{lem:asymmetric-core-witness}.  Thus
\begin{equation}
 Q_-,Q_0,Q_+
 \in\interior(H)\setminus\bigcup_{i=0}^5U_i,
 \qquad
 Q_-,Q_0,Q_+\in U_C.
 \label{eq:reader-forced-asymmetric}
\end{equation}

The public compact witness is
\begin{equation}
 \begin{gathered}
 K_{\rm wit}(a,b):=
 \mathbb B(0,r_\ast)\cup\{Q_-,Q_0,Q_+\},\\
 K_{\rm wit}(a,b)\subset U_C.
 \end{gathered}
 \label{eq:reader-witness-set}
\end{equation}

\begin{theorem}[Witness enclosure]
\label{thm:reader-witness-enclosure}
For every pair $(a,b)$ satisfying \eqref{eq:reader-ab-domain},
\[
 \Lambda\bigl(K_{\rm wit}(a,b)\bigr)\ge1.
\]
\end{theorem}

\begin{proof}
This is the zero-gap optimization proved in
\zcref{thm:zero-gap-enclosure-optimization}.
\end{proof}

\begin{theorem}[Zero-gap obstruction]
\label{thm:reader-zero-gap-obstruction}
There is no cover of $H$ by open unit equilateral triangles
$U_C,U_0,\ldots,U_5$ for which
\[
 N_{\mathrm{gap}}=0,
 \qquad N_+=1,
 \qquad(d,t)=(0,0).
\]
\end{theorem}

\begin{proof}
Equations \eqref{eq:reader-forced-disk} and
\eqref{eq:reader-forced-asymmetric} give the compact containment
$K_{\rm wit}(a,b)\subset U_C$, whereas
\zcref{thm:reader-witness-enclosure} gives
$\Lambda(K_{\rm wit}(a,b))\ge1$.  Every compact subset of an open unit
equilateral triangle lies in a closed equilateral triangle of side strictly
less than one, which would give $\Lambda(K_{\rm wit}(a,b))<1$.
\end{proof}

\begin{proposition}[Cases excluded by the nine-point obstruction]
\label{prop:reader-ab-core-branches}
Every branch satisfying
\[
 N_{\mathrm{gap}}=0,
 \qquad N_+=1,
 \qquad(d,t)=(0,0)
\]
is impossible, independently of the center class.
\end{proposition}

\begin{proof}
This is exactly \zcref{thm:reader-zero-gap-obstruction}.
\end{proof}
