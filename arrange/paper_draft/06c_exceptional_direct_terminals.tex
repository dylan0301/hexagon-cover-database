\subsection{Exactly one T3-like role}

\begin{proposition}[Direct T3-like rescuer witness]
\label{prop:new-one-t3-terminal}
Assume $N_+=1$, exactly one V triangle is T3-like, no V triangle is
Vd1/Vd2, and at least one V-gap is present.  Then the skeleton is not covered.
\end{proposition}

\input{figures/finite_enclosure/fe08_t3_rescuer}

\begin{proof}
Normalize the center midpoint to $M_0$.  Midpoint locality forces, after
reflection,
\[
 T_0\text{ T3-like},\qquad M_1\in T_0,\qquad
 T_1\text{ uniquely supercritical},
\]
while $T_2,T_3,T_4,T_5$ are nonsupercritical Vd0 roles.

Write the T3-like interval on $r_1$, measured from $V_1$ toward $O$, as
$[c,u]$, with $c\le1/2\le u$, and let $a$ be its boundary reach on
$e_{5,0}$.  The O-side endpoint
\[
 P_T=(1-u)V_1
 \tag{6.41}
\]
is missed by the open T3-like role, the adjacent supercritical role, and all
Vd0 or nonlocal roles.  Hence
\[
 \boxed{P_T\in U_C.}
 \tag{6.42}
\]

The translated T3-like normal form has
\[
 R_0=\frac{D+\sqrt{4-3D^2}}2,\qquad
 x=\frac{aD}{R_0},\qquad
 \theta=\frac{D-1}{R_0},
\]
and
\[
 c=x+\theta,\qquad
 \frac{1-4\theta+\theta^2}{2(1-2\theta)}
 \le x\le\frac12-\theta.
 \tag{6.43}
\]
Let $M=M_c^{\rm sup}$.  For
$s(z)=z(2-z)/(1+z)$, the equation $s(1-M)=c$ and the quadratic
\[
 2x^2+(\theta-1)x+\theta\ge0
\]
give
\[
 a\le x\le1-M,
 \qquad
 \frac{a}{a+1-u}=x\le1-M.
 \tag{6.44}
\]
Moreover $a+1-u=R_0/D\le1$.  Thus the one-gap branch is exactly
Lemma~\ref{lem:new-rescuer-tail-budget}.

If there are two gaps, the center is CE2.  The common pair
$(W-\alpha,R-\delta)$ controls the permitted T3-like neighboring support,
and Theorem~\ref{thm:new-ce2-short-ray} places $D_2$ or $D_4$ beyond the
corresponding C exit.
\end{proof}

\subsection{Exactly one Vd1/Vd2 role in CE2}

\begin{proposition}[Direct one-Vd placement assembly]
\label{prop:new-one-vd-assembly}
Assume CE2, $N_+=1$, exactly one Vd1/Vd2 role, and at least one V-gap.  Then
the skeleton is not covered.
\end{proposition}

\input{figures/finite_enclosure/fe09_vd_placements}

\begin{proof}
Let $T_\sigma$ be the supercritical role and $T_\tau$ the Vd1/Vd2 role.
They are distinct.  If a further role has positive adjacent support, the
skeleton-length theorem applies.  Assume every other role is Vd0.

\smallskip
\noindent\emph{Case 1: $\sigma=0$ and $\tau$ is adjacent.}
Reflect to $\tau=1$.  Let $\rho_R,\rho_L$ be the residual boundary reaches.
The exact residual calculation gives
\[
 \rho_R+\rho_L<\frac12,
 \qquad
 4\delta<\rho_L.
 \tag{6.45}
\]
The Vd supported endpoint is below $1-\rho_L<1-\delta$.  At $T_2$,
\[
 A_2>\frac12+\rho_R,\qquad B_2\ge\rho_L.
\]
Lemma~\ref{lem:new-one-third-radial-envelope} gives
\[
 C_2<
 1-\frac{\rho_L}{3}
 <1-\delta.
 \tag{6.46}
\]
Thus neither local V trace reaches the C interval on $r_2$, and a point is
uncovered.

\smallskip
\noindent\emph{Case 2: $\sigma=0$ and $\tau\in\{2,3,4\}$.}
The exact residual and diameter estimates give
\[
 \alpha+\delta<\min\{\rho_R,\rho_L\}.
 \tag{6.47}
\]
Every relevant C exit is at most $\alpha+\delta$, while the Vd own-radial
margin stops before
\[
 D_\tau=\min\{\rho_R,\rho_L\}V_\tau.
\]
The adjacent roles are Vd0 and the nonlocal roles fail by diameter, so
$D_\tau$ is uncovered.

\smallskip
\noindent\emph{Case 3: $\tau=0$.}
Midpoint rescue forces $\sigma\in\{1,5\}$; reflect to $\sigma=1$.
If $T_0$ is Vd2, the perimeter terminal gives
\[
 \frac12+\frac13+\frac2{\sqrt3}+4<6.
\]
If $T_0$ is Vd1, write its supported interval as $[c,u]$ and put
$\varepsilon=1-u$.  The O-side endpoint $\varepsilon V_1$ is center-forced.
The exact Vd1 calculation gives, with $M=M_c^{\rm sup}$,
\[
 a\le1-M,\qquad
 \frac{a}{a+\varepsilon}\le1-M.
 \tag{6.48}
\]
The Vd half-unit cap and $u\ge1/2$ give $a+\varepsilon<1$.
Lemma~\ref{lem:new-rescuer-tail-budget} gives the contradiction.

\smallskip
\noindent\emph{Case 4: $\sigma\ne0$ and $\tau\ne0$.}
Midpoint rescue makes the two roles adjacent.  Vd2 uses the preceding
perimeter terminal.  For Vd1, the corrected two-chart replacement preserves
the covered skeleton and produces six nonsupercritical Vd0 roles.  Recompute
the output gap rank: rank zero uses the boundary-complete length theorem,
rank one the complementary-gap theorem, and rank two the shortened CE2
short-ray theorem.  No preservation of the input gap rank is asserted.

The four cases are exhaustive.
\end{proof}
