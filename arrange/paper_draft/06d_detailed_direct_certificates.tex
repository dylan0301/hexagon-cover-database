\subsection{Detailed direct certificates for the nonzero-gap terminals}
\label{subsec:new-detailed-direct-certificates}

This subsection records the calculations suppressed by the case proofs.  The
new CE2 short-ray calculation replaces the former root-separation argument;
the remaining residual and local normal-form calculations are unchanged.

\subsubsection{The disk--gap support calculation}

Let $c=c_{\max}(p,q)$, $m=\min\{p,q\}$, and $d=1-c$.  The regular radial
hexagon has inradius $hd$, while the farther endpoint $G$ of the
complementary gap has
\[
 \|G\|^2=1-m+m^2.
\]
The disk-plus-point formula shows that
\[
 \Lambda(\mathcal D_{hd}\cup\{G\})\ge1
\]
is equivalent in the nontrivial branch to
\[
 3d(1-d)\ge m(1-m).
 \tag{6.51}
\]
For $c>h$, the exact local support inequalities imply
\[
 m\le c\left(\frac12-\sqrt{c^2-\frac34}\right).
 \tag{6.52}
\]
Substitution gives
\[
\begin{aligned}
&3c(1-c)-
 c\left(\frac12-\sqrt{c^2-\frac34}\right)
 \left[
 1-c\left(\frac12-\sqrt{c^2-\frac34}\right)
 \right]\\
&\quad=
\frac{c(1-c)}2
\left(5-2c-2c^2+2\sqrt{c^2-\frac34}\right)\ge0.
\end{aligned}
\]
This proves the complementary-gap inequality.

\subsubsection{The CE2 one-gap threshold inequalities}

Put
\[
 X=R-\delta,
 \qquad
 Q=\frac{\eta+\alpha+\delta}{2R}.
\]
The right-trace inequality gives
\[
 WX=RW-W\delta=\eta+P-W\delta>\eta+\alpha.
 \tag{6.53}
\]
The selected low root $\epsilon(\alpha)$ is the smaller root of
\[
 z^2-(1-\alpha)z+(1-\alpha)^2-(1-\alpha)^4=0.
\]
Testing this quadratic at the lower bound in (6.53) is equivalent to the
positivity of
\[
 \pi(q)=(\eta+q)(1-2q)-2qW.
\]
The function is concave and
\[
 \pi(0)=\eta>0,
 \qquad
 \pi(P)=\eta(\eta+2ER^2)>0.
\]
Thus
\[
 \boxed{\epsilon(\alpha)<X.}
 \tag{6.54}
\]

For the second inequality put $T=\alpha+\delta$.  Multiplying the two CE2
inequalities by $W,R$ and adding gives
\[
 E^2T<P=E\eta,
 \qquad
 T<\frac{\eta}{E}.
 \tag{6.55}
\]
If $d=\min\{\alpha,\delta\}$, then
\[
 \min\{\epsilon(\alpha),\epsilon(\delta)\}
 <
 \frac{2d}{1-2d}
 \le\frac{T}{1-T}.
\]
The comparison $T/(1-T)<Q$ is equivalent to
\[
 \chi(T)=(\eta+T)(1-T)-2RT>0.
\]
The function $\chi$ is concave,
\[
 \chi(0)=\eta>0,
 \qquad
 \chi\left(\frac{\eta}{E}\right)
 =
 \frac{\eta}{E^2}(E-R)^2>0.
\]
Hence
\[
 \boxed{
 \min\{\epsilon(\alpha),\epsilon(\delta)\}<Q.
 }
 \tag{6.56}
\]

\subsubsection{The shortened CE2 two-gap calculation}

Let
\[
 p=W-\alpha,\qquad q=R-\delta,\qquad
 e=\min\{\alpha,\delta\},\qquad M=\max\{R,W\}.
\]
The strict CE2 inequalities give
\[
 P>(1+M)e.
\]
Since $E^2=1-M+M^2$,
\[
 (1+M)^2-(3M-1)^2E^2
 =
 3M(1-M)(3M^2-2M+3)>0.
\]
Thus $\eta=P/E>(3M-1)e$.  Now
\[
 Wq>\eta+e>3Me,
 \qquad
 Rp>\eta+e>3Me,
\]
so
\[
 p,q>3e.
\]
Together with $p+q\le1-2e$, this gives
\[
 \boxed{3e<p,q<1-5e.}
 \tag{6.57}
\]
Furthermore
\[
 e<\frac{P}{1+M}
 \le\frac{2\sqrt3-3}{6}<\frac1{12}.
 \tag{6.58}
\]

For $c=1-e$, put $f_c(t)=c^4-c^2+ct-t^2$.  Concavity and
\[
 f_c(3e)=e(1-7e-4e^2+e^3)>0,
\]
\[
 f_c(1-5e)=e(2-15e-4e^2+e^3)>0
\]
give $f_c(p),f_c(q)>0$.  Since $c>p+q$ and $p+c,q+c>1$, the exact four
support-contact side lengths all exceed one.  Hence
$c_{\max}(p,q)<1-e$.

\subsubsection{Detailed adjacent Vd residual estimate}

In the adjacent placement, the residuals satisfy
\[
 \rho_R+\rho_L<\frac12.
 \tag{6.59}
\]
The signed CE2 domain gives
\[
 \alpha+\delta<\frac16\min\{R,W\}.
 \tag{6.60}
\]
If $\rho_L=W-\alpha$, then
\[
 4\delta+\alpha
 \le4(\alpha+\delta)
 <\frac{2W}{3}<W,
\]
so $4\delta<\rho_L$.

Otherwise $\rho_L=1-A_0$.  The opposite residual must be
\[
 \rho_R=1-(W+\delta),
 \qquad
 B_0\ge\frac{k}{R}.
\]
The boundary-endpoint diameter inequality gives
\[
 \rho_L>\frac{k}{2R}>\frac{\eta}{2R}.
 \tag{6.61}
\]
Combining this with (6.59) gives
\[
 \delta>R-\frac12+\frac{\eta}{2R}=:J(R).
\]
The function $J$ is increasing, $J(3/8)=1/24$, and (6.60) gives
$\delta<1/24$.  Thus $R<3/8$.  Then
\[
 (2-3R)^2-E^2=(3-8R)(1-R)>0,
\]
so $\eta/R>1/3$, and (6.61) yields
\[
 \rho_L>\frac16>4\delta.
 \tag{6.62}
\]
This proves the residual estimate used by the adjacent Vd terminal.  The
new one-third radial envelope then gives
\[
 C_2<1-\frac{\rho_L}{3}<1-\delta.
\]

\subsubsection{Detailed nonadjacent Vd separation}

Let $x=k/W$ and $y=k/R$ be the two center near endpoints, and let $B,U$ be
the farthest already-covered extents.  The endpoint-distance inequality and
monotonicity of $M_0$ give
\[
 B\le M_0(x),
 \qquad
 U\le M_0(y).
 \tag{6.63}
\]
Since
\[
 1-M_0(z)>\frac z2
 \qquad(z>0),
\]
the residual tails satisfy
\[
 \rho_R>\frac x2,
 \qquad
 \rho_L>\frac y2.
\]
The signed center-slack estimate
\[
 \alpha+\delta<\frac12\min\{x,y\}
\]
therefore gives
\[
 \alpha+\delta<\min\{\rho_R,\rho_L\}.
 \tag{6.64}
\]
The Vd own-radial margin then places
$\min\{\rho_R,\rho_L\}V_\tau$ beyond the Vd trace, while (6.64) places it
beyond the C exit.

\subsubsection{Detailed Vd1 rescue inequalities}

Use the Vd1 corner parameter $t\ge1$, let
$d=\sqrt{t^2+t+1}$, and write the supported interval as $[c,u]$.  Then
\[
 c=\frac{t(1-b)}{t+1},
 \qquad
 u=\frac{d-a-tb-1}{t}.
\]
The midpoint condition gives
\[
 a+tb\le d-1-\frac t2.
\]
Eliminating $tb=t-c(t+1)$ gives
\[
 a\le F(t,c):=d-1-\frac{3t}{2}+c(t+1).
\]
For fixed $c\le1/2$, the function $F$ decreases with $t$, so
\[
 a\le F(1,c)=\sqrt3-\frac52+2c=:L(c).
\]
The inequality $2L(c)\le1-M_c^{\rm sup}$ reduces to
\[
 20c^2+(18\sqrt3-52)c+47-24\sqrt3\ge0,
\]
which holds on $[0,1/2]$.  Hence
\[
 a\le1-M_c^{\rm sup}.
 \tag{6.65}
\]

Put $\varepsilon=1-u$.  Direct substitution gives
\[
 \varepsilon=\varepsilon_0+\frac at,
 \qquad
 \varepsilon_0=
 \frac{2t+1-c(t+1)-d}{t}>0.
\]
The map
\[
 z\longmapsto
 \frac{z}{z+\varepsilon_0+z/t}
\]
is increasing, while
$\varepsilon_0+F(t,c)/t=1/2$.  Therefore
\[
 \frac{a}{a+\varepsilon}
 \le
 \frac{F(t,c)}{F(t,c)+1/2}
 \le2L(c)
 \le1-M_c^{\rm sup}.
 \tag{6.66}
\]
Equations (6.65)--(6.66) are the local hypotheses of the common
rescuer-tail lemma.
