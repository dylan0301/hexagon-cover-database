\subsection{Slack-sensitive local capacity bounds}
Write $C_0(a,b)=c_{\max}(a,b)$ and $f(a,b)=1-C_0(a,b)$ for $a+b\le1$.
The subscript in $C_0$ denotes the own-ray capacity, not the actual reach of $T_0$.
Let \(h=\sqrt3/2\). For \(0\le m\le1/2\), define

\[
\ell(m)=1-c(m),\qquad c(m)\in[h,1],
\]

where \(c(m)\) is the unique root in this interval of

\[
F(c,m):=c^4-c^2+mc-m^2=0.
\]

Also set

\[
q(m)=\frac{m(1-m)}2,\qquad b(m)=\frac m2-\frac25m^2.
\]

\begin{lemma}[Own-ray deficit bounds and branchwise concavity]
\label{lem:bc-capacity-tools}
For $a,b\ge0$, $a+b\le1$, and $m=\min(a,b)$,
\[
q(m)\le\ell(m)\le f(a,b)\le m.
\]
For $m\le3/8$, also $b(m)\le\ell(m)$. The function $\ell$ is concave,
$\ell(m)\le m/2$, and $g(m)=m+\ell(m)$ is strictly increasing.
For $a\ge1/2$ and $b\le a$, the two capacity branches meet at $1-a=g(b)$;
each branch of $f$ is concave separately in either coordinate.
\end{lemma}
\begin{proof}
Use the local formula in \zcref{prop:new-exact-local-set}. The bound
$f(a,b)\le\min(a,b)$ and monotonicity follow from
\zcref{lem:new-common-pair-domination}.
\readerheading{Deficit bounds}

For \(a+b\le1\), with \(m=\min(a,b)\),

\[
\boxed{q(m)\le\ell(m)\le f(a,b)\le m.}
\]

Moreover,

\[
\boxed{b(m)\le\ell(m)\quad(0\le m\le3/8).}
\]

The upper bound \(f\le m\) is equivalent to the known own-ray bound \(C_0(a,b)\ge1-m\). To see \(C_0(a,b)\le c(m)\), first suppose \(m\le h/2\). The diagonal pair \((m,m)\) belongs to the quartic cell, because its cell polynomial is \(m^2(16m^2-3)\le0\). Thus \(C_0(a,b)\le C_0(m,m)=c(m)\). If \(m\ge h/2\), antitonicity gives \(C_0(a,b)\le C_0(h/2,h/2)=h\le c(m)\).

For completeness, both lower bounds are certified by substitution in the quartic. For \(v=1/2-m\),

\[
\begin{split}
\frac{1024F(1-q(m),m)}{m^2}
={}&64v^6+64v^5+496v^4+480v^3\\
&+1276v^2+644v+33.
\end{split}
\]

For \(v=3/8-m\),

\[
\begin{split}
\frac{10240000F(1-b(m),m)}{m^2}={}&262144v^6+720896v^5+3174400v^4\\
&+5826560v^3+11428800v^2+9721776v+3049.
\end{split}
\]

All coefficients are positive. The tested values of \(c\) exceed \(h\), and \(F_c=4c^3-2c+m>0\) on \([h,1]\). Hence these nonnegative quartic values prove both displayed lower bounds. At \(m=0\), take the continuous limits.

\readerheading{Concavity and the transition function}

Let \(D_0=4c^3-2c+m\). Implicit differentiation gives

\[
\ell'(m)=\frac{c-2m}{D_0},
\]

\[
\ell''(m)=-\frac{2\{8c^2m(c-m)+c^2-cm+m^2\}}{D_0^3}<0.
\]

Thus \(\ell\) is concave and, since \(\ell(0)=0\) and \(\ell'(0)=1/2\),

\[
\ell(m)\le m/2.
\]

It is increasing on \([0,3/8]\). The function

\[
g(m)=m+\ell(m)
\]

is strictly increasing on \([0,1/2]\), because

\[
g'(m)=\frac{4c^3-c-m}{D_0}>0.
\]

Also,

\[
g(3/8)\ge3/8+b(3/8)=81/160>1/2.
\]

Consequently \(g(z)\le1/2\) implies \(z<3/8\).

For \(a\ge1/2\), \(0\le b\le a\), and \(a+b\le1\), the exact capacity formula can be written

\[
f(a,b)=
\begin{cases}
\ell(b),&1-a\ge g(b),\\
1-a\,j(a+b),&1-a\le g(b),
\end{cases}
\]

where

\[
j(s)=\frac2{1+\sqrt{4s^2-3}}.
\]

The expressions agree at the transition; the second expression is used only in its real selected domain. This branch formula follows from the exact quartic/triangular cell formula. Indeed, its cell polynomial at sum \(s\) is \(F(s,b)\), whose selected zero is \(s=c(b)\).

Each branch of \(f\) is concave separately in either argument while the larger argument remains the first one. For the triangular branch, \(j''(s)>0\). If \(D=\sqrt{4s^2-3}\in(0,1]\), then

\[
2j'(s)+\frac s2j''(s)
=\frac{4s(-2D^3-4D^2+9D+3)}{D^3(1+D)^3}>0.
\]

The numerator bracket is at least \(3D+3\). Since \(a\ge(a+b)/2\), \(\partial_a^2[a\,j(a+b)]>0\); also \(\partial_b^2[a\,j(a+b)]=a j''(a+b)>0\). Negating these derivatives proves the asserted branchwise concavity. The quartic branch is constant in the first argument and concave in the second.

\end{proof}
\begin{lemma}[Slack-sensitive rational envelope]
\label{lem:bc-slack-envelope}
For $a,b\ge0$, $a+b\le1$, $m=\min(a,b)\le3/8$, and $\delta=1-a-b$, put
\[
b(m)=\frac m2-\frac25m^2,\qquad K(m)=\frac{5+4m}{5-4m}.
\]
Then
\[
 f(a,b)\ge\max\{b(m),m-K(m)\delta\}.
\]
The envelope is exact at $\delta=0$ and is not asserted with these coefficients for $m>3/8$.
\end{lemma}
\begin{proof}
Use \zcref{lem:bc-capacity-tools}.
If \(m=0\), the assertion is immediate. Otherwise write \(M=1-m-\delta\) and use the baseline bound \(f(M,m)\ge\ell(m)\ge b(m)\). For \(\delta\ge b(m)\), the affine entry is at most \(b(m)\), so nothing more is needed.

For \(0\le\delta\le b(m)\), we have

\[
M+m=1-\delta\ge1-b(m)\ge1-\ell(m)=c(m).
\]

Thus the selected triangular branch applies. On this interval,

\[
F_m(\delta)=f(1-m-\delta,m)
\]

is concave, by \zcref{lem:bc-capacity-tools}; here \(M\ge m\) since \(m\le3/8\) and \(1-m-b(m)>m\). Its endpoint values satisfy

\[
F_m(0)=m,\qquad F_m(b(m))\ge b(m).
\]

The chord inequality therefore gives

\[
F_m(\delta)\ge m-\frac{m-b(m)}{b(m)}\delta
 =m-\frac{5+4m}{5-4m}\delta.
\]

This proves the rational envelope.

The baseline bound used here can be audited without a root computation. For \(F(c,m)=c^4-c^2+mc-m^2\),

\[
F(1-b(m),m)=\frac{m^2}{10000}H(m),
\]

where

\[
H(m)=256m^6-1280m^5+4960m^4-11600m^3+20625m^2-21000m+5500.
\]

On \([0,3/8]\),

\[
H'(m)\le1536(3/8)^5+19840(3/8)^3+41250(3/8)-21000
 =-286311/64<0,
\]

while \(H(3/8)=3049/1024>0\). Also \(1-b(m)>\sqrt3/2\), where \(F\) increases with \(c\). This proves \(b(m)\le\ell(m)\).

\end{proof}
\begin{lemma}[Coupled nonuniform capacity inequality]
\label{lem:bc-coupled-capacity}
If $0<x\le y<1$ and $x/2\le z\le y$, put
\[
 r=f(1-y,z),\qquad t=f(1-z,x/2).
\]
Then
\[
 2(r+t)>(1-y+r)(2x-y+r).
\]
\end{lemma}
\begin{proof}
Put

\[
u=x/2,\qquad f(a,b)=1-C_0(a,b),
\]

so the domain is

\[
0<u,\quad 2u\le y<1,\quad u\le z\le y,
\]

and the desired positive quantity is

\[
P(y,r,t):=2(r+t)-(1-y+r)(4u-y+r).
\]

Both arguments of each capacity are nonnegative and have sum at most one. In particular, \(0<r,t\le1/2\). At fixed \(u,y\),

\[
P_r=1-4u+2y-2r\ge1-2r\ge0,\qquad P_t=2.
\]

Therefore lower bounds on either radius may be substituted throughout.

Use the local bounds and branchwise concavity of \zcref{lem:bc-capacity-tools} and the rational envelope \zcref{lem:bc-slack-envelope}.

\readerheading{Easy parameter regions}

\readerheading{The region $y\ge1/2$}

Put \(a=1-y\le1/2\). Then \(u\le(1-a)/2\), \(u\le z\le1-a\). Use the radius monotonicity above and the lower bounds the deficit bounds above.

If \(z\ge a\) and \(z\le1-u\), we may use \(r=q(a)\), \(t=q(u)\). The resulting expression is concave in \(u\); its endpoint at \(u=0\) is visibly nonnegative, and its endpoint at \(u=(1-a)/2\) is

\[
\frac{(1-a)^3(1+a)}4>0.
\]

If \(z\ge a\) and \(z\ge1-u\), then \(r,t\ge q(a)\), because \(a\le1-z\le u\le1/2\). After this replacement the expression decreases with \(u\), and at \(u=(1-a)/2\) it equals

\[
\frac{a(1-a)(a^2-a+2)}4>0.
\]

If \(z\le a\), then \(r,t\ge q(u)\). The resulting expression is concave in \(a\) on

\[
u\le a\le\min\{1/2,1-2u\}.
\]

Its possible endpoint values are

\[
\begin{array}{c|c|c}
a& P\text{ after replacement}&\text{range}\\\hline
u&\frac u4(14-43u+14u^2-u^3)&0\le u\le1/3\\
1/2&\frac14(1-17u^2+10u^3-u^4)&0\le u\le1/4\\
1-2u&\frac u4(-u^3+2u^2+9u-2)&1/4\le u\le1/3.
\end{array}
\]

The three polynomial brackets are positive. The first decreases to \(32/27\), the second decreases to \(23/256\), and the third increases from \(23/64\), respectively. This proves the region.

\readerheading{The region $y\le1/2$, $z\ge2u$}

Here \(r\ge q(z)\) and \(t\ge q(u)\). The replaced expression increases with \(y\), so set \(y=z\). It is then concave in \(u\in[0,z/2]\). Its value at zero is positive, and its value at \(u=z/2\) is

\[
\frac{z^3(2-z)}4>0.
\]

\readerheading{Reduction of the remaining region to three transitions}

It remains to consider

\[
0<u\le1/4,\quad u\le z\le2u\le y\le1/2.
\]

The first arguments of both capacities are their larger arguments, so \zcref{lem:bc-capacity-tools} applies throughout.

Fix \(u,z\) and vary \(y\). In the triangular branch, \(r''(y)\le0\), and direct differentiation gives

\[
\frac{d^2P}{dy^2}
=(1-4u+2y-2r)r''-2(1-r')^2\le0.
\]

In the quartic branch \(r\) is constant and \(dP/dy=1+4u+2r-2y>0\). Hence it suffices to check \(y=2u\), the already-proved endpoint \(y=1/2\), or the transition \(y=g(z)\), whenever it belongs to the interval.

On \(y=2u\),

\[
P=r(1-r)+2t-2u(1-2u).
\]

The function \(r(1-r)\) is concave and increasing for \(0\le r\le1/2\). By branchwise concavity of both radii, the displayed relation is concave on each interval in \(z\) separated by \(g(z)=2u\) and \(z=g(u)\). Thus its minimum is at \(z=u\), \(z=2u\), or one of these transitions. At \(z=u\), \(t=u\), so the displayed relation is \(r(1-r)+4u^2>0\). At \(z=2u\), \(r=2u\), so it is \(2t>0\).

On \(y=g(z)\), we have \(r=\ell(z)\) and \(z<3/8\), and

\[
P=2\ell(z)+2t-(1-z)(4u-z).
\]

The allowed interval for \(u\) is \([z/2,g(z)/2]\). This expression is branchwise concave in \(u\), with its only possible switch at \(z=g(u)\). Its endpoints are the already-checked case \(z=2u\) and the first transition below. Therefore only these three configurations remain:

\[
\begin{array}{c|l}
\mathrm A& y=2u=g(z),\ r=\ell(z),\\
\mathrm B& y=2u,\ z=g(u),\ t=\ell(u),\\
\mathrm C& y=g(z),\ z=g(u),\ r=\ell(z),\ t=\ell(u).
\end{array}
\]

This is an exhaustive reduction by concavity, not an assumption that capacities always lie at transitions.

\readerheading{Exact transition checks}

\readerheading{Transition A: one rational bound, no subdivision}

Here

\[
y=2u=z+r,\qquad r=\ell(z),\qquad 0<z<3/8,
\]

and

\[
P=2t-z(1-z-2r).
\]

Apply \zcref{lem:bc-slack-envelope} to the second capacity, whose smaller coordinate is \(u\) and whose slack is \(z-u\):

\[
t\ge u-K(u)(z-u).
\]

Therefore

\[
P\ge 2\{u-K(u)(z-u)\}-z(1-z-2r),\qquad u=(z+r)/2.
\]

This expression increases with \(r\). Indeed, at fixed \(z\),

\[
\frac{d}{du}\{u-K(u)(z-u)\}
 =\frac{10(5-4z)}{(5-4u)^2}>0,
\]

and the remaining contribution to the \(r\)-derivative is \(2z>0\). Replace \(r\) by \(b(z)\). Exact simplification gives

\[
\boxed{
 P\ge
 \frac{z^2(75-230z+100z^2-16z^3)}{5(25-15z+4z^2)}>0.
}
\]

The cubic in the numerator is decreasing on \([0,3/8]\), with value \(63/32>0\) at \(3/8\). This treats the whole transition in one calculation. The former split at \(z=1/4\) and its degree-eight certificate are unnecessary.

\readerheading{Transition B: one rational bound, no subdivision}

Write

\[
v=\ell(u),\qquad z=u+v,\qquad t=v,\qquad0<u\le1/4.
\]

Then

\[
P=r(1-r)+2v-2u(1-2u).
\]

Since \(z\le3u/2\le3/8\), \zcref{lem:bc-slack-envelope} applies to the first capacity and gives

\[
r\ge R(u,v):=u+v-K(u+v)(u-v).
\]

On \(b(u)\le v\le u/2\),

\[
\frac{\partial R}{\partial v}
 =\frac{10(5-8u)}{(5-4u-4v)^2}>0.
\]

Furthermore,

\[
R(u,b(u))=\frac{u(25-80u+16u^2)}{(5-2u)(5-4u)}>0,
\]

and \(R(u,v)\le u+v\le3/8\). Thus substituting \(R\) for \(r\), and then \(b(u)\) for \(v\), can only decrease the expression. We obtain

\[
\boxed{
 P\ge
 \frac{u^2(625+4500u-18400u^2+5440u^3-256u^4)}
 {5(5-2u)^2(5-4u)^2}>0.
}
\]

On \(0\le u\le1/4\), the quartic numerator bracket is at least

\[
625+4500u-4600u-1=624-100u\ge599.
\]

No split at \(u=1/6\) is needed, and both former degree-six and degree-seven certificates are removed.

\readerheading{Transition C}

Again write \(v=\ell(u)\), \(z=u+v\). Now \(r=\ell(z)\), and

\[
P=2(r+v)-(1-z)(4u-z).
\]

We have \(z\le3u/2\le3/8\). First replace \(r\) by \(b(z)\). The resulting expression, as a function of \(v\) with \(z=u+v\), has positive derivative

\[
4+4u-\frac{18}{5}(u+v)>0.
\]

Thus replace \(v\) by \(v_0=b(u)\), and \(z\) by \(z_0=u+v_0\), obtaining

\[
P\ge2\{b(z_0)+v_0\}-(1-z_0)(4u-z_0)
=\frac{u^2}{500}(175+280u-144u^2)>0.
\]

All three transitions are proved. The exhaustive concavity reduction and the easy regions therefore establish the coupled inequality throughout its stated domain.

\end{proof}
