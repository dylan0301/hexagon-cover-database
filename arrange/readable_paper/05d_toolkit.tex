\section{Finite-Enclosure Toolkit and Nonzero-Gap Obstructions}
\label{sec:finite-enclosure}

Method~3 has one contract.  Local boundary and radial demands first produce a
point or compact set missed by all six open V roles.  Coverage then forces the
object into the open C triangle.  The case closes either because its least
enclosing equilateral triangle has side at least one or because one forced
point lies beyond the C-triangle exit on the same ray.

The recurring notation is
\[
 \Lambda(K),\quad c_{\max}(a,b),\quad C_\pm(a,b),\quad
 D_i=(1-\Gamma_i)V_i,\quad J(p,q),\quad K_{\rm tr}.
\]
Here $\Lambda(K)$ is the least enclosing equilateral side, $c_{\max}$ is the
own-ray capacity, $C_\pm$ are the permitted neighboring-ray capacities,
$D_i$ is a type-aware radial witness, and $J(p,q)$ is a complementary
boundary gap.  In the one-gap, one-supercritical all-Vd0 row,
\[
 K_{\rm tr}
 =
 \{O,M_0,X(\ell),X(r),P_2,P_3,P_4\}
\]
is the transverse seven-point witness.  It replaces the former ten-point
notation $K_{410}$: the endpoints $P_0,P_1,P_5$ are not used by the
enclosure certificate.

The chapter is ordered by dependency: endpoint rules, exact local capacities,
witness generation, universal terminals, exceptional cases, and case
dispatch.

\subsection{Endpoint, compactness, and continuity}
\label{subsec:new-endpoint-selector-audit}

\begin{lemma}[Open trace endpoint]
\label{lem:new-open-trace-endpoint}
Let $U$ be an open equilateral triangle and $T=\overline U$.  If
$T\cap[V,O]$ is the closed interval from $V$ to a point $P$, then
$P\notin U$.  If a noncentral point $Q$ of a radial arm belongs to $U$, then
$U$ has positive-length intersection with that arm.
\end{lemma}

\begin{proof}
The first assertion follows because $P$ is a boundary point of the closed
convex intersection.  For the second, a small Euclidean ball about $Q$ lies
in $U$ and cuts the arm in a relative open interval.
\end{proof}

\begin{lemma}[Compact set in an open unit triangle]
\label{lem:new-compact-open-shrink}
If a compact set $K$ is contained in an open unit equilateral triangle, then
\[
 \Lambda(K)<1.
\]
\end{lemma}

\begin{proof}
The distance from $K$ to each side line is positive.  Move all three side
lines inward by a common smaller distance.
\end{proof}

\begin{lemma}[Continuity of the enclosure number]
\label{lem:new-lambda-continuity}
For nonempty compact sets $K,L$,
\[
 |\Lambda(K)-\Lambda(L)|
 \le\frac3h d_H(K,L),
\]
where $d_H$ is the Hausdorff distance.
\end{lemma}

\begin{proof}
For every unit normal,
$|h_K(n)-h_L(n)|\le d_H(K,L)$.  Apply the three-support formula in both
directions.
\end{proof}

These lemmas justify singleton-gap limits and every use of a radial trace
endpoint as a center-forced point.

\subsection{Trace-exact AB envelopes at an actual gap}
\label{subsec:trace-exact-ab-envelopes}

If the actual V-gap on $e_{i,i+1}$ is
\[
 J_i=X_i([\ell,r]),
 \qquad X_i(t)=V_i+t(V_{i+1}-V_i),
\]
then actual maximality gives
\[
 \boxed{\ell=B_i,\qquad r=1-A_{i+1}.}
 \tag{6.24a}
\]
The two incident open traces are
\[
 U_i\cap e_{i,i+1}=X_i([0,\ell)),
 \qquad
 U_{i+1}\cap e_{i,i+1}=X_i((r,1]),
 \tag{6.24b}
\]
so the whole closed interval $J_i$, including a singleton, is center-forced.

For a closed unit V triangle $S$, let $A_i(S),B_i(S)$ be its actual reaches
and define the source-conditioned unions
\[
 \mathcal E_i^{\rightarrow}(a\mid b)
 =
 \bigcup\{S\cap\mathcal C_i:A_i(S)\ge a,\ B_i(S)=b\},
 \tag{6.24c}
\]
\[
 \mathcal E_i^{\leftarrow}(a\mid b)
 =
 \bigcup\{S\cap\mathcal C_i:A_i(S)=a,\ B_i(S)\ge b\}.
 \tag{6.24d}
\]
Their relevant closed edge sections stop and start at the actual endpoints:
\[
 \mathcal E_i^{\rightarrow}(a\mid b)\cap e_{i,i+1}
 =X_i([0,b]),
\]
\[
 \mathcal E_{i+1}^{\leftarrow}(1-r\mid b')\cap e_{i,i+1}
 =X_i([r,1]).
 \tag{6.24e}
\]

\begin{proposition}[Trace-exact AB-envelope interface]
\label{prop:readable-trace-exact-ab-interface}
At an actual gap, the two source-conditioned envelopes contain the incident
closed role footprints and are contained in the corresponding ordinary
AB-unions:
\[
 \mathcal E_i^{\rightarrow}(a\mid b),\quad
 \mathcal E_i^{\leftarrow}(a\mid b)
 \subseteq\mathcal K_i(a,b).
\]
Therefore the ordinary capacities $c_{\max},C_+,C_-$ remain valid upper
bounds, while the full interval between the actual open traces is
center-forced.
\end{proposition}

The operation restricts the source family to triangles having the prescribed

\end{proof}

\subsection{Radial witnesses}

Suppose $T_i=\overline{U_i}$ contains selected boundary anchors with
$a_i\le A_i$ and $b_i\le B_i$.  Put
\[
 \Gamma_i=
 \max\{c_{\max}(a_i,b_i),C_+(a_{i-1},b_{i-1}),
 C_-(a_{i+1},b_{i+1})\},
\]
where neighboring terms forbidden by the actual V types are omitted, and
define
\[
 D_i=(1-\Gamma_i)V_i.
 \tag{6.25}
\]

\begin{lemma}[Type-aware radial forcing]
\label{lem:new-type-aware-radial-forcing}
Every $D_i$ is missed by all six open V roles.  Hence
\[
 \boxed{D_i\in U_C.}
\]
\end{lemma}

\begin{proof}
Membership in $U_i$ would increase the own-radial demand beyond
$c_{\max}$.  Membership in an adjacent role would either create forbidden
positive support or exceed its exact neighboring capacity.  If
$D_i=dV_i$, then
\[
 \|dV_i-V_{i\pm2}\|^2=1+d+d^2>1,
 \qquad
 \|dV_i-V_{i+3}\|=1+d>1,
\]
so the three nonlocal roles are excluded by diameter one.
\end{proof}

\begin{lemma}[Common-pair domination]
\label{lem:new-common-pair-domination}
Let $p,q\ge0$, $p+q\le1$, and put
\[
 c_*=c_{\max}(p,q),\qquad m=\min\{p,q\}.
\]
Then
\[
 c_*\ge1-m,
 \qquad
 C_+(p,q),C_-(p,q)\le1-m\le c_*.
 \tag{6.26}
\]
The same conclusion holds for every actual local pair dominating $(p,q)$.
\end{lemma}

\begin{proof}
Assume $q=m\le p$.  The demand $(p,q,1-q)$ is admissible because its
relevant support side has length
$q+\max\{p,1-q\}=1$, so $c_*\ge1-q$.  Every branch of the exact neighboring
formula is at most $1-q$.  Reflection handles the other order, and
coordinatewise antitonicity handles dominating pairs.
\end{proof}

\subsection{Common pairs supplied by actual gaps}

\begin{lemma}[One-gap common boundary pair]
\label{lem:readable-one-gap-common-pair}
Assume there is exactly one V-gap
\[
 J=[B_0,1-A_1]=[\ell,r]\subset e_{0,1}
\]
and every V role on the remaining path is nonsupercritical.  Put
\[
 p=1-r,\qquad q=\ell.
\]
Then every local boundary pair used in the one-gap argument dominates
$(p,q)$ coordinatewise.
\end{lemma}

\begin{proof}
Choose overlap coordinates $x_i$ on the other five edges.
Nonsupercriticality gives
\[
 x_1\le r,\qquad
 x_i\le x_{i-1}\ (2\le i\le5),\qquad
 x_5\ge\ell.
\]
The selected pairs are $(p,x_1)$ at $T_1$,
$(1-x_{i-1},x_i)$ at $T_i$ for $2\le i\le5$, and $(1-x_5,q)$ at
$T_0$.
\end{proof}

\begin{lemma}[Two-gap CE2 common boundary pair]
\label{lem:readable-two-gap-common-pair}
Suppose both signed CE2 center intervals contain V-gaps, and put
\[
 p=W-\alpha,\qquad q=R-\delta.
\]
If $T_1,\ldots,T_5$ are nonsupercritical, then all five roles dominate
$(p,q)$ coordinatewise.
\end{lemma}

\begin{proof}
The incident endpoint roles give $A_1\ge q$ and $B_5\ge p$.  Handoffs
$x_1,\ldots,x_4$ on the intervening center-free edges satisfy
\[
 p<x_4<x_3<x_2<x_1<1-q.
\]
Reading each incoming and outgoing pair proves the claim.
\end{proof}
