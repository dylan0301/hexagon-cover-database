\subsection{Universal terminal G: complementary gap}

\begin{lemma}[Disk-plus-point formula]
\label{lem:new-disk-point-formula}
Let $\mathcal D_\eta$ be the disk of radius $\eta$ about $O$, and let
$\lVert G\rVert=\rho$.  Then
\[
 \Lambda(\mathcal D_\eta\cup\{G\})=
 \begin{cases}
  3\eta/h,&\rho\le2\eta,\\[2mm]
  \dfrac{3\eta+\sqrt{3(\rho^2-\eta^2)}}{2h},&\rho>2\eta.
 \end{cases}
 \tag{6.27}
\]
\end{lemma}

\begin{proof}
For a unit normal $n$, write
$s_j=\langle G,\mathsf R^jn\rangle$.  Then
$s_0+s_1+s_2=0$ and
$\sum s_j^2=3\rho^2/2$.  The support sum is
$\sum\max\{\eta,s_j\}$.  If $\rho\le2\eta$, orient the point projections as
$\rho/2,\rho/2,-\rho$.  If $\rho>2\eta$, an angular minimum occurs when one
point projection equals $\eta$.  The other active projection is
\[
 \frac{-\eta+\sqrt{3(\rho^2-\eta^2)}}2.
\]
Substitution in Proposition~\ref{prop:new-enclosure-gauge} gives (6.27).
\end{proof}

For $p+q\le1$, let
\[
 J(p,q)=[X(q),X(1-p)]\subset e_{0,1},
 \qquad X(t)=V_0+t(V_1-V_0),
\]
and put
\[
 c_*=c_{\max}(p,q),
 \qquad d=1-c_*.
\]

\begin{theorem}[Complementary-gap enclosure]
\label{thm:new-complementary-gap}
\[
 \boxed{
 \Lambda\left(\mathcal D_{hd}\cup J(p,q)\right)\ge1.
 }
 \tag{6.28}
\]
\end{theorem}

\begin{proof}
Put $m=\min\{p,q\}$.  The farther endpoint of $J(p,q)$ has squared norm
\[
 \rho^2=1-m+m^2.
\]
By Lemma~\ref{lem:new-disk-point-formula}, it is enough to prove
\[
 3c_*(1-c_*)\ge m(1-m).
 \tag{6.29}
\]
Lemma~\ref{lem:new-common-pair-domination} gives $c_*\ge1-m\ge1/2$.
If $c_*\le h$, then the left side of (6.29) is at least
$3h(1-h)>1/4$, while $m(1-m)\le1/4$.

Assume $c_*>h$.  In the exact local support formula, let
$x=\cos|t-\pi/6|$.  The pure radial support gives $x\le h/c_*$.  The two
mixed supports imply
\[
 mx+\frac{c_*}{2}
 \left(x+\sqrt3\sqrt{1-x^2}\right)\le h.
\]
The left side is concave in $x$.  Its value at $x=h$ is greater than $h$
unless $c_*=1-m$, in which case (6.29) is immediate.  Hence its value at
$x=h/c_*$ is at most $h$, giving
\[
 m\le \zeta(c_*):=c_*
 \left(\frac12-\sqrt{c_*^2-\frac34}\right).
\]
A direct calculation gives
\[
\begin{aligned}
&3c_*(1-c_*)-\zeta(c_*)(1-\zeta(c_*))\\
&\quad=
\frac{c_*(1-c_*)}{2}
\left(5-2c_*-2c_*^2+2\sqrt{c_*^2-\frac34}\right)\ge0.
\end{aligned}
\]
Since $0\le m\le \zeta(c_*)<1/2$, this proves (6.29).
\end{proof}

\subsection{Universal terminal S: CE2 short ray}

\begin{theorem}[CE2 short-ray obstruction]
\label{thm:new-ce2-short-ray}
In the strict signed CE2 domain put
\[
 p=W-\alpha,
 \qquad q=R-\delta,
 \qquad e=\min\{\alpha,\delta\}.
\]
Then
\[
 \boxed{c_{\max}(p,q)<1-e.}
 \tag{6.30}
\]
Consequently, whenever both CE2 traces contain V-gaps and the five roles
$T_1,\ldots,T_5$ realize the common pair $(p,q)$, one of the radial witnesses
on $r_2,r_4$ lies beyond the corresponding center exit.
\end{theorem}

\begin{proof}
The CE2 inequalities are
\[
 \alpha+W\delta<P,
 \qquad R\alpha+\delta<P.
\]
Using $RW=\eta+P$, they give
\[
 p,q>\eta+e.
 \tag{6.31}
\]
If $M=\max\{R,W\}$, then
\[
 e<\frac{P}{1+M},
 \qquad
 \frac{\eta}{e}>\frac{1+M}{E}\ge\sqrt3.
\]
Hence
\[
 p,q>(1+\sqrt3)e.
 \tag{6.32}
\]
Moreover
\[
 0<e<e_*:=\frac{2\sqrt3-3}{6}.
 \tag{6.33}
\]
Put $c=1-e$ and
\[
 f_c(t)=c^4-c^2+ct-t^2.
\]
Its roots are
\[
 t_\pm=\frac c2\left(1\pm\sqrt{4c^2-3}\right).
\]
At $t_0=(1+\sqrt3)e$,
\[
 f_{1-e}(t_0)=eB(e),
\]
where
\[
 B(e)=e^3-4e^2-3\sqrt3e+\sqrt3-1.
\]
The function $B$ decreases on $[0,e_*]$ and
\[
 B(e_*)=\frac{302\sqrt3-501}{72}>0.
\]
Thus $t_-<(1+\sqrt3)e<p,q$.  Since
$p+q\le1-2e=c-e$, each of $p,q$ is also below $t_+$.  Therefore
\[
 f_c(p)>0,
 \qquad f_c(q)>0.
 \tag{6.34}
\]
At radial demand $c=1-e$, the four exact support-contact side lengths of
$K(p,q,c)$ are all greater than one: two are $p+c,q+c$, and the other two
exceed one exactly by (6.34).  Thus $c$ is not admissible, proving (6.30).

If $e=\delta$, the point $(1-c_{\max}(p,q))V_2$ lies farther than $\delta$
from $O$, while the center exit on $r_2$ is $\delta$.  If $e=\alpha$, the
analogous point on $r_4$ lies beyond the exit $\alpha$.
\end{proof}
