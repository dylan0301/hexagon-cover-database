\begin{proposition}[CE1 direct reverse-path certificate]
\label{prop:new-ce1-direct-certificate}
Use the signed CE1 center variables
\[
0<R<1,
\qquad w=1-R,
\qquad E=\sqrt{1-Rw},
\]
\[
\eta=1-E,
\qquad P=E(1-E),
\]
and write
\[
 A=\alpha,
 \qquad D=\delta.
\]
Assume
\[
 A>0,
 \qquad D>0,
 \qquad A+wD<P,
 \qquad D+RA\ge P.
 \tag{6.12}
\]
Put
\[
 X=R-D,
 \qquad m=\frac AR,
 \qquad h_0=\frac{\eta+A+D}{2R}.
 \tag{6.13}
\]
Let $T_0$ be the unique supercritical role and
$T_1,\ldots,T_5$ nonsupercritical $\Vdzero$ roles in a one-gap skeleton
cover.  Then
\[
 \boxed{A_0>1-h_0.}
 \tag{6.14}
\]
\end{proposition}

\begin{proof}
The direct path $T_5\to T_4\to T_3\to T_2\to T_1$, its two selected-chord
steps, and the final low-root contradiction are proved in
Appendix~\ref{subsec:calc-ce1-reverse-path}.  No reach map is introduced or
composed.
\end{proof}
